const PROD_CONFIG = {
  SHEET_NAME: "Pomelnic",
  SUMMARY_SHEET: "_Sync_Summary",
  AUDIT_SHEET: "_Sync_Audit_Details",
  ERROR_SHEET: "_Sync_Errors",
  BATCH_SIZE: 100,
  CALENDAR_NASTERE: "DIiH - Zile de nastere",
  CALENDAR_ADORMIRE: "DIiH - Zile de adormire",
  START_HOUR_DEFAULT: 9,
  EVENT_DURATION_HOURS: 1,
  SEND_EMAIL_REPORT: true,
  NIGHTLY_TRIGGER_PROP: "DIiH_NIGHTLY_TRIGGER_ID",
  ACTIVE_RUN_ID_PROP: "DIiH_CURRENT_RUN_ID"
};

/**
 * Funcția principală de producție apelată automat de trigger
 */
function syncPomelnicProduction() {
  executeSyncEngine(false);
}

 * FUNCȚIE DE SIMULARE PENTRU TESTARE MANUALĂ (FĂRĂ AFECTARE TRIGGER ZILNIC)
 * Rulează această funcție direct din editorul Apps Script pentru a testa remedierea!
 */
function simulateNightlySyncTest() {
  Logger.log(">>> PORNIT SIMULARE TEST MANUALA PENTRU SINCRONIZARE CALENDAR <<<");
  executeSyncEngine(true);
}

/**
 * Motorul central de sincronizare
 */
function executeSyncEngine(isSimulation) {
  const startTimeObj = new Date();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(PROD_CONFIG.SHEET_NAME);
  if (!sheet) {
    Logger.log("Foaia " + PROD_CONFIG.SHEET_NAME + " nu a fost găsită!");
    return;
  }

  const props = PropertiesService.getScriptProperties();

  // Generare sau preluare RunID
  let runId = props.getProperty(PROD_CONFIG.ACTIVE_RUN_ID_PROP);
  if (!runId || props.getProperty("LAST_PROD_ROW") === "2" || !props.getProperty("LAST_PROD_ROW")) {
    runId = "RUN-" + Utilities.formatDate(startTimeObj, "Europe/Bucharest", "yyyyMMdd-HHmmss");
    props.setProperty(PROD_CONFIG.ACTIVE_RUN_ID_PROP, runId);
  }

  const calNastere = getOrCreateCalendar(PROD_CONFIG.CALENDAR_NASTERE);
  const calAdormire = getOrCreateCalendar(PROD_CONFIG.CALENDAR_ADORMIRE);
  const totalRows = getLastRowWithContent(sheet);

  let startRow = parseInt(props.getProperty("LAST_PROD_ROW") || "2", 10);
  if (startRow > totalRows) startRow = 2;

  let endRow = Math.min(startRow + PROD_CONFIG.BATCH_SIZE - 1, totalRows);
  const numRowsToFetch = endRow - startRow + 1;
  const rows = sheet.getRange(startRow, 1, numRowsToFetch, 22).getValues();

  let stats = {
    processed: 0,
    created: 0,
    updated: 0,
    skipped: 0,
    errors: 0,
    birthAttempts: 0,
    deathAttempts: 0
  };

  let auditLogs = [];
  let errorLogs = [];
  const currentYear = new Date().getFullYear();

  for (let i = 0; i < rows.length; i++) {
    const row = rows[i];
    const currentRowNumber = startRow + i;

    const rowId = row[0] || currentRowNumber;
    const isActiv = (row[2] === true || row[2] === "TRUE" || row[2] === 1 || row[2] === "1");
    const prenume = (row[4] || "").toString().trim();
    const nume = (row[5] || "").toString().trim();
    const info = (row[6] || "").toString().trim();
    const dataNasteriiRaw = row[8];
    const dataAdormiriiRaw = row[10];
    const stareaSufleteasca = (row[13] || "Viu").toString().trim();
    const grad = (row[14] || "").toString().trim();
    const copii = (row[15] || "").toString().trim();
    const sot = (row[16] || "").toString().trim();
    const parinti = (row[17] || "").toString().trim();
    const apropiati = (row[18] || "").toString().trim();

    if (!isActiv || (!prenume && !nume)) {
      stats.skipped++;
      continue;
    }

    const numeComplet = (prenume + " " + nume).trim();
    const parsedNastere = parseDate(dataNasteriiRaw);
    const parsedAdormire = parseDate(dataAdormiriiRaw);

    if (!parsedNastere && !parsedAdormire) {
      stats.skipped++;
      continue;
    }

    stats.processed++;

    [currentYear, currentYear + 1].forEach(targetYear => {
      // 1. SINCRONIZARE EVENIMENT NAȘTERE
      if (parsedNastere) {
        stats.birthAttempts++;
        try {
          const varstaInLume = targetYear - parsedNastere.year;
          let titlu = "";
          if (stareaSufleteasca === "Adormit" && parsedAdormire) {
            const aniAdormit = targetYear - parsedAdormire.year;
            titlu = `🕯️ ${numeComplet} — (adormit de ${aniAdormit} ani • ar fi avut ${varstaInLume} ani)`;
          } else {
            titlu = `🎂 ${numeComplet} (${varstaInLume} ani)`;
          }

          const syncTag = `BIRTH_${rowId}_${targetYear}`;
          const slotHour = findFreeSlotInCalendar(calNastere, targetYear, parsedNastere.month, parsedNastere.day, syncTag);

          const startDate = new Date(targetYear, parsedNastere.month - 1, parsedNastere.day, slotHour, 0, 0);
          const endDate = new Date(targetYear, parsedNastere.month - 1, parsedNastere.day, slotHour + PROD_CONFIG.EVENT_DURATION_HOURS, 0, 0);
          const dataCalendarFormatted = Utilities.formatDate(startDate, "Europe/Bucharest", "dd.MM.yyyy HH:mm");

          const desc = buildDescription({
            rowId, numeComplet, stare: stareaSufleteasca, grad,
            dataNasterii: parsedNastere.str, dataAdormirii: parsedAdormire ? parsedAdormire.str : "-",
            varstaNastere: varstaInLume, aniAdormit: parsedAdormire ? (targetYear - parsedAdormire.year) : null,
            copii, sot, parinti, apropiati, info
          });

          const res = syncEventWithAudit(calNastere, syncTag, titlu, startDate, endDate, desc, auditLogs, numeComplet, "Naștere", dataCalendarFormatted, runId, currentRowNumber);
          if (res === "created") stats.created++;
          if (res === "updated") stats.updated++;
        } catch (e) {
          stats.errors++;
          errorLogs.push([
            Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss"),
            runId,
            currentRowNumber,
            rowId,
            numeComplet,
            "Naștere",
            targetYear,
            PROD_CONFIG.CALENDAR_NASTERE,
            `BIRTH_${rowId}_${targetYear}`,
            String(dataNasteriiRaw),
            stareaSufleteasca,
            e.name || "Error",
            e.message || String(e),
            e.stack || "-"
          ]);
        }
      }

      // 2. SINCRONIZARE EVENIMENT ADORMIRE (CORECTAT PENTRU DATA DE ADORMIRE)
      if (parsedAdormire && stareaSufleteasca === "Adormit") {
        stats.deathAttempts++;
        try {
          const aniAdormit = targetYear - parsedAdormire.year;
          let varstaLaAdormire = parsedNastere ? (parsedAdormire.year - parsedNastere.year) : "?";
          const titlu = `† ${numeComplet} (adormit de ${aniAdormit} ani | adormit la ${varstaLaAdormire} ani ✝)`;

          const syncTag = `DEATH_${rowId}_${targetYear}`;
          const slotHour = findFreeSlotInCalendar(calAdormire, targetYear, parsedAdormire.month, parsedAdormire.day, syncTag);

          // CORECTAT: parsedAdormire.month - 1 și parsedAdormire.day
          const startDate = new Date(targetYear, parsedAdormire.month - 1, parsedAdormire.day, slotHour, 0, 0);
          const endDate = new Date(targetYear, parsedAdormire.month - 1, parsedAdormire.day, slotHour + PROD_CONFIG.EVENT_DURATION_HOURS, 0, 0);
          const dataCalendarFormatted = Utilities.formatDate(startDate, "Europe/Bucharest", "dd.MM.yyyy HH:mm");

          const desc = buildDescription({
            rowId, numeComplet, stare: "Adormit", grad,
            dataNasterii: parsedNastere ? parsedNastere.str : "-", dataAdormirii: parsedAdormire.str,
            varstaLaAdormire, aniAdormit, copii, sot, parinti, apropiati, info
          });

          const res = syncEventWithAudit(calAdormire, syncTag, titlu, startDate, endDate, desc, auditLogs, numeComplet, "Adormire", dataCalendarFormatted, runId, currentRowNumber);
          if (res === "created") stats.created++;
          if (res === "updated") stats.updated++;
        } catch (e) {
          stats.errors++;
          errorLogs.push([
            Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss"),
            runId,
            currentRowNumber,
            rowId,
            numeComplet,
            "Adormire",
            targetYear,
            PROD_CONFIG.CALENDAR_ADORMIRE,
            `DEATH_${rowId}_${targetYear}`,
            String(dataAdormiriiRaw),
            stareaSufleteasca,
            e.name || "Error",
            e.message || String(e),
            e.stack || "-"
          ]);
        }
      }
    });
  }

  const endTimeObj = new Date();

  // Actualizare acumulatori în proprietățile de script
  let totalProcessed = parseInt(props.getProperty("ACC_PROCESSED") || "0", 10) + stats.processed;
  let totalCreated = parseInt(props.getProperty("ACC_CREATED") || "0", 10) + stats.created;
  let totalUpdated = parseInt(props.getProperty("ACC_UPDATED") || "0", 10) + stats.updated;
  let totalErrors = parseInt(props.getProperty("ACC_ERRORS") || "0", 10) + stats.errors;

  props.setProperty("ACC_PROCESSED", totalProcessed.toString());
  props.setProperty("ACC_CREATED", totalCreated.toString());
  props.setProperty("ACC_UPDATED", totalUpdated.toString());
  props.setProperty("ACC_ERRORS", totalErrors.toString());

  saveAuditLogs(ss, auditLogs);
  saveErrorLogs(ss, errorLogs);
  saveSummaryLog(ss, runId, startTimeObj, endTimeObj, `Rândurile ${startRow}–${endRow} (din util: ${totalRows})`, stats);

  let nextRow = endRow + 1;

  if (nextRow <= totalRows) {
    props.setProperty("LAST_PROD_ROW", nextRow.toString());
    if (!isSimulation) {
      scheduleNextStepSafely();
    }
    Logger.log(`Calup terminat. Rândurile ${startRow}–${endRow}. Următorul pas programat de la rândul ${nextRow}.`);
  } else {
    props.setProperty("LAST_PROD_ROW", "2");
    if (!isSimulation) {
      clearTemporaryTriggersOnly();
    }

    if (PROD_CONFIG.SEND_EMAIL_REPORT) {
      sendDetailedNotificationEmail(ss.getUrl(), {
        runId: runId,
        processed: totalProcessed,
        created: totalCreated,
        updated: totalUpdated,
        errors: totalErrors
      });
    }

    // Resetare acumulatori
    props.deleteProperty("ACC_PROCESSED");
    props.deleteProperty("ACC_CREATED");
    props.deleteProperty("ACC_UPDATED");
    props.deleteProperty("ACC_ERRORS");
    props.deleteProperty(PROD_CONFIG.ACTIVE_RUN_ID_PROP);

    Logger.log(`Sincronizarea bazei a fost finalizată complet! RunID: ${runId}`);
  }
}

/**
 * Programează următorul pas curățând EXCLUSIV triggerele temporare
 */
function scheduleNextStepSafely() {
  clearTemporaryTriggersOnly();
  ScriptApp.newTrigger("syncPomelnicProduction")
    .timeBased()
    .after(60 * 1000)
    .create();
}

/**
 * Șterge doar triggerele temporare, păstrând triggerul recurent de la ora 03:00
 */
function clearTemporaryTriggersOnly() {
  const props = PropertiesService.getScriptProperties();
  const nightlyId = props.getProperty(PROD_CONFIG.NIGHTLY_TRIGGER_PROP);
  const triggers = ScriptApp.getProjectTriggers();

  triggers.forEach(t => {
    if (t.getHandlerFunction() === "syncPomelnicProduction") {
      // Dacă triggerul nu este cel de noapte, este temporar și se șterge
      if (t.getUniqueId() !== nightlyId) {
        try {
          ScriptApp.deleteTrigger(t);
        } catch (e) {}
      }
    }
  });

  ensureNightlyTriggerExists();
}

/**
 * Verifică și garantează existența triggerului nocturn zilnic
 */
function ensureNightlyTriggerExists() {
  const props = PropertiesService.getScriptProperties();
  let nightlyId = props.getProperty(PROD_CONFIG.NIGHTLY_TRIGGER_PROP);
  const triggers = ScriptApp.getProjectTriggers();
  let exists = false;

  triggers.forEach(t => {
    if (t.getHandlerFunction() === "syncPomelnicProduction" && t.getUniqueId() === nightlyId) {
      exists = true;
    }
  });

  if (!exists) {
    setupNightlyTrigger();
  }
}

/**
 * Configurare inițială a triggerului nocturn (ora 03:00)
 */
function setupNightlyTrigger() {
  const props = PropertiesService.getScriptProperties();
  const triggers = ScriptApp.getProjectTriggers();

  triggers.forEach(t => {
    if (t.getHandlerFunction() === "syncPomelnicProduction") {
      ScriptApp.deleteTrigger(t);
    }
  });

  const t = ScriptApp.newTrigger("syncPomelnicProduction")
    .timeBased()
    .atHour(3)
    .everyDays(1)
    .create();

  props.setProperty(PROD_CONFIG.NIGHTLY_TRIGGER_PROP, t.getUniqueId());
  Logger.log(`Trigger nocturn programat pentru ora 03:00 AM (ID: ${t.getUniqueId()})`);
}

function findFreeSlotInCalendar(calendar, year, month, day, myTag) {
  const dayStart = new Date(year, month - 1, day, 0, 0, 0);
  const dayEnd = new Date(year, month - 1, day, 23, 59, 59);
  const events = calendar.getEvents(dayStart, dayEnd);

  let occupiedHours = new Set();
  events.forEach(ev => {
    let desc = ev.getDescription() || "";
    if (!desc.includes(myTag)) occupiedHours.add(ev.getStartTime().getHours());
  });

  let candidateHour = PROD_CONFIG.START_HOUR_DEFAULT;
  while (occupiedHours.has(candidateHour) && candidateHour < 21) {
    candidateHour++;
  }
  return candidateHour;
}

function getLastRowWithContent(sheet) {
  const maxRows = sheet.getLastRow();
  if (maxRows < 2) return 2;
  const names = sheet.getRange(1, 5, maxRows, 2).getValues();
  for (let r = maxRows - 1; r >= 1; r--) {
    let prenume = (names[r][0] || "").toString().trim();
    let nume = (names[r][1] || "").toString().trim();
    if (prenume !== "" || nume !== "") return r + 1;
  }
  return 2;
}

function syncEventWithAudit(cal, tag, newTitle, start, end, newDesc, auditList, numePersoana, tipEv, dataCalendarFormatted, runId, currentRow) {
  const dayStart = new Date(start.getFullYear(), start.getMonth(), start.getDate(), 0, 0, 0);
  const dayEnd = new Date(start.getFullYear(), start.getMonth(), start.getDate(), 23, 59, 59);
  const existing = cal.getEvents(dayStart, dayEnd, { search: tag });
  const timestamp = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");

  if (existing.length > 0) {
    const ev = existing[0];
    const oldTitle = ev.getTitle();
    const oldDesc = ev.getDescription();
    const oldHour = ev.getStartTime().getHours();
    const newHour = start.getHours();

    if (oldTitle !== newTitle || oldDesc !== (newDesc + "\n\nTag: " + tag) || oldHour !== newHour) {
      ev.setTitle(newTitle);
      ev.setDescription(newDesc + "\n\nTag: " + tag);
      ev.setTime(start, end);
      auditList.push([
        timestamp,
        runId,
        currentRow,
        numePersoana,
        tipEv,
        dataCalendarFormatted,
        "ACTUALIZAT",
        oldTitle,
        newTitle,
        `${oldHour}:00`,
        `${newHour}:00`
      ]);
      return "updated";
    }
    return "skipped";
  } else {
    const ev = cal.createEvent(newTitle, start, end, { description: newDesc + "\n\nTag: " + tag });
    ev.addPopupReminder(0);
    auditList.push([
      timestamp,
      runId,
      currentRow,
      numePersoana,
      tipEv,
      dataCalendarFormatted,
      "CREAT",
      "-",
      newTitle,
      "-",
      `${start.getHours()}:00`
    ]);
    return "created";
  }
}

function buildDescription(d) {
  let l = ["━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"];
  l.push(`Sursă: DIiH | Foaie: Pomelnic | RowID: ${d.rowId}`);
  l.push(`Stare: ${d.stare} | Grad/Cler: ${d.grad || "-"}`);
  l.push("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
  l.push(`Nume: ${d.numeComplet}`);
  if (d.dataNasterii && d.dataNasterii !== "-") l.push(`Data nașterii: ${d.dataNasterii}`);
  if (d.dataAdormirii && d.dataAdormirii !== "-") l.push(`Data adormirii: ${d.dataAdormirii}`);
  if (d.varstaLaAdormire) l.push(`Vârstă la adormire: ${d.varstaLaAdormire} ani`);
  if (d.aniAdormit) l.push(`Trecut la Domnul de: ${d.aniAdormit} ani`);
  if (d.varstaNastere) l.push(`Vârstă de la naștere: ${d.varstaNastere} ani`);
  l.push("");
  if (d.info) l.push(`Informații: ${d.info}`);
  if (d.copii && d.copii !== "-") l.push(`Copii: ${d.copii}`);
  if (d.sot && d.sot !== "-") l.push(`Soț / Soție: ${d.sot}`);
  if (d.parinti && d.parinti !== "-") l.push(`Părinți: ${d.parinti}`);
  if (d.apropiati && d.apropiati !== "-") l.push(`Frați, bunici, apropiați: ${d.apropiati}`);
  l.push("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
  return l.join("\n");
}

function parseDate(val) {
  if (!val) return null;
  if (Object.prototype.toString.call(val) === '[object Date]') {
    if (isNaN(val.getTime())) return null;
    return { day: val.getDate(), month: val.getMonth() + 1, year: val.getFullYear(), str: Utilities.formatDate(val, "Europe/Bucharest", "dd.MM.yyyy") };
  }
  const s = val.toString().trim();
  const p = s.split(/[\.\/\-]/);
  if (p.length === 3) {
    const d = parseInt(p[0], 10), m = parseInt(p[1], 10), y = parseInt(p[2], 10);
    if (!isNaN(d) && !isNaN(m) && !isNaN(y)) {
      return { day: d, month: m, year: y, str: `${d < 10 ? '0' + d : d}.${m < 10 ? '0' + m : m}.${y}` };
    }
  }
  return null;
}

function getOrCreateCalendar(name) {
  const c = CalendarApp.getCalendarsByName(name);
  if (c.length > 0) return c[0];
  return CalendarApp.createCalendar(name, { timeZone: "Europe/Bucharest" });
}

function saveAuditLogs(ss, logs) {
  if (logs.length === 0) return;
  let sh = ss.getSheetByName(PROD_CONFIG.AUDIT_SHEET);
  if (!sh) {
    sh = ss.insertSheet(PROD_CONFIG.AUDIT_SHEET);
    sh.appendRow(["Timestamp", "RunID", "Rând Sheet", "Persoană", "Tip Eveniment", "Data Eveniment", "Operațiune", "Titlu Vechi", "Titlu Nou", "Ora Veche", "Ora Nouă"]);
    sh.getRange(1, 1, 1, 11).setFontWeight("bold");
  }
  sh.getRange(sh.getLastRow() + 1, 1, logs.length, logs[0].length).setValues(logs);
}

function saveErrorLogs(ss, errs) {
  if (errs.length === 0) return;
  let sh = ss.getSheetByName(PROD_CONFIG.ERROR_SHEET);
  if (!sh) {
    sh = ss.insertSheet(PROD_CONFIG.ERROR_SHEET);
    sh.appendRow(["Timestamp", "RunID", "Rând Sheet", "RowID", "Persoană", "Tip Eveniment", "An Țintă", "Calendar", "Tag", "Data Brută", "Stare", "Tip Eroare", "Mesaj Eroare", "Stack Trace"]);
    sh.getRange(1, 1, 1, 14).setFontWeight("bold").setFontColor("red");
  }
  sh.getRange(sh.getLastRow() + 1, 1, errs.length, errs[0].length).setValues(errs);
}

function saveSummaryLog(ss, runId, startTime, endTime, rangeText, stats) {
  let sh = ss.getSheetByName(PROD_CONFIG.SUMMARY_SHEET);
  if (!sh) {
    sh = ss.insertSheet(PROD_CONFIG.SUMMARY_SHEET);
    sh.appendRow(["RunID", "Start Time", "End Time", "Durată (s)", "Interval Rânduri", "Procesate", "Create", "Actualizate", "Sărite", "Erori", "Încercări Naștere", "Încercări Adormire"]);
    sh.getRange(1, 1, 1, 12).setFontWeight("bold");
  }

  const startStr = Utilities.formatDate(startTime, "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
  const endStr = Utilities.formatDate(endTime, "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
  const durationSec = Math.round((endTime.getTime() - startTime.getTime()) / 1000);

  sh.appendRow([
    runId,
    startStr,
    endStr,
    durationSec + "s",
    rangeText,
    stats.processed,
    stats.created,
    stats.updated,
    stats.skipped,
    stats.errors,
    stats.birthAttempts,
    stats.deathAttempts
  ]);
}

function sendDetailedNotificationEmail(sheetUrl, stats) {
  const email = Session.getActiveUser().getEmail();
  if (!email) return;

  const hasErrors = stats.errors > 0;
  const subject = hasErrors 
    ? `[DIiH Sync][ATENȚIE: ${stats.errors} ERORI] Sincronizare nocturnă finalizată`
    : `[DIiH Sync][SUCCES TOTAL] Sincronizare nocturnă finalizată fără erori`;

  const body = 
`S-au verificat toate persoanele din Pomelnic (Integral rândurile 2–${getLastRowWithContent(SpreadsheetApp.getActiveSpreadsheet().getSheetByName(PROD_CONFIG.SHEET_NAME))}).

ID Rulare (RunID): ${stats.runId}
Stare: ${hasErrors ? "FINALIZAT CU ERORI (vezi foaia _Sync_Errors)" : "SUCCES INTEGRAL"}

Rezumat cumulativ:
• Persoane valide analizate: ${stats.processed}
• Evenimente noi create: ${stats.created}
• Evenimente actualizate: ${stats.updated}
• Erori înregistrate: ${stats.errors}

Deschide registrul pentru detalii complete de audit:
${sheetUrl}
`;
  MailApp.sendEmail(email, subject, body);
}
