function syncPsaltireToCalendar() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(PSALTIRE_CONFIG.SHEET_NAME);
  if (!sheet || sheet.getLastRow() < 7) {
    setupPsaltireSheet();
    sheet = ss.getSheetByName(PSALTIRE_CONFIG.SHEET_NAME);
  }

  autoExtendPsaltireSheet(sheet);

  const cal = getOrCreateCalendar(PSALTIRE_CONFIG.CALENDAR_NAME);
  const totalRows = sheet.getLastRow();
  const rows = sheet.getRange(7, 1, totalRows - 6, 5).getValues();

  let createdCount = 0;
  let updatedCount = 0;
  let skippedCount = 0;

  rows.forEach((r) => {
    let rawDate = r[0];
    let catismaStr = (r[2] || "").toString();
    let titlu = (r[4] || "").toString();

    let targetDate = parseRowDate(rawDate);
    if (!targetDate) return;

    let dataStr = Utilities.formatDate(targetDate, "Europe/Bucharest", "dd.MM.yyyy");
    let tag = `PSALTIRE_${dataStr}`;

    targetDate.setHours(PSALTIRE_CONFIG.EVENT_HOUR, 0, 0, 0);
    let endDate = new Date(targetDate.getTime() + PSALTIRE_CONFIG.EVENT_DURATION_MIN * 60 * 1000);

    let numMatches = catismaStr.match(/\((\d+)\)/);
    let cNum = numMatches ? parseInt(numMatches[1], 10) : 1;
    let cInfo = CATISME_DETALIATE[cNum];
    let rugaciuniComplete = RUGACIUNI_CATISME[cNum] || "";

    let descriere = 
`━━━━━━━━━━━━━━━━━━━━━━━━━━━━
RÂND CITIRE PSALTIRE
(Doxologia.ro)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Catisma ${cInfo.roman} (${cNum})
Data: ${dataStr} (${r[1]})

${RUGACIUNI_INAINTE_PSALTIRE}

ÎMPĂRȚIRE PE STĂRI (SLAVĂ):
• Starea 1: ${cInfo.s1}
  ⤹ Slavă Tatălui...
• Starea 2: ${cInfo.s2}
  ⤹ Slavă Tatălui...
• Starea 3: ${cInfo.s3}
  ⤹ Slavă Tatălui...

━━━━━━━━━━━━━━━━━━━━━━━━━━━━
RÂNDUIALA DUPĂ CATISMA ${cInfo.roman}
━━━━━━━━━━━━━━━━━━━━━━━━━━━━
${rugaciuniComplete}
━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n\nTag: ${tag}`;

    let existing = cal.getEvents(targetDate, endDate, { search: tag });
    if (existing.length > 0) {
      let ev = existing[0];
      if (ev.getTitle() !== titlu || ev.getDescription() !== descriere) {
        ev.setTitle(titlu);
        ev.setDescription(descriere);
        ev.removeAllReminders();
        updatedCount++;
      } else {
        skippedCount++;
      }
    } else {
      let ev = cal.createEvent(titlu, targetDate, endDate, { description: descriere });
      ev.removeAllReminders();
      createdCount++;
    }
  });

  sheet.getRange("B4").setValue(`Actualizat: ${Utilities.formatDate(new Date(), "Europe/Bucharest", "dd.MM.yyyy HH:mm:ss")} (Create: ${createdCount}, Modificate: ${updatedCount}, Nemodificate: ${skippedCount})`);
  Logger.log(`Psaltire: ${createdCount} create, ${updatedCount} modificate, ${skippedCount} neschimbate.`);
}

function autoExtendPsaltireSheet(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 7) return;

  const lastDateRaw = sheet.getRange(lastRow, 1).getValue();
  const lastDate = parseRowDate(lastDateRaw);
  if (!lastDate) return;

  const today = new Date();
  const daysDiff = Math.ceil((lastDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));

  if (daysDiff < 20) {
    const baseDateRaw = sheet.getRange("B2").getValue();
    const baseDate = parseRowDate(baseDateRaw) || new Date(2026, 8, 22);
    const baseRand = parseInt(sheet.getRange("B3").getValue() || "6", 10);
    const zileNume = ["Duminică", "Luni", "Marți", "Miercuri", "Joi", "Vineri", "Sâmbătă"];

    let newRows = [];
    for (let i = 1; i <= 60; i++) {
      let d = new Date(lastDate.getTime());
      d.setDate(d.getDate() + i);

      let diffFromBase = Math.round((d.getTime() - baseDate.getTime()) / (1000 * 60 * 60 * 24));
      let catismaNum = ((baseRand - 1 + diffFromBase) % 20 + 20) % 20 + 1;
      let cInfo = CATISME_DETALIATE[catismaNum];

      let titlu = `Psaltire: Catisma ${cInfo.roman} [Ps. ${cInfo.display}]`;
      let dataStr = Utilities.formatDate(d, "Europe/Bucharest", "dd.MM.yyyy");
      let ziStr = zileNume[d.getDay()];

      newRows.push([dataStr, ziStr, `Catisma ${cInfo.roman} (${catismaNum})`, cInfo.display, titlu]);
    }
    sheet.getRange(lastRow + 1, 1, newRows.length, 5).setValues(newRows);
  }
}

function setupPsaltireSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(PSALTIRE_CONFIG.SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(PSALTIRE_CONFIG.SHEET_NAME);
  sheet.clear();

  sheet.getRange("A1:E1").setValues([["CONFIGURARE GRUP CITIRE PSALTIRE", "", "", "", ""]]).setFontWeight("bold").setBackground("#d9ead3");
  sheet.getRange("A2:B2").setValues([["Dată pornire serie de referință:", "2026-09-22"]]);
  sheet.getRange("A3:B3").setValues([["Rândul meu în grup (la data de start):", 6]]);
  sheet.getRange("A4:B4").setValues([["Status sincronizare:", "Pregatit de rulare"]]);
  sheet.getRange("A2:A4").setFontWeight("bold");

  const headers = ["Data", "Ziua Săptămânii", "Catisma", "Împărțire pe Stări (Slavă)", "Titlu Calendar"];
  sheet.getRange("A6:E6").setValues([headers]).setFontWeight("bold").setBackground("#cfe2f3");

  const baseDate = new Date(2026, 8, 22);
  const baseRand = 6;
  let rowsData = [];
  const zileNume = ["Duminică", "Luni", "Marți", "Miercuri", "Joi", "Vineri", "Sâmbătă"];

  for (let i = 0; i < PSALTIRE_CONFIG.DAYS_AHEAD; i++) {
    let d = new Date(baseDate.getTime());
    d.setDate(d.getDate() + i);

    let catismaNum = ((baseRand - 1 + i) % 20) + 1;
    let cInfo = CATISME_DETALIATE[catismaNum];
    let titlu = `Psaltire: Catisma ${cInfo.roman} [Ps. ${cInfo.display}]`;
    let dataStr = Utilities.formatDate(d, "Europe/Bucharest", "dd.MM.yyyy");
    let ziStr = zileNume[d.getDay()];

    rowsData.push([dataStr, ziStr, `Catisma ${cInfo.roman} (${catismaNum})`, cInfo.display, titlu]);
  }

  sheet.getRange(7, 1, rowsData.length, 5).setValues(rowsData);
  sheet.autoResizeColumns(1, 5);
}

function parseRowDate(val) {
  if (!val) return null;
  if (Object.prototype.toString.call(val) === '[object Date]') return new Date(val.getTime());
  let s = val.toString().trim();
  let parts = s.split(/[\.\/\-]/);
  if (parts.length === 3) {
    let d = parseInt(parts[0], 10), m = parseInt(parts[1], 10) - 1, y = parseInt(parts[2], 10);
    return new Date(y, m, d);
  }
  return null;
}

function getOrCreateCalendar(name) {
  const c = CalendarApp.getCalendarsByName(name);
  if (c.length > 0) return c[0];
  return CalendarApp.createCalendar(name, { timeZone: "Europe/Bucharest" });
}

function setupPsaltireTrigger() {
  const triggers = ScriptApp.getProjectTriggers();
  triggers.forEach(t => {
    if (t.getHandlerFunction() === "syncPsaltireToCalendar") ScriptApp.deleteTrigger(t);
  });

  ScriptApp.newTrigger("syncPsaltireToCalendar")
    .timeBased()
    .atHour(4)
    .everyDays(1)
    .create();

  Logger.log("Trigger Psaltire configurat pentru ora 04:00 AM.");
}
