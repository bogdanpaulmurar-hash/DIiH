/**

 *
 * Ce face: se loghează în Adservio cu datele părintelui și aduce absențele, calificativele/notele
 * și temele copilului. Absențele și notele se scriu în tab-urile Adservio_Absente / Adservio_Note
 * (rescrise complet la fiecare sincronizare); temele intră în Scoala_Teme (ID „ADS-<id>”, sursa „adservio”),
 * fără să strice bifele „Gata” puse din aplicație.

 *  1. Setări proiect (⚙️) → Proprietăți script → adaugă:
 *       ADSERVIO_USER = emailul/utilizatorul de Adservio
 *       ADSERVIO_PASS = parola de Adservio
 *     (nu se scriu niciodată în sheet și nu ajung în aplicație)
 *  2. Rulează o dată testeazaAdservio() → în Execuții/Jurnal vezi ce a adus.
 *  3. Rulează o dată instaleazaSyncAdservio() → sincronizare automată zilnic la 07:00 și 16:00.
 *
 * v24.42: actualizarea se face din aplicație (Școală → Adservio → „Actualizează din Adservio”).
 *  Dacă Adservio cere cod (2FA), adminul îl scrie o dată în aplicație; serverul ține minte dispozitivul
 *  (ADSERVIO_TFA_COOKIES în Proprietăți script) și creează singur triggerele de la 07:00 și 16:00.

const ADSERVIO_BASE = "https://www.adservio.ro";
const ADS_ABS_SHEET = "Adservio_Absente";
const ADS_NOTE_SHEET = "Adservio_Note";
const ADS_CALIFICATIVE = { 10: "FB", 8: "B", 6: "S", 4: "I" };

function adsCredentials() {
  const p = PropertiesService.getScriptProperties();
  const user = p.getProperty("ADSERVIO_USER");
  const pass = p.getProperty("ADSERVIO_PASS");
  if (!user || !pass) throw new Error("Lipsesc ADSERVIO_USER / ADSERVIO_PASS din Proprietăți script.");
  return { user: user, pass: pass };
}

// Extrage cookie-urile „nume=valoare” din antetele Set-Cookie ale unui răspuns
function adsCookiesFrom(resp, jar) {
  const h = resp.getAllHeaders();
  let sc = h["Set-Cookie"] || h["set-cookie"] || [];
  if (!Array.isArray(sc)) sc = [sc];
  sc.forEach(c => {
    const first = String(c).split(";")[0];
    const i = first.indexOf("=");
    if (i > 0) jar[first.substring(0, i).trim()] = first.substring(i + 1).trim();
  });
  return jar;
}

function adsCookieHeader(jar) {
  return Object.keys(jar).map(k => k + "=" + jar[k]).join("; ");
}

// Login: POST /api/v2/auth (uaUserName, uaPassword, tokenLifetime). Întoarce sesiunea (cookie-uri + token).
// opts (v24.42, opțional): { tfaOptID, tfaResponse, tfaSave } — pasul al doilea, cu codul de verificare (2FA).
// Erorile au marcaje pentru aplicație: err.tfaRequired (cere cod), err.badCode (cod greșit), err.blocked (reCAPTCHA).
function adsLogin(opts) {
  opts = opts || {};
  const cred = adsCredentials();
  const jar = {};
  // cookie-urile inițiale ale site-ului (unele servere cer sesiunea înainte de login)
  try { adsCookiesFrom(UrlFetchApp.fetch(ADSERVIO_BASE + "/ro/auth", { muteHttpExceptions: true, followRedirects: false }), jar); } catch (e) {}
  adsTrustedCookies(jar);

  // Exact câmpurile trimise de aplicația Adservio din browser (intent=auth, setCookie=1 → sesiune prin cookie)
  const fields = {
    intent: "auth", uaUserName: cred.user, uaPassword: cred.pass, setCookie: "1",
    tfaSave: opts.tfaSave ? "true" : "false", tfaOptID: opts.tfaOptID || "", tfaResponse: opts.tfaResponse || "",
    uaNewPassword: "", uaNewPassword2: "", tokenLifetime: "30", recaptcha: ""
  };
  const resp = UrlFetchApp.fetch(ADSERVIO_BASE + "/api/v2/auth", {
    method: "post",
    contentType: "application/x-www-form-urlencoded",
    payload: fields,
    headers: { Accept: "application/json", Cookie: adsCookieHeader(jar) },
    muteHttpExceptions: true,
    followRedirects: false
  });
  const code = resp.getResponseCode();
  const text = resp.getContentText();
  const fresh = adsCookiesFrom(resp, {});
  Object.assign(jar, fresh);
  let js = {};
  try { js = JSON.parse(text); } catch (e) {}
  if (code === 200 && js.atToken) {
    if (opts.tfaOptID && opts.tfaSave) adsSaveTrusted(js, fresh);
    return { jar: jar, token: js.atToken, uaID: js.uaID, mode: null };
  }
  const errs = Array.isArray(js) ? js : (js && Array.isArray(js.errors) ? js.errors : []);
  const has = (field, c) => errs.some(x => x && x.errorField === field && (c === undefined || String(x.errorCode) === String(c)));
  let err;
  if (has("recaptcha")) {
    err = new Error("Adservio cere verificarea „nu sunt robot”, pe care serverul nu o poate face. Folosește marcajul din Adservio.");
    err.blocked = true;
  } else if (opts.tfaOptID && has("tfaResponse")) {
    err = new Error("Codul nu e bun sau a expirat. Scrie-l din nou sau cere altul.");
    err.badCode = true;
  } else if (errs.some(x => x && String(x.errorCode) === "107")) {
    err = new Error("Adservio cere codul de verificare (2FA). Deschide Școală → Adservio în aplicație și apasă „Actualizează din Adservio”.");
    err.tfaRequired = true;
  } else {
    err = new Error("Login Adservio eșuat (HTTP " + code + ": " + text.substring(0, 200) + "). Verifică utilizatorul și parola din Proprietăți script.");
  }
  throw err;
}

// v24.42 — „dispozitivul de încredere”: cookie-urile primite după login cu cod și „ține minte”.
// Stau doar în Proprietăți script (ADSERVIO_TFA_COOKIES); ADSERVIO_TFA_TOKEN e varianta veche, păstrată.
function adsTrustedCookies(jar) {
  const p = PropertiesService.getScriptProperties();
  const old = p.getProperty("ADSERVIO_TFA_TOKEN");
  if (old) jar.tfaToken = old;
  try { Object.assign(jar, JSON.parse(p.getProperty("ADSERVIO_TFA_COOKIES") || "{}")); } catch (e) {}
  return jar;
}

function adsSaveTrusted(js, fresh) {
  const keep = {};
  Object.keys(fresh).forEach(k => { if (/tfa|trust|device/i.test(k)) keep[k] = fresh[k]; });
  if (js.tfaToken) keep.tfaToken = js.tfaToken;
  if (!Object.keys(keep).length) Object.assign(keep, fresh); // numele cookie-ului nu e documentat: le păstrăm pe toate
  const p = PropertiesService.getScriptProperties();
  p.setProperty("ADSERVIO_TFA_COOKIES", JSON.stringify(keep));
  p.setProperty("ADSERVIO_TFA_SALVAT", authNow());
}

// Variantele de cod ale contului: 1 = SMS, 2 = e-mail, 3 = aplicație de autentificare (ca pe site).
function adsTfaOptions() {
  const cred = adsCredentials();
  const jar = adsTrustedCookies({});
  const body = { _method: "GET", uaUserName: cred.user, uaPassword: cred.pass };
  const tries = [
    { contentType: "application/json", payload: JSON.stringify(body) },
    { contentType: "application/x-www-form-urlencoded", payload: body }
  ];
  let last = "";
  for (let i = 0; i < tries.length; i++) {
    const resp = UrlFetchApp.fetch(ADSERVIO_BASE + "/api/v2/auth/tfa_options", Object.assign({
      method: "post", headers: { Accept: "application/json", Cookie: adsCookieHeader(jar) }, muteHttpExceptions: true, followRedirects: false
    }, tries[i]));
    last = "HTTP " + resp.getResponseCode() + ": " + resp.getContentText().substring(0, 200);
    if (resp.getResponseCode() !== 200) continue;
    let js = null;
    try { js = JSON.parse(resp.getContentText()); } catch (e) {}
    const list = adsData(js).filter(o => o && o.tfaOptID);
    if (list.length) return list.map(o => ({ id: String(o.tfaOptID), type: String(o.tfaOptType), hint: String(o.tfaOptHint || "") }));
  }
  throw new Error("Nu am putut afla cum trimite Adservio codul (" + last + ").");
}

// Cere trimiterea codului pe SMS / e-mail (POST /api/v2/auth/tfa_challenge, ca pe site).
function adsTfaSend(optId) {
  const cred = adsCredentials();
  const resp = UrlFetchApp.fetch(ADSERVIO_BASE + "/api/v2/auth/tfa_challenge", {
    method: "post", contentType: "application/x-www-form-urlencoded",
    payload: { uaUserName: cred.user, uaPassword: cred.pass, tfaOptID: String(optId || "") },
    headers: { Accept: "application/json", Cookie: adsCookieHeader(adsTrustedCookies({})) },
    muteHttpExceptions: true, followRedirects: false
  });
  const c = resp.getResponseCode();
  if (c < 200 || c >= 300) throw new Error("Adservio nu a trimis codul (HTTP " + c + "). Încearcă din nou peste un minut.");
  return true;
}

// Corp multipart/form-data construit manual (pentru cazul în care serverul acceptă doar FormData)
function adsMultipart(fields) {
  const boundary = "----DIiH" + Utilities.getUuid().replace(/-/g, "");
  let body = "";
  Object.keys(fields).forEach(k => {
    body += "--" + boundary + "\r\nContent-Disposition: form-data; name=\"" + k + "\"\r\n\r\n" + fields[k] + "\r\n";
  });
  body += "--" + boundary + "--\r\n";
  return { boundary: boundary, body: body };
}

// Cerere GET autentificată. Încearcă pe rând: cookie-uri, Bearer token, cookie cu token; reține varianta care merge.
function adsGet(sess, path) {
  const modes = sess.mode ? [sess.mode] : ["cookie", "bearer", "tokencookie"];
  let last = "";
  for (let i = 0; i < modes.length; i++) {
    const headers = { Accept: "application/json" };
    const jarCopy = Object.assign({}, sess.jar);
    if (modes[i] === "bearer") headers.Authorization = "Bearer " + sess.token;
    if (modes[i] === "tokencookie") jarCopy.atToken = sess.token;
    if (Object.keys(jarCopy).length) headers.Cookie = adsCookieHeader(jarCopy);
    const resp = UrlFetchApp.fetch(ADSERVIO_BASE + path, { headers: headers, muteHttpExceptions: true, followRedirects: false });
    const code = resp.getResponseCode();
    if (code === 200) {
      sess.mode = modes[i];
      return JSON.parse(resp.getContentText());
    }
    last = "HTTP " + code;
  }
  throw new Error("Adservio a refuzat cererea " + path.split("?")[0] + " (" + last + ").");
}

function adsData(js) {
  return (js && Array.isArray(js.data)) ? js.data : (Array.isArray(js) ? js : []);
}

function adsDay(s) {
  return s ? String(s).substring(0, 10) : "";
}

function adsStripHtml(s) {
  return String(s || "").replace(/<br\s*\/?>/gi, "\n").replace(/<\/p>/gi, "\n").replace(/<[^>]+>/g, "")
    .replace(/&nbsp;/g, " ").replace(/&amp;/g, "&").replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/&quot;/g, "\"")
    .replace(/\n{2,}/g, "\n").trim();
}

function adsPersonName(p) {
  return p ? [p.uaFirstName, p.uaLastName].filter(Boolean).join(" ") : "";
}

// Copilul (sau copiii) contului de părinte, cu clasa din anul școlar activ
function adsChildren(sess) {
  const kids = [];
  const parentId = sess.uaID;
  try {
    const el = adsGet(sess, "/api/v2/parinti/elev?epID=" + encodeURIComponent(parentId));
    (Array.isArray(el) ? el : [el]).forEach(e => { if (e && e.elevID) kids.push({ elevID: String(e.elevID), name: e.elevPrenume || e.elevNume || "Copil" }); });
  } catch (e) {}
  if (!kids.length) {
    const acc = adsGet(sess, "/api/v2/accounts/adservio?uaID=" + encodeURIComponent(parentId) + "&withNonLinked=true");
    (Array.isArray(acc) ? acc : []).forEach(a => { if (a.elev && a.elev.elevID) kids.push({ elevID: String(a.elev.elevID), name: a.elev.elevPrenume || "Copil" }); });
  }
  kids.forEach(k => {
    const cls = adsData(adsGet(sess, "/api/v2/elevi/clase?elevID=" + k.elevID));
    const active = cls.find(c => c.asc && String(c.asc.ascIsActiv) === "1") || cls[0] || {};
    k.clID = active.clID || "";
    k.ascID = active.asc ? active.asc.ascID : "";
    k.className = active.clNume || "";
  });
  return kids;
}

function adsFetchChild(sess, k) {
  const abs = adsData(adsGet(sess, "/api/v2/absente?elevID=" + k.elevID + "&cuMaterie=1&ascID=" + k.ascID + "&_limit=-1"));
  const rep = adsGet(sess, "/api/v2/elevi/repartizare?elevID=" + k.elevID + "&clID=" + k.clID + "&cuNote=1&cuMedii=1&cuActivitate=1");
  const hw = adsData(adsGet(sess, "/api/v2/teme?elevID=" + k.elevID + "&cuProfesor=1&cuMaterie=1&cuNota=1&cuRating=1&cuHwsSubmitted=1&_limit=9999"));

  const absente = abs.map(a => ({
    id: String(a.absID),
    copil: k.name,
    data: adsDay(a.absData),
    materie: a.materie ? a.materie.matNume : "",
    motivata: !!(a.absMotivareData || a.absMotivareCadruID),
    modul: a.absSanID || "",
    profesor: adsPersonName(a.cadruInfo),
    obs: a.absObs || ""
  }));

  const note = [];
  (Array.isArray(rep) ? rep : []).forEach(m => {
    const materie = m.cmcMaterie ? m.cmcMaterie.matNume : "";
    (m.note || []).forEach(n => {
      const scale = n.type && n.type.scale;
      const v = n.notaValoare;
      note.push({
        id: String(n.notaID),
        copil: k.name,
        data: adsDay(n.notaData || n.notaDataInsert),
        materie: materie,
        valoare: v,
        afisare: scale === "calificative" ? (ADS_CALIFICATIVE[v] || String(v)) : String(v),
        tip: n.type ? n.type.name : "",
        modul: n.notaSanID || "",
        obs: n.notaObs || ""
      });
    });
  });

  // doar temele din anul școlar curent (din 1 septembrie); cele vechi ar apărea ca „întârziate”
  const now = new Date();
  const startAn = (now.getMonth() >= 8 ? now.getFullYear() : now.getFullYear() - 1) + "-09-01";
  const teme = hw.filter(h => adsDay(h.hwDueDate || h.hwInsertDate) >= startAn).map(h => ({
    id: "ADS-" + h.hwID,
    child: k.name,
    subject: h.materie ? h.materie.matNume : "",
    text: (h.hwName || "Temă") + (h.hwDescription ? " — " + adsStripHtml(h.hwDescription) : ""),
    givenDate: adsDay(h.hwInsertDate),
    dueDate: adsDay(h.hwDueDate),
    submitted: !!h.hwsSubmitted
  }));

  return { absente: absente, note: note, teme: teme };
}

function adsWriteSheet(ss, name, header, rows) {
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.setFrozenRows(1);
  }
  sh.clearContents();
  sh.getRange(1, 1, 1, header.length).setValues([header]).setFontWeight("bold");
  sh.getRange("A:Z").setNumberFormat("@");
  if (rows.length) sh.getRange(2, 1, rows.length, header.length).setValues(rows);
}

// Temele Adservio intră în Scoala_Teme. O temă bifată „Gata” în aplicație rămâne Gata.
function adsMergeTeme(ss, teme) {
  const sh = getOrCreateSchoolTemeSheet(ss);
  const now = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
  const last = sh.getLastRow();
  const existing = {};
  if (last >= 2) {
    sh.getRange(2, 1, last - 1, 12).getValues().forEach((r, i) => { if (r[0]) existing[String(r[0])] = { row: i + 2, status: r[6], doneDate: r[7], created: r[9] }; });
  }
  let added = 0;
  teme.forEach(t => {
    const ex = existing[t.id];
    const status = t.submitted ? "Gata" : (ex && ex.status === "Gata" ? "Gata" : "De făcut");
    const doneDate = status === "Gata" ? ((ex && ex.doneDate) || Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd")) : "";
    const row = [t.id, t.child, t.subject, t.text.substring(0, 1500), t.givenDate, t.dueDate, status, doneDate, "adservio", (ex && ex.created) || now, now, "adservio"];
    if (ex) sh.getRange(ex.row, 1, 1, row.length).setValues([row]);
    else { sh.appendRow(row); added++; }
  });
  return added;
}

// Sincronizarea completă. Folosită de trigger, de butonul din aplicație și de testeazaAdservio().
// v24.42: serverul face aceleași cereri ca marcajul și folosește aceeași interpretare (adsHandleRawPush),
// deci tab-urile Adservio_* primesc aceleași coloane, indiferent de unde vin datele.
function adservioSync(motiv, loginOpts) {
  const ss = getSpreadsheet();
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(30000)) throw new Error("O actualizare Adservio e deja în curs.");
  try {
    const sess = adsLogin(loginOpts);
    const calls = adsServerCalls(sess);
    const rez = adsHandleRawPush(ss, { v: 2, calls: calls, sursa: motiv || "server" });
    if (!rez.success) throw new Error(rez.error || "Interpretarea datelor a eșuat.");
    return rez;
  } catch (err) {
    setSettingValue(ss.getSheetByName("Setari"), "adservio_eroare", authNow() + " — " + (err.message || String(err)));
    logToAppSheet(ss, { modul: "Adservio", actiune: "sincronizare", status: "EROARE", detalii: (err.message || String(err)) + "; " + motiv, user: "-" });
    throw err;
  } finally {
    lock.releaseLock();
  }
}

// Cererile marcajului (adservio-marcaj.js), făcute din server. Întoarce lista {kind, path, status, ctx, json}.
function adsServerCalls(sess) {
  const calls = [];
  const one = (kind, path, ctx) => {
    try {
      const js = adsGet(sess, path);
      calls.push({ kind: kind, path: path, status: 200, ctx: ctx || {}, json: js });
      return js;
    } catch (e) {
      calls.push({ kind: kind, path: path, status: 0, ctx: ctx || {}, error: String(e.message || e) });
      return null;
    }
  };
  // mai multe cereri deodată, după ce știm ce fel de autentificare merge (sess.mode)
  const many = list => {
    const reqs = list.map(x => {
      const headers = { Accept: "application/json" };
      const jarCopy = Object.assign({}, sess.jar);
      if (sess.mode === "bearer") headers.Authorization = "Bearer " + sess.token;
      if (sess.mode === "tokencookie") jarCopy.atToken = sess.token;
      headers.Cookie = adsCookieHeader(jarCopy);
      return { url: ADSERVIO_BASE + x.path, headers: headers, muteHttpExceptions: true, followRedirects: false };
    });
    UrlFetchApp.fetchAll(reqs).forEach((r, i) => {
      let js = null;
      try { js = JSON.parse(r.getContentText()); } catch (e) {}
      calls.push({ kind: list[i].kind, path: list[i].path, status: r.getResponseCode(), ctx: list[i].ctx || {}, json: js });
    });
  };
  const ids = (x, key, out) => {
    if (!x || typeof x !== "object") return out;
    if (Array.isArray(x)) { x.forEach(v => ids(v, key, out)); return out; }
    Object.keys(x).forEach(k => {
      const v = x[k];
      if (k === key && v !== null && typeof v !== "object") { if (out.indexOf(String(v)) === -1) out.push(String(v)); }
      else ids(v, key, out);
    });
    return out;
  };

  one("profil", "/api/v2/utilizatori/profile");
  const par = one("parinti", "/api/v2/parinti/elev?epID=" + encodeURIComponent(sess.uaID));
  const elevi = ids([par], "elevID", []);
  if (!elevi.length) throw new Error("Nu am găsit niciun elev asociat contului.");

  const now = new Date();
  const y1 = now.getMonth() >= 8 ? now.getFullYear() : now.getFullYear() - 1;
  const azi = Utilities.formatDate(now, "Europe/Bucharest", "yyyy-MM-dd");
  elevi.forEach(e => {
    const c = { elevID: e };
    const clase = adsList(one("clase", "/api/v2/elevi/clase?elevID=" + e + "&cuCiclu=1", c));
    const batch = [
      { kind: "detalii", path: "/api/v2/elevi/detalii?elevID=" + e + "&withAccessLevel=1", ctx: c },
      { kind: "teme", path: "/api/v2/teme?elevID=" + e + "&cuProfesor=1&cuMaterie=1&cuNota=1&cuRating=1&cuHwsSubmitted=1&_limit=9999", ctx: c }
    ];
    clase.forEach(cl => {
      cl = cl || {};
      const clID = cl.clID, ascID = cl.ascID || (cl.asc && cl.asc.ascID);
      if (!clID || !ascID) return;
      const activ = !!(cl.asc && String(cl.asc.ascIsActiv) === "1");
      const ctx = { elevID: e, clID: String(clID), ascID: String(ascID), clasa: cl.clNume || "", an: (cl.asc && cl.asc.ascNume) || "", activ: activ };
      batch.push(
        { kind: "perioade", path: "/api/v2/ani/periods?_limit=-1&sanAscID=" + ascID, ctx: ctx },
        { kind: "repartizare", path: "/api/v2/elevi/repartizare?elevID=" + e + "&clID=" + clID + "&cuNote=1&cuMedii=1&cuActivitate=1", ctx: ctx },
        { kind: "repartizareAbsente", path: "/api/v2/elevi/repartizare?elevID=" + e + "&clID=" + clID + "&cuAbsente=1", ctx: ctx },
        { kind: "medii", path: "/api/v2/medii/elev?elevID=" + e + "&clID=" + clID, ctx: ctx },
        { kind: "absente", path: "/api/v2/absente?elevID=" + e + "&cuMaterie=1&ascID=" + ascID + "&_limit=-1", ctx: ctx },
        { kind: "scutiri", path: "/api/v2/exemptions?clID=" + clID + "&elevID=" + e + "&ascID=" + ascID + "&_limit=-1&_order=%3EcreatedAt", ctx: ctx },
        { kind: "avertismente", path: "/api/v2/avertismente?elevID=" + e + "&ascID=" + ascID + "&_limit=-1&_order=%3EavtData", ctx: ctx }
      );
      if (activ) batch.push(
        { kind: "vacante", path: "/api/v2/ani/vacations?_limit=-1&ascID=" + ascID, ctx: ctx },
        { kind: "orarZi", path: "/api/v2/timetables/day?uaID=" + e + "&date=" + azi + "&clID=" + clID + "&withLessonPlan=1&_limit=-1", ctx: ctx }
      );
    });
    for (let i = 0; i < batch.length; i += 20) many(batch.slice(i, i + 20));
  });
  many([{ kind: "sarbatori", path: "/api/v2/holidays?startDate=" + y1 + "-09-01&endDate=" + (y1 + 1) + "-08-31" }]);
  return calls;
}

// v24.42 — ACTUALIZARE DIN APLICAȚIE (butonul „Actualizează din Adservio”), cu pasul pentru cod (2FA)
// Codul ajunge la părinte (SMS / e-mail / aplicație de autentificare); aplicația îl trimite doar către Adservio.
// După „Ține minte”, Adservio nu mai cere cod, iar triggerele de la 07:00 și 16:00 merg singure.
// =====================================================================================================
function adsTfaPick(options) {
  return options.find(o => o.type === "2") || options.find(o => o.type === "1") || options.find(o => o.type === "3") || options[0];
}

function adsResult(ss, rez, extra) {
  return Object.assign({ success: true, rezultat: rez, adservio: getAdservioData(ss) }, extra || {});
}

// Butonul din aplicație (scoala/edit). Dacă e nevoie de cod: adminul primește variantele și codul e cerut automat.
function adservioConnect(ss, auth) {
  const cine = (auth && auth.profile && auth.profile.email) || "-";
  try {
    return adsResult(ss, adservioSync("din aplicație (" + cine + ")"));
  } catch (err) {
    if (err.blocked) return { success: false, blocked: true, error: err.message };
    if (!err.tfaRequired) return { success: false, error: err.message || String(err) };
    if (!auth || !auth.profile || auth.profile.rol !== "admin") {
      return { success: false, tfaRequired: true, adminOnly: true, error: "Adservio cere un cod de verificare. Îl poate introduce doar administratorul, o singură dată." };
    }
    try {
      const options = adsTfaOptions();
      const pick = adsTfaPick(options);
      let sent = false, sendError = "";
      if (pick && pick.type !== "3") { try { sent = adsTfaSend(pick.id); } catch (e2) { sendError = e2.message; } }
      return { success: false, tfaRequired: true, options: options, chosen: pick ? pick.id : "", sent: sent, sendError: sendError };
    } catch (e3) {
      return { success: false, tfaRequired: true, options: [], error: e3.message };
    }
  }
}

// Doar admin: cere (din nou) codul pe varianta aleasă
function adservioTfaSendAction(optId) {
  try { return { success: true, sent: adsTfaSend(optId) }; }
  catch (err) { return { success: false, error: err.message || String(err) }; }
}

// Doar admin: login cu codul, „ține minte” dispozitivul, apoi actualizarea completă și triggerele zilnice
function adservioTfaVerify(ss, auth, optId, code, trust) {
  const cine = (auth && auth.profile && auth.profile.email) || "-";
  try {
    const rez = adservioSync("din aplicație, cu cod (" + cine + ")", { tfaOptID: String(optId || ""), tfaResponse: String(code || "").trim(), tfaSave: trust !== "0" });
    let programat = false;
    try { programat = adsEnsureTriggers(); } catch (e) {}
    return adsResult(ss, rez, { tinutMinte: trust !== "0", programat: programat });
  } catch (err) {
    return { success: false, badCode: !!err.badCode, blocked: !!err.blocked, tfaRequired: !!err.tfaRequired, error: err.message || String(err) };
  }
}

// ADSERVIO PRIN BROWSER (v24.7) — datele vin din sesiunea Adservio deja autentificată a părintelui,
// trimise de marcajul (bookmarklet) din browser. Fără parolă pe server, fără a atinge 2FA.
// Se autorizează cu ADSERVIO_PUSH_KEY (Proprietăți script). Payload (text/plain, JSON): {absente, note, teme}.
// =====================================================================================================
function adservioHandlePush(ss, e) {
  const key = PropertiesService.getScriptProperties().getProperty("ADSERVIO_PUSH_KEY");
  if (!key) return { success: false, error: "Lipsește ADSERVIO_PUSH_KEY din Proprietăți script. Rulează aratamCheiaPush() o dată." };
  const given = (e && e.parameter && e.parameter.key) || "";
  if (given !== key) return { success: false, error: "Cheie invalidă." };

  // Corpul vine ca text (fetch) sau, dacă site-ul blochează fetch, din formularul de rezervă (câmpul „payload”)
  let payload = {};
  try {
    const raw = (e.parameter && e.parameter.payload) || (e.postData && e.postData.contents) || "";
    payload = JSON.parse(raw);
  } catch (err) { return { success: false, error: "Date invalide: " + err }; }

  // v2 (din v24.19): marcajul trimite răspunsurile brute; interpretarea se face aici
  if (Array.isArray(payload.calls)) return adsHandleRawPush(ss, payload);

  // v1 (marcajul vechi): {absente, note, teme} deja interpretate
  const absente = Array.isArray(payload.absente) ? payload.absente : [];
  const note = Array.isArray(payload.note) ? payload.note : [];
  const teme = Array.isArray(payload.teme) ? payload.teme : [];
  adsWriteParsed(ss, { absente: absente, note: note, teme: teme, medii: [], scutiri: [], avertismente: [], vacante: [] });
  const temeNoi = adsMergeTeme(ss, teme);
  const rez = { success: true, absente: absente.length, note: note.length, teme: teme.length, temeNoi: temeNoi };
  adsMarkDone(ss, rez, "marcaj vechi");
  return rez;
}

// Rulează O DATĂ din editor: creează cheia de trimitere (dacă nu există) și o afișează în jurnal.
function aratamCheiaPush() {
  const props = PropertiesService.getScriptProperties();
  let key = props.getProperty("ADSERVIO_PUSH_KEY");
  if (!key) { key = Utilities.getUuid().replace(/-/g, ""); props.setProperty("ADSERVIO_PUSH_KEY", key); }
  Logger.log("ADSERVIO_PUSH_KEY = " + key);
  return key;
}

// ADSERVIO v2 (v24.19) — interpretarea răspunsurilor brute trimise de marcaj (adservio-marcaj.js din aplicație)
// Câmpurile Adservio nu sunt documentate: fiecare valoare se caută după mai multe nume posibile, iar
// tabul Adservio_Raw păstrează mostre din fiecare răspuns, ca interpretarea să poată fi corectată pe date reale.
// Documentele (ex. scutirea medicală la o absență) rămân în Adservio: păstrăm doar linkul spre ele.
// =====================================================================================================
const ADS_RAW_SHEET = "Adservio_Raw";
const ADS_MEDII_SHEET = "Adservio_Medii";
const ADS_EXTRA_SHEET = "Adservio_Extra";
const ADS_ABS_HEADER = ["ID", "Copil", "Data", "Materie", "Motivata", "Modul", "Profesor", "Observatii", "Motivare", "DataMotivare", "Document", "AnScolar"];
const ADS_NOTE_HEADER = ["ID", "Copil", "Data", "Materie", "Valoare", "Afisare", "Tip", "Modul", "Observatii", "DataPrimirii", "Profesor", "Perioada", "AnScolar"];
const ADS_TIP_NOTA = { NT_TEST: "Test", NT_ORAL: "Oral", NT_PROIECT: "Proiect", NT_TEZA: "Teză", NT_LUCRARE: "Lucrare", NT_TEMA: "Temă", NT_ACTIVITATE: "Activitate" };
const ADS_TIP_SCUTIRE = { medical: "Medical", parental: "Cererea părintelui", school: "Școala", sport: "Sport" };

function adsList(js) {
  if (js && Array.isArray(js.data)) return js.data;
  if (Array.isArray(js)) return js;
  if (js && js.data && typeof js.data === "object") return [js.data];
  return js && typeof js === "object" ? [js] : [];
}

function adsTextOf(v) {
  return v === null || v === undefined ? "" : String(v);
}

// „san_modul_1” → „Modulul 1”; altfel numele perioadei așa cum vine
function adsPeriodName(n) {
  const m = /modul_?(\d+)/i.exec(n || "");
  if (m) return "Modulul " + m[1];
  const s = /sem(estrul)?_?(\d+)/i.exec(n || "");
  return s ? "Semestrul " + s[2] : adsTextOf(n);
}

// Media (0 = încă nu există). În anul școlar curent, o medie neîncheiată (…IsLocked ≠ 1) e „provizorie”.
function adsMedie(oficial, calculat, incheiata, anCurent) {
  const v = (oficial && String(oficial) !== "0") ? oficial : ((calculat && String(calculat) !== "0") ? calculat : "");
  if (!v) return null;
  return { v: v, eticheta: anCurent && String(incheiata) !== "1" ? " (provizorie)" : "" };
}

// Calificativele (sistemul SN_CALIFICATIVE) vin ca 10/8/6/4
function adsDisplayGrade(v, scale) {
  if (v === "" || v === null || v === undefined) return "";
  return scale === "calificative" ? (ADS_CALIFICATIVE[Number(v)] || String(v)) : String(v);
}

// Răspunsurile brute de la marcaj (adservio-marcaj.js), cu structura verificată pe date reale (2026-10-19)
function adsHandleRawPush(ss, payload) {
  const calls = payload.calls.filter(c => c && typeof c === "object");
  adsWriteRawSamples(ss, calls);
  const byKind = k => calls.filter(c => c.kind === k);

  const names = {};
  byKind("parinti").concat(byKind("detalii")).forEach(c => adsList(c.json).forEach(el => {
    if (el && el.elevID && el.elevPrenume) names[String(el.elevID)] = String(el.elevPrenume);
  }));
  const childName = ctx => (ctx && names[ctx.elevID]) || "Copil";
  // Anul școlar al fiecărei clase (ex. „2026/2027”), din lista claselor elevului
  const yearsByAsc = {};
  byKind("clase").forEach(c => adsList(c.json).forEach(cl => { if (cl.asc && cl.asc.ascID) yearsByAsc[String(cl.asc.ascID)] = adsTextOf(cl.asc.ascNume); }));
  const yearOf = ctx => String((ctx && (ctx.an || yearsByAsc[ctx.ascID])) || "").replace("/", "-");

  const periods = {};
  byKind("perioade").forEach(c => adsList(c.json).forEach(pr => { if (pr.sanID) periods[String(pr.sanID)] = adsPeriodName(pr.sanNume); }));
  const periodOf = id => periods[String(id || "")] || "";

  // Scutirile: motivul și documentul absențelor din intervalul lor (dateStart–dateEnd)
  const extra = [];
  const scutiri = [];
  byKind("scutiri").forEach(c => adsList(c.json).forEach(x => {
    const files = (x.files || []).map(f => f.fsUrl || (f.fsID ? ADSERVIO_BASE + "/api/v2/fisiere/view?fsID=" + f.fsID : "")).filter(Boolean);
    const tip = ADS_TIP_SCUTIRE[x.type] || adsTextOf(x.type) || "Scutire";
    const materie = x.subject ? adsTextOf(x.subject.matNume || x.subject.name || x.subject) : "";
    const sc = { elevID: c.ctx.elevID, start: adsDay(x.dateStart), end: adsDay(x.dateEnd) || adsDay(x.dateStart), tip: tip,
      detalii: adsStripHtml(x.details), files: files, motiveaza: String(x.justifyAbsences) === "1" };
    scutiri.push(sc);
    extra.push({ tip: "Scutire", copil: childName(c.ctx), anScolar: yearOf(c.ctx), data: sc.start, sfarsit: sc.end,
      titlu: tip + (materie ? " — " + materie : "") + (String(x.isFullYear) === "1" ? " (tot anul)" : ""),
      text: sc.detalii + (x.justifiedAbsences ? " · " + x.justifiedAbsences + " absențe motivate" : ""), link: files[0] || "" });
  }));

  const absente = [];
  byKind("absente").forEach(c => adsList(c.json).forEach(a => {
    const data = adsDay(a.absData);
    const motivata = String(a.absStatus) === "2" || !!a.absMotivareData;
    const sc = motivata ? scutiri.find(x => x.elevID === c.ctx.elevID && x.start && data >= x.start && data <= x.end) : null;
    absente.push({
      id: adsTextOf(a.absID), copil: childName(c.ctx), data: data,
      materie: a.materie ? adsTextOf(a.materie.matNume) : "",
      motivata: motivata, modul: periodOf(a.absSanID), profesor: adsPersonName(a.cadruInfo), obs: adsStripHtml(a.absObs),
      motivare: sc ? sc.tip + (sc.detalii ? ": " + sc.detalii : "") : "",
      dataMotivare: adsDay(a.absMotivareData),
      document: sc && sc.files.length ? sc.files[0] : "",
      anScolar: yearOf(c.ctx)
    });
  }));

  const note = [];
  const medii = [];
  byKind("repartizare").forEach(c => adsList(c.json).forEach(m => {
    const materie = m.cmcMaterie ? adsTextOf(m.cmcMaterie.matNume) : "";
    let scale = "";
    (Array.isArray(m.note) ? m.note : []).forEach(n => {
      scale = (n.type && n.type.scale) || scale;
      const tipCod = n.type ? adsTextOf(n.type.name) : "";
      note.push({
        id: adsTextOf(n.notaID), copil: childName(c.ctx), data: adsDay(n.notaData || n.notaDataInsert), materie: materie,
        valoare: n.notaValoare, afisare: adsDisplayGrade(n.notaValoare, n.type && n.type.scale),
        tip: adsTextOf(n.ntNume) || ADS_TIP_NOTA[tipCod] || tipCod.replace(/^NT_/, "").toLowerCase(),
        modul: periodOf(n.notaSanID),
        obs: [adsStripHtml(n.notaObs), adsStripHtml(n.notaExtraInfo)].filter(Boolean).join(" · "),
        dataPrimirii: adsTextOf(n.notaDataInsert).substring(0, 16),
        profesor: adsPersonName(n.cadru) || adsPersonName(m.cmcCadru),
        anScolar: yearOf(c.ctx)
      });
    });
    // Plusuri / minusuri de activitate la oră (contoare pe materie, fără dată)
    const plus = parseInt(m.e2mAoPlus, 10) || 0, minus = parseInt(m.e2mAoMinus, 10) || 0;
    if (plus || minus) medii.push({ copil: childName(c.ctx), materie: materie, perioada: "Activitate", medie: (plus ? "+" + plus : "") + (plus && minus ? " / " : "") + (minus ? "−" + minus : ""), anScolar: yearOf(c.ctx) });
    // Mediile pe materie: doar cele încheiate (valoare diferită de 0)
    const md = m.medii || {};
    (md.semestriale || md.perioade || []).forEach(x => {
      const mv = adsMedie(x.msmValoare, x.msmValoareAds, x.msmIsLocked, c.ctx.activ);
      if (mv) medii.push({ copil: childName(c.ctx), materie: materie, perioada: periodOf(x.sanID) + mv.eticheta, medie: adsDisplayGrade(mv.v, scale || "calificative"), anScolar: yearOf(c.ctx) });
    });
    if (md.anuala) {
      const mv = adsMedie(md.anuala.mgmValoare, md.anuala.mgmValoareAds, md.anuala.mgmIsLocked, c.ctx.activ);
      if (mv) medii.push({ copil: childName(c.ctx), materie: materie, perioada: "Anual" + mv.eticheta, medie: adsDisplayGrade(mv.v, scale || "calificative"), anScolar: yearOf(c.ctx) });
    }
  }));
  // Media generală (pe toate materiile)
  byKind("medii").forEach(c => {
    const md = (c.json && c.json.data) || c.json || {};
    (md.semestriale || md.perioade || []).forEach(x => {
      const mv = adsMedie(x.mgsValoare, x.mgsValoareAds, x.mgsIsLocked, c.ctx.activ);
      if (mv) medii.push({ copil: childName(c.ctx), materie: "Media generală", perioada: periodOf(x.sanID) + mv.eticheta, medie: String(mv.v), anScolar: yearOf(c.ctx) });
    });
    if (md.anuala) {
      const mv = adsMedie(md.anuala.mgaValoare, md.anuala.mgaValoareAds, md.anuala.mgaIsLocked, c.ctx.activ);
      if (mv) medii.push({ copil: childName(c.ctx), materie: "Media generală", perioada: "Anual" + mv.eticheta, medie: String(mv.v), anScolar: yearOf(c.ctx) });
    }
  });

  byKind("avertismente").forEach(c => adsList(c.json).forEach(x => extra.push({
    tip: "Avertisment", copil: childName(c.ctx), anScolar: yearOf(c.ctx), data: adsDay(x.avtData), sfarsit: "",
    titlu: adsTextOf(x.avtTitlu || x.avtTip || "Avertisment"), text: adsStripHtml(x.avtText || x.avtMotiv || x.avtObs), link: ""
  })));
  byKind("vacante").forEach(c => adsList(c.json).forEach(x => extra.push({
    tip: "Vacanță", copil: "", anScolar: yearOf(c.ctx), data: adsDay(x.v2pStartDate), sfarsit: adsDay(x.v2pEndDate),
    titlu: adsTextOf(x.v2pName) || "Vacanță", text: "", link: ""
  })));
  byKind("sarbatori").forEach(c => adsList(c.json).forEach(x => extra.push({
    tip: "Zi liberă", copil: "", anScolar: "", data: adsDay(x.holDate), sfarsit: adsDay(x.holDate),
    titlu: adsTextOf(x.holName) || "Zi liberă", text: "", link: ""
  })));

  // Temele: doar din anul școlar curent (cele vechi ar apărea ca întârziate)
  const now = new Date();
  const startAn = (now.getMonth() >= 8 ? now.getFullYear() : now.getFullYear() - 1) + "-09-01";
  const teme = [];
  byKind("teme").forEach(c => adsList(c.json).forEach(h => {
    if (!h.hwID || adsDay(h.hwDueDate || h.hwInsertDate) < startAn) return;
    teme.push({
      id: "ADS-" + h.hwID, child: childName(c.ctx), subject: h.materie ? adsTextOf(h.materie.matNume) : "",
      text: (h.hwName || "Temă") + (h.hwDescription ? " — " + adsStripHtml(h.hwDescription) : ""),
      givenDate: adsDay(h.hwInsertDate), dueDate: adsDay(h.hwDueDate), submitted: !!h.hwsSubmitted
    });
  }));

  adsWriteParsed(ss, { absente: absente, note: note, medii: medii, extra: extra });
  const temeNoi = adsMergeTeme(ss, teme);
  const rez = { success: true, cereri: calls.length, absente: absente.length, note: note.length, medii: medii.length, extra: extra.length, teme: teme.length, temeNoi: temeNoi };
  adsMarkDone(ss, rez, payload.sursa || "marcaj v2");
  return rez;
}

function adsWriteParsed(ss, d) {
  const absente = (d.absente || []).slice().sort((a, b) => String(b.data).localeCompare(String(a.data)));
  const note = (d.note || []).slice().sort((a, b) => String(b.data).localeCompare(String(a.data)));
  adsWriteSheet(ss, ADS_ABS_SHEET, ADS_ABS_HEADER, absente.map(a => [String(a.id), a.copil || "", a.data || "", a.materie || "",
    a.motivata ? "DA" : "NU", a.modul || "", a.profesor || "", a.obs || "", a.motivare || "", a.dataMotivare || "", a.document || "", a.anScolar || ""]));
  adsWriteSheet(ss, ADS_NOTE_SHEET, ADS_NOTE_HEADER, note.map(n => [String(n.id), n.copil || "", n.data || "", n.materie || "", String(n.valoare),
    n.afisare || "", n.tip || "", n.modul || "", n.obs || "", n.dataPrimirii || "", n.profesor || "", n.perioada || n.modul || "", n.anScolar || ""]));
  if (d.medii) adsWriteSheet(ss, ADS_MEDII_SHEET, ["Copil", "Materie", "Perioada", "Medie", "AnScolar"],
    d.medii.map(m => [m.copil, m.materie, m.perioada, m.medie, m.anScolar]));
  if (d.extra) adsWriteSheet(ss, ADS_EXTRA_SHEET, ["Tip", "Copil", "Data", "Sfarsit", "Titlu", "Text", "Link", "AnScolar"],
    d.extra.sort((a, b) => String(b.data).localeCompare(String(a.data))).map(x => [x.tip, x.copil, x.data, x.sfarsit, x.titlu, String(x.text || "").substring(0, 2000), x.link, x.anScolar]));
}

// Mostre din fiecare răspuns (primele 2 elemente), ca să vedem câmpurile reale. Se rescrie la fiecare trimitere.
function adsWriteRawSamples(ss, calls) {
  const rows = calls.map(c => {
    const items = adsList(c.json);
    const keys = {};
    items.slice(0, 20).forEach(it => { if (it && typeof it === "object") Object.keys(it).forEach(k => { keys[k] = 1; }); });
    let sample = c.error ? "EROARE: " + c.error : JSON.stringify(items.slice(0, 2));
    if (sample.length > 45000) sample = sample.substring(0, 45000) + "…";
    return [c.kind || "", String(c.path || "").substring(0, 300), String(c.status || ""), items.length, Object.keys(keys).join(", ").substring(0, 5000), sample];
  });
  adsWriteSheet(ss, ADS_RAW_SHEET, ["Tip", "Cerere", "Status", "Elemente", "Chei", "Mostra (primele 2)"], rows);
}

function adsMarkDone(ss, rez, sursa) {
  const sh = ss.getSheetByName("Setari");
  setSettingValue(sh, "adservio_ultim", authNow() + " — " + rez.absente + " absențe, " + rez.note + " note" +
    (rez.medii !== undefined ? ", " + rez.medii + " medii" : "") + ", " + rez.teme + " teme (" + sursa + ")");
  setSettingValue(sh, "adservio_eroare", "");
  logToAppSheet(ss, { modul: "Adservio", actiune: /marcaj/.test(sursa) ? "push_browser" : "sincronizare", status: "OK", detalii: JSON.stringify(rez) + "; " + sursa, user: "-" });
}
