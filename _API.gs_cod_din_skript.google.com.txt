const BACKEND_VERSION = "24.23";
const SPREADSHEET_ID = "107cZOxLdMMTuOsytriNmZt_d3FM0Nwz1z2ykzpnCcfY";

function getSpreadsheet() {
  try {
    return SpreadsheetApp.openById(SPREADSHEET_ID);
  } catch (e) {
    return SpreadsheetApp.getActiveSpreadsheet();
  }
}

function doGet(e) {
  const ss = getSpreadsheet();
  const action = e.parameter.action;

  // 0. AUTENTIFICARE: login Google / cod de asociere sunt publice; restul cere o sesiune Google validă
  //    și dreptul pe modulul acțiunii (din v23.3 tokenul vechi nu mai e acceptat)
  if (AUTH_PUBLIC_ACTIONS.indexOf(action) !== -1) return authHandleAction(ss, e, action, null);
  const auth = authenticateRequest(ss, e);
  if (!auth.ok) return returnResponse(e, { error: auth.error, authRequired: true });
  if (!authIsActionAllowed(auth, action)) return returnResponse(e, { error: "Nu ai drept pentru această acțiune.", forbidden: true });
  if (action && action.indexOf("auth") === 0) return authHandleAction(ss, e, action, auth);

  // 0b. Familie.gs (v24.9): centru de notificări, calendarul familiei, sfinții zilei
  if (typeof familieHandleAction === "function") {
    const famResp = familieHandleAction(ss, e, action, auth);
    if (famResp) return famResp;
  }

  // 1. AUDIT AVANSAT ÎN LOG_Sistem
  if (action === "logEvent") {
    logToSystemSheet(ss, {
      categorie: e.parameter.category || "AUDIT_APP",
      runId: e.parameter.runId || "-",
      sursa: e.parameter.deviceName || e.parameter.deviceId || "App",
      modul: e.parameter.entity || "General",
      actiune: e.parameter.appAction || "-",
      status: e.parameter.status || "INFO",
      detalii: e.parameter.message || "",
      eroare: e.parameter.errorMessage || "",
      stack: e.parameter.stack || ""
    });
    return returnResponse(e, { success: true });
  }

  // 1b. AUDIT LA NIVEL DE APLICAȚIE ÎN LOG_app (separat de LOG_Sistem; folosit ex. la revenirea unei plăți pe neachitat)
  if (action === "logAppEvent") {
    logToAppSheet(ss, {
      modul: e.parameter.modul || "General",
      actiune: e.parameter.actiune || "-",
      status: e.parameter.status || "INFO",
      detalii: e.parameter.detalii || "",
      user: e.parameter.user || "-"
    });
    return returnResponse(e, { success: true });
  }

  // 2. SETĂRI GLOBALE (Inclusiv Număr Persoane Pomelnic Zilnic, Iconiță & Configurare Partajare)
  if (action === "updateGlobalSettings") {
    let sh = ss.getSheetByName("Setari");
    if (!sh) {
      sh = ss.insertSheet("Setari");
      sh.appendRow(["Cheie", "Valoare", "UltimaModificare"]);
      sh.getRange("A1:C1").setFontWeight("bold");
    }
    if (e.parameter.appTitle) setSettingValue(sh, "app_title", e.parameter.appTitle);
    if (e.parameter.appLogo) setSettingValue(sh, "app_logo", e.parameter.appLogo);
    if (e.parameter.appIconPath) setSettingValue(sh, "app_icon_path", e.parameter.appIconPath);
    if (e.parameter.pomelnicBatchSize) setSettingValue(sh, "pomelnic_batch_size", e.parameter.pomelnicBatchSize);
    if (e.parameter.psaltireBatchSize) setSettingValue(sh, "psaltire_batch_size", e.parameter.psaltireBatchSize);
    if (e.parameter.randuialaTrackerJson) setSettingValue(sh, "randuiala_tracker_json", e.parameter.randuialaTrackerJson);
    return returnResponse(e, { success: true });
  }

  // 2b. SETARE / CITIRE CHEIE ARBITRARĂ ÎN Setari (folosit de funcții noi: ModelCitire activ, iconiță/logo per profil,
  //     calendar școlar per an, repere zoom, date Săptămâna Verde/Altfel) — fără să modifice cheile cunoscute existente
  if (action === "setSettingRaw") {
    let sh = ss.getSheetByName("Setari");
    if (!sh) {
      sh = ss.insertSheet("Setari");
      sh.appendRow(["Cheie", "Valoare", "UltimaModificare"]);
      sh.getRange("A1:C1").setFontWeight("bold");
    }
    // Cheile auth_* (ex. auth_mode) se schimbă doar prin acțiunile de autentificare
    if (e.parameter.key && e.parameter.key.indexOf("auth_") === 0) return returnResponse(e, { success: false, error: "Cheie rezervată." });
    if (e.parameter.key) setSettingValue(sh, e.parameter.key, e.parameter.value || "");
    return returnResponse(e, { success: true });
  }

  if (action === "getSettingsRaw") {
    return returnResponse(e, { settingsRaw: getAllSettingsRaw(ss) });
  }

  // Returnează o imagine salvată în Drive (calendar școlar, iconiță/logo custom) ca base64, pentru <img src="data:...">
  if (action === "getDriveImage") {
    return returnResponse(e, getDriveImageAsBase64(ss, e.parameter.fileId));
  }

  // 3. SALVARE ACTIVITĂȚI ȘCOALĂ (Păstrează structura de 14 coloane a foii existente + stocare metadate în Col 13
  //    + coloane NOI O/P: Data_Achitat, Comentariu_Plata)
  if (action === "saveSchoolActivities") {
    let sh = ss.getSheetByName("Scoala_Activitati") || ss.getSheetByName("Activitati_Scoala");
    if (!sh) {
      sh = ss.insertSheet("Scoala_Activitati");
      sh.appendRow(["ID", "Data", "Copil", "Institutie", "Titlu", "TipPlata", "Ore", "TarifOra", "TotalPlata", "StatusPlata", "TermenPlata", "MesajZi", "RemindereJson", "DataCreare"]);
      sh.getRange("A1:N1").setFontWeight("bold");
    }
    ensureSheetColumns(sh, ["Data_Achitat", "Comentariu_Plata"]);
    const activities = JSON.parse(e.parameter.data || "[]");
    const dataCreare = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");

    activities.forEach(item => {
      deleteRowById(sh, item.id, 1);
      // Păstrăm metadatele extinse (culoare, ore început/sfârșit, autor) în Col 13 (RemindereJson)
      const metaObj = {
        reminders: item.reminders || [60, 30, 10],
        color: item.color || "auto",
        startHour: item.startHour || "08:00",
        endHour: item.endHour || "16:00",
        author: item.author || "Bogdan"
      };

      sh.appendRow([
        item.id,
        item.date,
        item.child || "Copil",
        item.institution || "-",
        item.title,
        item.paymentType || "Pe oră",
        item.hours || 1,
        item.rate || 0,
        item.total || 0,
        item.paymentStatus || "În așteptare",
        item.deadline || "-",
        item.dayMessage || "",
        JSON.stringify(metaObj),
        dataCreare,
        item.dataAchitat || "",
        item.comentariuPlata || ""
      ]);
    });
    return returnResponse(e, { success: true, count: activities.length });
  }

  if (action === "deleteSchoolActivity") {
    let sh = ss.getSheetByName("Scoala_Activitati") || ss.getSheetByName("Activitati_Scoala");
    if (sh && e.parameter.id) deleteRowById(sh, e.parameter.id, 1);
    return returnResponse(e, { success: true });
  }

  // Actualizare în calup status plată (pentru restanțe rezolvate direct din sumar)
  if (action === "batchUpdateSchoolStatus") {
    let sh = ss.getSheetByName("Scoala_Activitati") || ss.getSheetByName("Activitati_Scoala");
    const ids = JSON.parse(e.parameter.ids || "[]");
    const newStatus = e.parameter.status || "Achitat";
    // dataAchitat/comentariuPlata: completate la achitare; golite (string gol) la revenire la neachitat
    const dataAchitat = e.parameter.dataAchitat !== undefined ? e.parameter.dataAchitat : null;
    const comentariuPlata = e.parameter.comentariuPlata !== undefined ? e.parameter.comentariuPlata : null;
    if (sh && ids.length > 0) {
      const cols = ensureSheetColumns(sh, ["Data_Achitat", "Comentariu_Plata"]);
      const maxRows = sh.getLastRow();
      if (maxRows >= 2) {
        const idCol = sh.getRange(2, 1, maxRows - 1, 1).getValues();
        for (let i = 0; i < idCol.length; i++) {
          if (ids.indexOf(String(idCol[i][0]).trim()) !== -1) {
            sh.getRange(i + 2, 10).setValue(newStatus); // Col J = StatusPlata
            if (dataAchitat !== null) sh.getRange(i + 2, cols["Data_Achitat"]).setValue(dataAchitat);
            if (comentariuPlata !== null) sh.getRange(i + 2, cols["Comentariu_Plata"]).setValue(comentariuPlata);
          }
        }
      }
    }
    return returnResponse(e, { success: true });
  }

  // 4. CRUD FOAIA Psaltire_Grup
  if (action === "addPsaltirePerson") {
    let sh = ss.getSheetByName("Psaltire_Grup");
    if (!sh) {
      sh = ss.insertSheet("Psaltire_Grup");
      sh.appendRow(["Sectiune", "Subgrup_Coloana", "Numar_Rand", "Catisma", "Persoana_Raw", "Exclus_La_Pomenire"]);
      sh.getRange("A1:F1").setFontWeight("bold");
    }
    sh.appendRow([
      e.parameter.sectiune || "Grup Psaltire",
      e.parameter.subgrup || "Coloana 1",
      parseInt(e.parameter.numarRand, 10) || sh.getLastRow(),
      e.parameter.catisma || "",
      e.parameter.nume,
      e.parameter.exclus === "true" ? "DA" : "NU"
    ]);
    return returnResponse(e, { success: true });
  }

  if (action === "updatePsaltirePerson") {
    let sh = ss.getSheetByName("Psaltire_Grup");
    let rowId = parseInt(e.parameter.rowId, 10);
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      if (e.parameter.nume) sh.getRange(rowId, 5).setValue(e.parameter.nume);
      if (e.parameter.sectiune) sh.getRange(rowId, 1).setValue(e.parameter.sectiune);
      if (e.parameter.subgrup) sh.getRange(rowId, 2).setValue(e.parameter.subgrup);
      if (e.parameter.catisma) sh.getRange(rowId, 4).setValue(e.parameter.catisma);
      if (e.parameter.exclus !== undefined) sh.getRange(rowId, 6).setValue(e.parameter.exclus === "true" ? "DA" : "NU");
      return returnResponse(e, { success: true });
    }
  }

  if (action === "deletePsaltirePerson") {
    let sh = ss.getSheetByName("Psaltire_Grup");
    let rowId = parseInt(e.parameter.rowId, 10);
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      sh.deleteRow(rowId);
      return returnResponse(e, { success: true });
    }
  }

  // 5. NOTIȚE & GÂNDURI (Suport pentru tab separat "Notite" sau fallback pe "Ganduri")
  if (action === "addGand" || action === "addNote" || action === "addNotita") {
    let sh = ss.getSheetByName("Notite") || ss.getSheetByName("Ganduri");
    if (!sh) {
      sh = ss.insertSheet("Notite");
      sh.appendRow(["ID", "Gand", "Sursa", "DataCreare"]);
      sh.getRange("A1:D1").setFontWeight("bold");
    }
    const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
    const dataCreare = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
    const noteContent = e.parameter.gand || e.parameter.nota || e.parameter.notita || e.parameter.text || "";
    const noteCols = ensureSheetColumns(sh, NOTE_OWNER_COLUMNS);
    // Fiecare notiță are un proprietar (profilul care o vede). Doar adminul poate crea notițe pentru altcineva.
    let proprietar = auth.profile.email;
    if (auth.profile.rol === "admin" && e.parameter.proprietar && authFindProfile(ss, e.parameter.proprietar)) {
      proprietar = e.parameter.proprietar.toString().trim().toLowerCase();
    }
    const writeNoteOwner = () => {
      const r = sh.getLastRow();
      sh.getRange(r, noteCols["Proprietar"]).setValue(proprietar);
      sh.getRange(r, noteCols["CreatDe"]).setValue(auth.profile.email);
      // Text simplu: altfel Sheets transformă „2026-10-18T20:00” în dată, în fusul registrului, și ora se decala
      sh.getRange(r, noteCols["Reminder"]).setNumberFormat("@").setValue((e.parameter.reminder || "").toString());
    };
    if (header === "ID") {
      const gandId = e.parameter.id || ("NOTE-" + Date.now());
      sh.appendRow([
        gandId,
        noteContent,
        e.parameter.sursa || "Personal",
        dataCreare,
        e.parameter.user || e.parameter.author || ""
      ]);
      writeNoteOwner();
    } else {
      sh.appendRow([
        noteContent,
        e.parameter.sursa || "Personal",
        dataCreare,
        e.parameter.user || e.parameter.author || ""
      ]);
      writeNoteOwner();
    }
    const newNoteRow = sh.getLastRow();
    if (typeof familieOnNoteSaved === "function") familieOnNoteSaved(ss, auth, { rowId: newNoteRow, text: noteContent, proprietar: proprietar, reminder: (e.parameter.reminder || "").toString(), nou: true });
    return returnResponse(e, { success: true, rowId: newNoteRow });
  }

  if (action === "updateGand" || action === "updateNote" || action === "updateNotita") {
    let sh = ss.getSheetByName("Notite") || ss.getSheetByName("Ganduri");
    let rowId = parseInt(e.parameter.rowId, 10);
    const noteContent = e.parameter.gand || e.parameter.nota || e.parameter.notita || e.parameter.text || "";
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      if (!noteRowAllowed(ss, sh, rowId, auth.profile)) return returnResponse(e, { success: false, forbidden: true, error: "Notița nu îți aparține." });
      const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
      const colOffset = (header === "ID") ? 1 : 0;
      sh.getRange(rowId, 1 + colOffset).setValue(noteContent);
      if (e.parameter.sursa) sh.getRange(rowId, 2 + colOffset).setValue(e.parameter.sursa);
      if (e.parameter.reminder !== undefined) {
        const noteCols = ensureSheetColumns(sh, NOTE_OWNER_COLUMNS);
        sh.getRange(rowId, noteCols["Reminder"]).setNumberFormat("@").setValue(e.parameter.reminder.toString());
        if (typeof familieOnNoteSaved === "function") {
          const propNote = (sh.getRange(rowId, noteCols["Proprietar"]).getValue() || auth.profile.email).toString();
          familieOnNoteSaved(ss, auth, { rowId: rowId, text: noteContent, proprietar: propNote, reminder: e.parameter.reminder.toString(), nou: false });
        }
      }
      return returnResponse(e, { success: true });
    }
  }

  if (action === "deleteGand" || action === "deleteNote" || action === "deleteNotita") {
    let sh = ss.getSheetByName("Notite") || ss.getSheetByName("Ganduri");
    if (!sh || sh.getLastRow() < 2) return returnResponse(e, { success: true });
    
    let targetGand = (e.parameter.gand || e.parameter.nota || e.parameter.notita || e.parameter.text || "").toString().trim();
    let rowId = parseInt(e.parameter.rowId, 10);
    let deleted = false;

    // 1. Căutăm după textul gândului / notiței (asigură ștergerea corectă chiar dacă rândurile s-au decalat)
    if (targetGand) {
      const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
      const col = (header === "ID") ? 2 : 1;
      const vals = sh.getRange(2, col, sh.getLastRow() - 1, 1).getValues();
      for (let i = vals.length - 1; i >= 0; i--) {
        if (vals[i][0] && vals[i][0].toString().trim() === targetGand && noteRowAllowed(ss, sh, i + 2, auth.profile)) {
          sh.deleteRow(i + 2);
          deleted = true;
          break;
        }
      }
    }

    // 2. Fallback după rowId dacă nu s-a găsit după text
    if (!deleted && !isNaN(rowId) && rowId >= 2 && rowId <= sh.getLastRow() && noteRowAllowed(ss, sh, rowId, auth.profile)) {
      sh.deleteRow(rowId);
      deleted = true;
    }

    return returnResponse(e, { success: true, deleted: deleted });
  }

  // 5b. ȘTERGERE SOFT NOTIȚE (marcaj Sters=DA/NU, nu elimină rânduri; suportă selecție multiplă via rowIds="2,5,7")
  if (action === "softDeleteGand" || action === "restoreGand") {
    let sh = ss.getSheetByName("Notite") || ss.getSheetByName("Ganduri");
    if (!sh) return returnResponse(e, { success: true });
    const cols = getNoteStersColumns(sh);
    const isDelete = (action === "softDeleteGand");
    const timestamp = isDelete ? Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss") : "";
    const rowIdsStr = e.parameter.rowIds || e.parameter.rowId || "";
    const ids = rowIdsStr.toString().split(",").map(x => parseInt(x.trim(), 10)).filter(n => !isNaN(n));
    ids.forEach(rId => {
      if (rId >= 2 && rId <= sh.getLastRow() && noteRowAllowed(ss, sh, rId, auth.profile)) {
        sh.getRange(rId, cols.stersCol).setValue(isDelete ? "DA" : "NU");
        sh.getRange(rId, cols.dataStergereCol).setValue(timestamp);
        // notiță ștearsă → și evenimentul ei din calendar dispare (la restaurare se recreează din reminder)
        if (typeof familieOnNoteSaved === "function") {
          const nc = ensureSheetColumns(sh, NOTE_OWNER_COLUMNS);
          const remVal = sh.getRange(rId, nc["Reminder"]).getDisplayValue();
          familieOnNoteSaved(ss, auth, { rowId: rId, text: sh.getRange(rId, (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase() === "ID" ? 2 : 1).getValue().toString(),
            proprietar: (sh.getRange(rId, nc["Proprietar"]).getValue() || auth.profile.email).toString(), reminder: isDelete ? "" : remVal, nou: false });
        }
      }
    });
    return returnResponse(e, { success: true, count: ids.length });
  }

  // 6. CUMPĂRĂTURI (Structură suportată: cu ID pe Col 1 sau direct Magazin pe Col 1; suport autor & opțiune "Nu partaja")
  if (action === "addCumparaturi") {
    let sh = ss.getSheetByName("lista_cumparaturi");
    if (!sh) {
      sh = ss.insertSheet("lista_cumparaturi");
      sh.appendRow(["ID", "Magazin", "Produs", "Cantitate", "DataCreare", "NuPartaja", "Owner"]);
      sh.getRange("A1:G1").setFontWeight("bold");
    }
    const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
    const dataCreare = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
    // Autorul NU se mai lipește de numele produsului (cine a adăugat e deja în coloana Owner).
    const produsComplet = (e.parameter.produs || "").toString().trim();
    const nuPartaja = (e.parameter.nuPartaja === "true" || e.parameter.nuPartaja === true) ? "DA" : "NU";
    const owner = auth.profile.email;
    if (header === "ID") {
      const shopId = e.parameter.id || ("SHOP-" + Date.now());
      sh.appendRow([
        shopId,
        e.parameter.magazin || "General",
        produsComplet,
        e.parameter.cantitate || "1",
        dataCreare,
        nuPartaja,
        owner
      ]);
    } else {
      sh.appendRow([
        e.parameter.magazin || "General",
        produsComplet,
        e.parameter.cantitate || "1",
        nuPartaja,
        owner
      ]);
    }
    if (typeof familieOnShopAdded === "function") familieOnShopAdded(ss, auth, produsComplet, e.parameter.magazin || "", nuPartaja === "DA");
    return returnResponse(e, { success: true });
  }

  if (action === "deleteCumparaturi") {
    let sh = ss.getSheetByName("lista_cumparaturi");
    if (!sh || sh.getLastRow() < 2) return returnResponse(e, { success: true });

    let targetProdus = (e.parameter.produs || "").toString().trim();
    let targetMagazin = (e.parameter.magazin || "").toString().trim();
    let rowId = parseInt(e.parameter.rowId, 10);
    let deleted = false;

    // 0. După ID (exact, sigur) — din v24.9 aplicația trimite ID-ul produsului
    const targetId = (e.parameter.id || "").toString().trim();
    if (targetId) {
      const hdr = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
      if (hdr === "ID") {
        const idVals = sh.getRange(2, 1, sh.getLastRow() - 1, Math.max(sh.getLastColumn(), 7)).getValues();
        for (let i = idVals.length - 1; i >= 0; i--) {
          if ((idVals[i][0] || "").toString().trim() === targetId && shopItemVisibleTo(shopRowToItem(idVals[i], true), auth.profile)) {
            sh.deleteRow(i + 2);
            return returnResponse(e, { success: true, deleted: true });
          }
        }
      }
    }

    // 1. Căutăm după produs și magazin (asigură ștergerea corectă chiar dacă rândurile s-au decalat)
    if (targetProdus) {
      const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
      const hasIdCol = (header === "ID");
      const magazinCol = hasIdCol ? 2 : 1;
      const produsCol = hasIdCol ? 3 : 2;
      const maxRows = sh.getLastRow() - 1;
      const vals = sh.getRange(2, 1, maxRows, Math.max(sh.getLastColumn(), hasIdCol ? 7 : 5)).getValues();
      for (let i = vals.length - 1; i >= 0; i--) {
        let pVal = (vals[i][produsCol - 1] || "").toString().trim();
        let mVal = (vals[i][magazinCol - 1] || "").toString().trim();
        if (!shopItemVisibleTo(shopRowToItem(vals[i], hasIdCol), auth.profile)) continue;
        if ((pVal === targetProdus || pVal.startsWith(targetProdus + " (")) && (!targetMagazin || mVal === targetMagazin || !mVal)) {
          sh.deleteRow(i + 2);
          deleted = true;
          break;
        }
      }
    }

    // 2. Fallback după rowId dacă nu s-a găsit după conținut
    if (!deleted && !isNaN(rowId) && rowId >= 2 && rowId <= sh.getLastRow()) {
      const hasIdColFb = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase() === "ID";
      const rowVals = sh.getRange(rowId, 1, 1, Math.max(sh.getLastColumn(), hasIdColFb ? 7 : 5)).getValues()[0];
      if (shopItemVisibleTo(shopRowToItem(rowVals, hasIdColFb), auth.profile)) {
        sh.deleteRow(rowId);
        deleted = true;
      }
    }

    return returnResponse(e, { success: true, deleted: deleted });
  }

  if (action === "deleteMagazinCumparaturi") {
    let sh = ss.getSheetByName("lista_cumparaturi");
    const magazin = e.parameter.magazin;
    if (sh && magazin && sh.getLastRow() >= 2) {
      const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
      const hasIdCol = header === "ID";
      const magazinCol = hasIdCol ? 2 : 1;
      const vals = sh.getRange(2, 1, sh.getLastRow() - 1, Math.max(sh.getLastColumn(), hasIdCol ? 7 : 5)).getValues();
      for (let i = vals.length - 1; i >= 0; i--) {
        const m = vals[i][magazinCol - 1];
        // Produsele private ale altora rămân pe listă (nici nu le vezi)
        if (m && m.toString().trim() === magazin.toString().trim() && shopItemVisibleTo(shopRowToItem(vals[i], hasIdCol), auth.profile)) {
          sh.deleteRow(i + 2);
        }
      }
    }
    return returnResponse(e, { success: true });
  }

  // 7. CRUD COMPLET PERSOANĂ POMELNIC (Adăugare pe ultimul rând & Editare completă)
  if (action === "addPerson") {
    let sh = ss.getSheetByName("Pomelnic");
    if (!sh) {
      return returnResponse(e, { error: "Foaia Pomelnic nu a fost găsită" });
    }
    const newRowId = sh.getLastRow() + 1;
    const isActiv = (e.parameter.isActivPomelnic === "true" || e.parameter.isActivPomelnic === true || e.parameter.isActivPomelnic === "1" || e.parameter.isActivPomelnic === 1);
    const stare = (e.parameter.stare || "Viu").toString().trim();
    const van = (stare === "Adormit") ? "A" : "V";
    const necesitaAtentie = (e.parameter.necesitaAtentie === "true" || e.parameter.necesitaAtentie === true);
    const bolnav = (e.parameter.bolnav === "true" || e.parameter.bolnav === true);
    const dnAprox = (e.parameter.dnAprox === "true" || e.parameter.dnAprox === true);
    const daAprox = (e.parameter.daAprox === "true" || e.parameter.daAprox === true);
    const pomelnicAndreea = (e.parameter.pomelnicAndreea === "true" || e.parameter.pomelnicAndreea === true);

    sh.appendRow([
      newRowId,                                       // Col 1 (A): RowID
      van,                                            // Col 2 (B): (F) V/A/N
      isActiv,                                        // Col 3 (C): (F) Activ_Inactiv/calcul_pomelnic
      pomelnicAndreea,                                // Col 4 (D): Pomelnic_Andreea
      false,                                          // Col 5 (E): Pomelnic_calendar
      (e.parameter.prenume || "").trim(),             // Col 6 (F): Prenume
      (e.parameter.nume || "").trim(),                // Col 7 (G): Nume
      (e.parameter.info || "").trim(),                // Col 8 (H): Informatii
      (e.parameter.istoricMedical || "").trim(),      // Col 9 (I): Istoric medical
      (e.parameter.dn || "").trim(),                  // Col 10 (J): Data nasterii
      dnAprox,                                        // Col 11 (K): D A N A
      (e.parameter.da || "").trim(),                  // Col 12 (L): Data adormirii
      daAprox,                                        // Col 13 (M): D A D A
      bolnav,                                         // Col 14 (N): Bolnav
      stare,                                          // Col 15 (O): Starea sufleteasca
      (e.parameter.grad || "").trim(),                // Col 16 (P): Grad (Coloana P!)
      (e.parameter.copii || "-").trim(),              // Col 17 (Q): Copil
      (e.parameter.sot || "-").trim(),                // Col 18 (R): Sot/Sotie
      (e.parameter.parinti || "-").trim(),            // Col 19 (S): Parinti
      (e.parameter.apropiati || "-").trim(),          // Col 20 (T): Frati, bunici, apropiati
      "09:00",                                        // Col 21 (U): StartTime
      "10:00",                                        // Col 22 (V): EndTime
      necesitaAtentie                                 // Col 23 (W): NECESITA_Atentie!!!
    ]);
    return returnResponse(e, { success: true, rowId: newRowId });
  }

  if (action === "updatePerson") {
    let sh = ss.getSheetByName("Pomelnic");
    let rowId = parseInt(e.parameter.rowId, 10);
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      if (e.parameter.prenume !== undefined) sh.getRange(rowId, 6).setValue(e.parameter.prenume);
      if (e.parameter.nume !== undefined) sh.getRange(rowId, 7).setValue(e.parameter.nume);
      if (e.parameter.info !== undefined) sh.getRange(rowId, 8).setValue(e.parameter.info);
      if (e.parameter.istoricMedical !== undefined) sh.getRange(rowId, 9).setValue(e.parameter.istoricMedical);
      if (e.parameter.dn !== undefined) sh.getRange(rowId, 10).setValue(e.parameter.dn);
      if (e.parameter.dnAprox !== undefined) {
        const dnVal = (e.parameter.dnAprox === "true" || e.parameter.dnAprox === true);
        sh.getRange(rowId, 11).setValue(dnVal);
      }
      if (e.parameter.da !== undefined) sh.getRange(rowId, 12).setValue(e.parameter.da);
      if (e.parameter.daAprox !== undefined) {
        const daVal = (e.parameter.daAprox === "true" || e.parameter.daAprox === true);
        sh.getRange(rowId, 13).setValue(daVal);
      }
      if (e.parameter.bolnav !== undefined) {
        const bVal = (e.parameter.bolnav === "true" || e.parameter.bolnav === true);
        sh.getRange(rowId, 14).setValue(bVal);
      }
      if (e.parameter.stare !== undefined) {
        sh.getRange(rowId, 15).setValue(e.parameter.stare);
        sh.getRange(rowId, 2).setValue(e.parameter.stare === "Adormit" ? "A" : "V");
      }
      if (e.parameter.grad !== undefined) sh.getRange(rowId, 16).setValue(e.parameter.grad);
      if (e.parameter.copii !== undefined) sh.getRange(rowId, 17).setValue(e.parameter.copii);
      if (e.parameter.sot !== undefined) sh.getRange(rowId, 18).setValue(e.parameter.sot);
      if (e.parameter.parinti !== undefined) sh.getRange(rowId, 19).setValue(e.parameter.parinti);
      if (e.parameter.apropiati !== undefined) sh.getRange(rowId, 20).setValue(e.parameter.apropiati);
      if (e.parameter.necesitaAtentie !== undefined) {
        const valBool = (e.parameter.necesitaAtentie === "true" || e.parameter.necesitaAtentie === true);
        sh.getRange(rowId, 23).setValue(valBool);
      }
      if (e.parameter.isActivPomelnic !== undefined) {
        const valAct = (e.parameter.isActivPomelnic === "true" || e.parameter.isActivPomelnic === true);
        sh.getRange(rowId, 3).setValue(valAct);
      }
      if (e.parameter.pomelnicAndreea !== undefined) {
        const valAndreea = (e.parameter.pomelnicAndreea === "true" || e.parameter.pomelnicAndreea === true || e.parameter.pomelnicAndreea === "1" || e.parameter.pomelnicAndreea === 1);
        sh.getRange(rowId, 4).setValue(valAndreea);
      }
      return returnResponse(e, { success: true });
    }
  }

  // Actualizare în calup Pomelnic Andreea (Coloana D)
  if (action === "batchUpdateAndreea") {
    let sh = ss.getSheetByName("Pomelnic");
    const rowIdsStr = e.parameter.rowIds || "";
    const flagValue = (e.parameter.pomelnicAndreea === "true" || e.parameter.pomelnicAndreea === true);
    if (sh && rowIdsStr) {
      const ids = rowIdsStr.split(",").map(x => parseInt(x.trim(), 10)).filter(n => !isNaN(n));
      ids.forEach(rId => {
        if (rId >= 2 && rId <= sh.getLastRow()) {
          sh.getRange(rId, 4).setValue(flagValue);
        }
      });
      return returnResponse(e, { success: true, count: ids.length, flag: flagValue });
    }
    return returnResponse(e, { error: "Parametri lipsă" });
  }

  // Actualizare în calup Necesită Atenție (Pomelnic - Coloana W = 23)
  if (action === "batchUpdateAtentie") {
    let sh = ss.getSheetByName("Pomelnic");
    const rowIdsStr = e.parameter.rowIds || "";
    const flagValue = (e.parameter.necesitaAtentie === "true" || e.parameter.necesitaAtentie === true);
    if (sh && rowIdsStr) {
      const ids = rowIdsStr.split(",").map(x => parseInt(x.trim(), 10)).filter(n => !isNaN(n));
      ids.forEach(rId => {
        if (rId >= 2 && rId <= sh.getLastRow()) {
          sh.getRange(rId, 23).setValue(flagValue);
        }
      });
    }
    return returnResponse(e, { success: true });
  }

  // 8. SALVARE CONFIGURĂRI NOTIFICĂRI ÎN SHEET — per profil (cheia notif_cfg_<email>), fiecare doar pe ale lui
  if (action === "saveNotificationConfigs") {
    let sh = ss.getSheetByName("Setari");
    if (!sh) {
      sh = ss.insertSheet("Setari");
      sh.appendRow(["Cheie", "Valoare", "UltimaModificare"]);
    }
    setSettingValue(sh, "notif_cfg_" + auth.profile.email, e.parameter.configJson);
    return returnResponse(e, { success: true });
  }

  // 8a. ȘCOALĂ → TEME (tab Scoala_Teme; adăugare/editare/bifare/ștergere) — v24.5
  if (action === "saveSchoolTema") {
    const sh = getOrCreateSchoolTemeSheet(ss);
    const now = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
    const id = e.parameter.id || ("TEMA-" + Date.now());
    const p = e.parameter;
    const row = [id, p.child || "", p.subject || "", p.text || "", p.givenDate || "", p.dueDate || "",
      p.status === "Gata" ? "Gata" : "De făcut", p.doneDate || "", p.source || "manual",
      p.dataCreare || now, now, auth.profile.email || ""];
    const idx = findRowIndexById(sh, id);
    if (idx > 0) {
      row[9] = sh.getRange(idx, 10).getValue() || row[9];
      sh.getRange(idx, 1, 1, row.length).setValues([row]);
    } else {
      sh.appendRow(row);
    }
    return returnResponse(e, { success: true, id: id });
  }

  if (action === "deleteSchoolTema") {
    const sh = ss.getSheetByName("Scoala_Teme");
    if (sh && e.parameter.id) deleteRowById(sh, e.parameter.id, 1);
    return returnResponse(e, { success: true });
  }

  // 8a2. ADSERVIO (v24.6) — datele aduse de sincronizare (fișierul Adservio.gs) + sincronizare la cerere
  if (action === "getAdservioData") {
    return returnResponse(e, { adservio: getAdservioData(ss) });
  }

  // Cheia de trimitere pentru marcajul „Adservio → DIiH” — doar admin (acțiune nelistată în AUTH_ACTION_RULES)
  if (action === "adservioMarcajKey") {
    if (typeof aratamCheiaPush !== "function") return returnResponse(e, { success: false, error: "Lipsește fișierul Adservio.gs din proiect." });
    return returnResponse(e, { success: true, key: aratamCheiaPush() });
  }

  if (action === "adservioSyncNow") {
    try {
      const rez = adservioSync("manual, din aplicație (" + (auth.profile.email || "-") + ")");
      return returnResponse(e, { success: true, rezultat: rez, adservio: getAdservioData(ss) });
    } catch (err) {
      return returnResponse(e, { success: false, error: err.message || String(err) });
    }
  }

  // 8b. CRUD INVENTAR PERSONAL (Obiecte, spații de depozitare, mentenanță)
  if (action === "getInventoryData") {
    return returnResponse(e, { inventar: getInventoryData(ss) });
  }

  if (action === "addInventoryItem" || action === "updateInventoryItem") {
    let sh = ss.getSheetByName("Inventar");
    if (!sh) {
      sh = ss.insertSheet("Inventar");
      sh.appendRow(["ID", "Denumire", "Categorie", "Spatiu", "Cantitate", "Stare", "Valoare", "DataAchizitie", "UrmatoareaMentenanta", "Observatii", "PozaBase64", "IstoricMutariJson", "DataCreare", "DataModificare"]);
      sh.getRange("A1:N1").setFontWeight("bold");
    }
    const now = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
    const itemId = e.parameter.id || ("INV-" + Date.now());

    if (action === "updateInventoryItem") {
      deleteRowById(sh, itemId, 1);
    }

    sh.appendRow([
      itemId,
      e.parameter.name || "Obiect",
      e.parameter.category || "-",
      e.parameter.space || "-",
      e.parameter.qty || "1",
      e.parameter.status || "Funcțional",
      e.parameter.value || "",
      e.parameter.purchaseDate || "",
      e.parameter.nextMaintenance || "",
      e.parameter.notes || "",
      e.parameter.photoBase64 || "",
      e.parameter.historyJson || "[]",
      action === "addInventoryItem" ? now : (e.parameter.dataCreare || now),
      now
    ]);
    return returnResponse(e, { success: true, id: itemId });
  }

  if (action === "deleteInventoryItem") {
    let sh = ss.getSheetByName("Inventar");
    if (sh && e.parameter.id) deleteRowById(sh, e.parameter.id, 1);
    return returnResponse(e, { success: true });
  }

  // 8c. GÂNDUL ZILEI (tab nou GandulZilei; adăugare/editare/dezactivare — niciodată ștergere de rând)
  if (action === "getGandulZileiData") {
    return returnResponse(e, { gandulZilei: getGandulZileiData(ss) });
  }

  if (action === "addGandulZilei") {
    let sh = getOrCreateGandulZileiSheet(ss);
    const now = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
    const newId = "GZ-" + Date.now();
    sh.appendRow([newId, e.parameter.text || "", e.parameter.autor || "", e.parameter.tema || "Înțelepciune", "DA", now, now]);
    return returnResponse(e, { success: true, id: newId });
  }

  if (action === "updateGandulZilei") {
    let sh = getOrCreateGandulZileiSheet(ss);
    const id = e.parameter.id;
    const maxRows = sh.getLastRow();
    if (sh && id && maxRows >= 2) {
      const ids = sh.getRange(2, 1, maxRows - 1, 1).getValues();
      for (let i = 0; i < ids.length; i++) {
        if ((ids[i][0] || "").toString().trim() === id.toString().trim()) {
          const r = i + 2;
          if (e.parameter.text !== undefined) sh.getRange(r, 2).setValue(e.parameter.text);
          if (e.parameter.autor !== undefined) sh.getRange(r, 3).setValue(e.parameter.autor);
          if (e.parameter.tema !== undefined) sh.getRange(r, 4).setValue(e.parameter.tema);
          if (e.parameter.activ !== undefined) sh.getRange(r, 5).setValue(e.parameter.activ === "true" || e.parameter.activ === true ? "DA" : "NU");
          sh.getRange(r, 7).setValue(Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss"));
          break;
        }
      }
    }
    return returnResponse(e, { success: true });
  }

  // 8d. MODEL CITIRE (Secțiunea 4-5): tab NOU ModelCitire, singurul tab complet nou cerut — populat
  //     idempotent cu cele 3 modele predefinite DOAR dacă foaia e goală la prima utilizare
  if (action === "getModelCitireData") {
    return returnResponse(e, { modelCitire: getModelCitireData(ss) });
  }

  if (action === "setModelCitirePozitieBifat") {
    let sh = getOrCreateModelCitireSheet(ss);
    const rowId = parseInt(e.parameter.rowId, 10);
    const bifat = (e.parameter.bifat === "true" || e.parameter.bifat === true);
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      sh.getRange(rowId, 6).setValue(bifat ? "DA" : "NU");
      sh.getRange(rowId, 7).setValue(bifat ? Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss") : "");
    }
    return returnResponse(e, { success: true });
  }

  // 8e. EDITARE NOTE BIBLIA ANANIA — actualizează doar câmpurile trimise, completează DataModificare
  if (action === "updateNoteAnania") {
    let sh = ss.getSheetByName("noteBIBLIA_Anania") || ss.getSheetByName("Biblia_Note");
    const rowId = parseInt(e.parameter.rowId, 10);
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      const cols = ensureSheetColumns(sh, ["Sters", "DataStergere", "DataModificare"]);
      if (e.parameter.textVerset !== undefined) sh.getRange(rowId, 6).setValue(e.parameter.textVerset);
      if (e.parameter.context !== undefined) sh.getRange(rowId, 7).setValue(e.parameter.context);
      if (e.parameter.nota !== undefined) sh.getRange(rowId, 8).setValue(e.parameter.nota);
      if (e.parameter.derventUrl !== undefined) sh.getRange(rowId, 9).setValue(e.parameter.derventUrl);
      if (e.parameter.capitolProgres !== undefined) sh.getRange(rowId, 10).setValue(e.parameter.capitolProgres);
      sh.getRange(rowId, cols["DataModificare"]).setValue(Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss"));
    }
    return returnResponse(e, { success: true });
  }

  // 8f. ȘTERGERE SOFT NOTE BIBLIA ANANIA (ca la Notițe: marcaj Sters=DA/NU, nu elimină rânduri)
  if (action === "softDeleteNoteAnania" || action === "restoreNoteAnania") {
    let sh = ss.getSheetByName("noteBIBLIA_Anania") || ss.getSheetByName("Biblia_Note");
    if (!sh) return returnResponse(e, { success: true });
    const cols = ensureSheetColumns(sh, ["Sters", "DataStergere"]);
    const isDelete = (action === "softDeleteNoteAnania");
    const timestamp = isDelete ? Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss") : "";
    const ids = (e.parameter.rowIds || e.parameter.rowId || "").toString().split(",").map(x => parseInt(x.trim(), 10)).filter(n => !isNaN(n));
    ids.forEach(rId => {
      if (rId >= 2 && rId <= sh.getLastRow()) {
        sh.getRange(rId, cols["Sters"]).setValue(isDelete ? "DA" : "NU");
        sh.getRange(rId, cols["DataStergere"]).setValue(timestamp);
      }
    });
    return returnResponse(e, { success: true, count: ids.length });
  }

  // 8g. PROGRES SCRIPTURĂ PER PROFIL (v24.1) — capitole citite, model activ, lectura curentă, istoric
  if (action === "saveScripturaProgress") {
    const row = scripturaGetRow(ss, auth.profile.email);
    if (e.parameter.capitole !== undefined) row.capitole = e.parameter.capitole.toString().slice(0, 40000);
    if (e.parameter.modelActiv !== undefined) row.modelActiv = e.parameter.modelActiv.toString().slice(0, 100);
    if (!row.inceputLa) row.inceputLa = authNow();
    scripturaSaveRow(ss, row);
    return returnResponse(e, { success: true, scripturaProgres: scripturaPublic(row) });
  }

  // Lectură nouă: lectura curentă intră în istoric (cu copie a capitolelor), progresul o ia de la zero
  if (action === "newScripturaReading") {
    const row = scripturaArchive(ss, auth.profile.email, "încheiată");
    return returnResponse(e, { success: true, scripturaProgres: scripturaPublic(row) });
  }

  // Admin: resetează progresul unui profil (cu copie de siguranță) sau restaurează ultima copie
  if (action === "adminResetScriptura") {
    if (!authFindProfile(ss, e.parameter.email)) return returnResponse(e, { success: false, error: "Profil inexistent." });
    const row = scripturaArchive(ss, e.parameter.email, "resetată de admin");
    logToAppSheet(ss, { modul: "Scriptura", actiune: "reset_progres", status: "OK", detalii: e.parameter.email, user: auth.profile.email });
    return returnResponse(e, { success: true, scripturaProgres: scripturaPublic(row) });
  }

  if (action === "adminRestoreScriptura") {
    const row = scripturaGetRow(ss, (e.parameter.email || "").toString().toLowerCase());
    const last = row.istoric.length ? row.istoric[row.istoric.length - 1] : null;
    if (!last || last.capitole === undefined) return returnResponse(e, { success: false, error: "Nu există o copie de restaurat." });
    row.istoric.pop();
    row.capitole = last.capitole;
    row.modelActiv = last.model || row.modelActiv;
    row.inceputLa = last.inceput || row.inceputLa;
    row.nrLectura = Math.max(1, row.nrLectura - 1);
    scripturaSaveRow(ss, row);
    logToAppSheet(ss, { modul: "Scriptura", actiune: "restaurare_progres", status: "OK", detalii: row.email, user: auth.profile.email });
    return returnResponse(e, { success: true, scripturaProgres: scripturaPublic(row) });
  }

  // 8h. COPII DE SIGURANȚĂ (v24.4) — doar admin (acțiuni nelistate în AUTH_ACTION_RULES)
  if (action === "adminBackupStatus") {
    return returnResponse(e, backupStatus(ss));
  }
  if (action === "adminBackupNow") {
    try {
      backupCreeazaCopie(ss, "manual, din aplicație (" + auth.profile.email + ")");
      return returnResponse(e, backupStatus(ss));
    } catch (err) {
      return returnResponse(e, { success: false, error: "Copia nu s-a putut face: " + (err.message || err) });
    }
  }

  // Populare inițială unică: scrie citatele trimise doar dacă foaia GandulZilei este încă goală (nu suprascrie date reale)
  if (action === "seedGandulZileiIfEmpty") {
    let sh = getOrCreateGandulZileiSheet(ss);
    if (sh.getLastRow() < 2) {
      const items = JSON.parse(e.parameter.itemsJson || "[]");
      const now = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
      const rows = items.map((it, idx) => ["GZ-SEED-" + idx, it.quote || it.text || "", it.author || it.autor || "", it.tema || "Înțelepciune", "DA", now, now]);
      if (rows.length > 0) sh.getRange(2, 1, rows.length, 7).setValues(rows);
      return returnResponse(e, { success: true, seeded: rows.length });
    }
    return returnResponse(e, { success: true, seeded: 0 });
  }

  // 9. PRELUARE COMPLETĂ DATE PENTRU APLICAȚIE (RĂSPUNS 1:1)
  if (action === "getLiveData") {
    // Sincronizare ușoară/rapidă (fără Pomelnic 990 rânduri și fără Biblia 360) - folosită la polling frecvent
    const notesList = getGanduriData(ss);
    const liveData = {
      settings: getSettingsData(ss),
      settingsRaw: getAllSettingsRaw(ss),
      cumparaturi: getCumparaturiData(ss),
      ganduri: notesList,
      notite: notesList,
      gandulZilei: getGandulZileiData(ss),
      modelCitire: getModelCitireData(ss),
      schoolActivities: getSchoolActivitiesData(ss),
      schoolTeme: getSchoolTemeData(ss),
      adservio: getAdservioDataSafe(ss),
      psaltireGrup: getPsaltireGrupData(ss)
    };
    const liveOut = authFilterResponse(liveData, auth, ss);
    if (typeof familieGetNotificari === "function") liveOut.notificari = familieGetNotificari(ss, auth);
    return returnResponse(e, liveOut);
  }

  const notesList = getGanduriData(ss);
  const responseData = {
    backendVersion: BACKEND_VERSION,
    settings: getSettingsData(ss),
    settingsRaw: getAllSettingsRaw(ss),
    pomelnic: getPomelnicData(ss),
    cumparaturi: getCumparaturiData(ss),
    ganduri: notesList,
    notite: notesList,
    gandulZilei: getGandulZileiData(ss),
    modelCitire: getModelCitireData(ss),
    noteAnania: getNoteAnaniaData(ss),
    bibliaBVA: getBiblia360Data(ss),
    schoolActivities: getSchoolActivitiesData(ss),
    schoolTeme: getSchoolTemeData(ss),
    adservio: getAdservioDataSafe(ss),
    psaltireGrup: getPsaltireGrupData(ss)
  };

  const allOut = authFilterResponse(responseData, auth, ss);
  if (typeof familieGetNotificari === "function") allOut.notificari = familieGetNotificari(ss, auth);
  return returnResponse(e, allOut);
}

function returnResponse(e, data) {
  const callback = e.parameter.callback;
  if (callback) {
    if (callback === "void") {
      return ContentService.createTextOutput("/* void */")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(`${callback}(${JSON.stringify(data)})`)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

const SETTINGS_RAW_HIDDEN_KEYS = ["sharing_config_json", "user_profiles_json"];

function getAllSettingsRaw(ss) {
  let sh = ss.getSheetByName("Setari");
  const out = {};
  if (!sh || sh.getLastRow() < 2) return out;
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues();
  rows.forEach(r => {
    const k = (r[0] || "").toString().trim();
    if (SETTINGS_RAW_HIDDEN_KEYS.indexOf(k) !== -1 || k.indexOf("auth_") === 0 || k.indexOf("notif_cfg_") === 0) return;
    if (k) out[k] = (r[1] === undefined || r[1] === null) ? "" : r[1].toString();
  });
  return out;
}

// Citește valoarea unei chei din Setari (inclusiv chei ascunse aplicației, ex. notif_cfg_<email>)
function getSettingValueRaw(ss, key) {
  const sh = ss.getSheetByName("Setari");
  if (!sh || sh.getLastRow() < 2) return "";
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues();
  for (let i = 0; i < rows.length; i++) {
    if ((rows[i][0] || "").toString().trim() === key) return (rows[i][1] || "").toString();
  }
  return "";
}

function setSettingValue(sheet, key, value) {
  if (value === undefined || value === null) return;
  const maxRows = sheet.getLastRow();
  const timestamp = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
  if (maxRows >= 2) {
    const keys = sheet.getRange(2, 1, maxRows - 1, 1).getValues();
    for (let i = 0; i < keys.length; i++) {
      if (keys[i][0] && keys[i][0].toString().trim() === key) {
        sheet.getRange(i + 2, 2, 1, 2).setValues([[value, timestamp]]);
        return;
      }
    }
  }
  sheet.appendRow([key, value, timestamp]);
}

// Helper generic: citește header-ul unei foi, creează (idempotent) coloanele lipsă din listă la finalul foii
// fără să atingă coloanele existente. Returnează un obiect { numeColoana: indexColoana(1-based) }.
function ensureSheetColumns(sheet, columnNames) {
  const lastCol = Math.max(sheet.getLastColumn(), 1);
  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => (h || "").toString().trim());
  const result = {};
  let nextCol = lastCol;
  columnNames.forEach(name => {
    const idx = headers.indexOf(name);
    if (idx !== -1) {
      result[name] = idx + 1;
    } else {
      nextCol += 1;
      sheet.getRange(1, nextCol).setValue(name);
      headers.push(name);
      result[name] = nextCol;
    }
  });
  return result;
}

// Scrie un eveniment de audit la nivel de aplicație în tabul LOG_app (separat de LOG_Sistem, folosit de sincronizarea Pomelnic)
function logToAppSheet(ss, d) {
  let sh = ss.getSheetByName("LOG_app");
  if (!sh) {
    sh = ss.insertSheet("LOG_app");
    sh.appendRow(["Timestamp", "Modul", "Actiune", "Status", "Detalii", "User"]);
    sh.getRange("A1:F1").setFontWeight("bold");
  }
  sh.appendRow([
    Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss"),
    d.modul || "General",
    d.actiune || "-",
    d.status || "INFO",
    d.detalii || "",
    d.user || "-"
  ]);
  if (sh.getLastRow() > 2000) sh.deleteRows(2, 500);
}

function getSchoolActivitiesData(ss) {
  let sh = ss.getSheetByName("Scoala_Activitati") || ss.getSheetByName("Activitati_Scoala");
  if (!sh || sh.getLastRow() < 2) return [];
  const cols = ensureSheetColumns(sh, ["Data_Achitat", "Comentariu_Plata"]);
  const maxCol = Math.max(14, cols["Data_Achitat"], cols["Comentariu_Plata"]);
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, maxCol).getValues();
  return rows.map(r => {
    let dateVal = r[1];
    let dateStr = "";
    if (dateVal instanceof Date) {
      dateStr = Utilities.formatDate(dateVal, "Europe/Bucharest", "yyyy-MM-dd");
    } else if (typeof dateVal === "string") {
      dateStr = dateVal.substring(0, 10);
    }
    const meta = parseJsonSafe(r[12]);
    const isMetaObj = meta && typeof meta === "object" && !Array.isArray(meta);
    const reminders = isMetaObj ? (meta.reminders || [60, 30, 10]) : (Array.isArray(meta) ? meta : [60, 30, 10]);
    const color = isMetaObj ? (meta.color || "auto") : "auto";
    const startHour = isMetaObj ? (meta.startHour || "08:00") : "08:00";
    const endHour = isMetaObj ? (meta.endHour || "16:00") : "16:00";
    const author = isMetaObj ? (meta.author || "Bogdan") : "Bogdan";

    return {
      id: r[0],
      date: dateStr,
      child: r[2] || "Copil",
      institution: r[3] || "-",
      title: r[4] || "Activitate",
      paymentType: r[5] || "Pe oră",
      hours: parseFloat(r[6]) || 1,
      rate: parseFloat(r[7]) || 0,
      total: parseFloat(r[8]) || 0,
      paymentStatus: r[9] || "În așteptare",
      deadline: r[10] ? r[10].toString().substring(0, 10) : "-",
      dayMessage: r[11] || "",
      reminders: reminders,
      color: color,
      startHour: startHour,
      endHour: endHour,
      // true = rândul nu are oră salvată (importuri vechi / metadate fără ore): nu ocupă un interval orar
      noHours: !(isMetaObj && meta.startHour),
      author: author,
      dataAchitat: r[cols["Data_Achitat"] - 1] ? parseDateToText(r[cols["Data_Achitat"] - 1]) : "",
      comentariuPlata: r[cols["Comentariu_Plata"] - 1] || ""
    };
  });
}

function parseJsonSafe(val) {
  try { return JSON.parse(val || "[]"); } catch (e) { return []; }
}

// v24.0.3: un produs marcat „Nu partaja” e vizibil doar celui care l-a adăugat. Owner = emailul contului;
// la produsele vechi, owner e un nume (ex. „Bogdan Murar”), potrivit cu numele profilului; fără owner = adminul.
function shopItemVisibleTo(it, prof) {
  if (!prof) return false;
  if (!it.nuPartaja) return true;
  const o = (it.owner || "").toString().toLowerCase().trim();
  if (!o) return prof.rol === "admin";
  if (o.indexOf("@") !== -1) return o === prof.email;
  const nm = (prof.nume || "").toLowerCase().trim();
  return nm === o || (nm && nm.split(" ")[0] === o.split(" ")[0]);
}

  };
}

function getGanduriData(ss) {
  const sh = ss.getSheetByName("Notite") || ss.getSheetByName("Ganduri");
  if (!sh || sh.getLastRow() < 2) return [];
  const maxRows = sh.getLastRow();
  const maxCols = Math.min(Math.max(sh.getLastColumn(), 3), 8);
  const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
  const hasIdCol = (header === "ID");
  const cols = getNoteStersColumns(sh);
  const ownerCols = ensureSheetColumns(sh, NOTE_OWNER_COLUMNS);
  const lastNeeded = Math.max(maxCols, cols.stersCol, ownerCols["Proprietar"], ownerCols["CreatDe"], ownerCols["Reminder"]);
  const rows = sh.getRange(2, 1, maxRows - 1, lastNeeded).getValues();
  return rows.map((r, i) => {
    let gand = hasIdCol ? (r[1] || "") : (r[0] || "");
    let sursa = hasIdCol ? (r[2] || "Personal") : (r[1] || "Personal");
    let dataVal = hasIdCol ? r[3] : r[2];
    let user = hasIdCol ? (r[4] || "").toString().trim() : (r[3] || "").toString().trim();
    let sters = (r[cols.stersCol - 1] || "").toString().trim().toUpperCase() === "DA";
    const rem = r[ownerCols["Reminder"] - 1];
    return {
      rowId: i + 2,
      gand: gand.toString().trim(),
      nota: gand.toString().trim(),
      sursa: sursa.toString().trim(),
      data: dataVal ? parseDateToText(dataVal) : "",
      user: user,
      proprietar: (r[ownerCols["Proprietar"] - 1] || "").toString().trim().toLowerCase(),
      creatDe: (r[ownerCols["CreatDe"] - 1] || "").toString().trim().toLowerCase(),
      // Remindere vechi salvate ca dată: se citesc în fusul registrului (cel în care au fost interpretate), nu forțat în București
      reminder: rem instanceof Date ? Utilities.formatDate(rem, ss.getSpreadsheetTimeZone() || "Europe/Bucharest", "yyyy-MM-dd'T'HH:mm") : (rem || "").toString().trim(),
      sters: sters
    };
  }).filter(x => x.gand !== "" && !x.sters);
}

// v24.0.2: notițe per profil. Proprietar = emailul profilului care vede notița; CreatDe = cine a scris-o
// (adminul poate scrie pentru alții); Reminder = data și ora (yyyy-MM-ddTHH:mm) pentru notificare.
const NOTE_OWNER_COLUMNS = ["Proprietar", "CreatDe", "Reminder"];

// Notițele vechi (dinainte de v24.0.2) nu au proprietar: aparțin profilului al cărui nume se potrivește cu
// coloana „user” (ex. „Andreea Murar”), iar cele fără user, adminului.
function noteVisibleTo(n, prof) {
  if (!prof) return false;
  if (n.proprietar) return n.proprietar === prof.email || n.creatDe === prof.email;
  const u = (n.user || "").toString().toLowerCase().trim();
  if (!u) return prof.rol === "admin";
  const nm = (prof.nume || "").toLowerCase().trim();
  return nm === u || (nm && nm.split(" ")[0] === u.split(" ")[0]);
}

function noteRowAllowed(ss, sh, rowId, prof) {
  const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
  const ownerCols = ensureSheetColumns(sh, NOTE_OWNER_COLUMNS);
  const width = Math.max(sh.getLastColumn(), ownerCols["Reminder"]);
  const r = sh.getRange(rowId, 1, 1, width).getValues()[0];
  return noteVisibleTo({
    user: (header === "ID" ? r[4] : r[3]) || "",
    proprietar: (r[ownerCols["Proprietar"] - 1] || "").toString().trim().toLowerCase(),
    creatDe: (r[ownerCols["CreatDe"] - 1] || "").toString().trim().toLowerCase()
  }, prof);
}

// Returnează indicii coloanelor Sters / DataStergere pentru foaia Notite/Ganduri, creându-le idempotent dacă lipsesc
function getNoteStersColumns(sheet) {
  const lastCol = Math.max(sheet.getLastColumn(), 1);
  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => (h || "").toString().trim().toUpperCase());
  let stersCol = headers.indexOf("STERS") + 1;
  let dataStergereCol = headers.indexOf("DATASTERGERE") + 1;
  if (stersCol === 0) {
    stersCol = lastCol + 1;
    sheet.getRange(1, stersCol).setValue("Sters");
  }
  if (dataStergereCol === 0) {
    dataStergereCol = Math.max(lastCol, stersCol) + 1;
    sheet.getRange(1, dataStergereCol).setValue("DataStergere");
  }
  return { stersCol: stersCol, dataStergereCol: dataStergereCol };
}

function getNoteAnaniaData(ss) {
  const sh = ss.getSheetByName("noteBIBLIA_Anania") || ss.getSheetByName("Biblia_Note");
  if (!sh || sh.getLastRow() < 2) return [];
  // Sters/DataStergere/DataModificare: coloane adăugate idempotent la finalul foii
  // (pentru ștergere cu confirmare și editare, ca la Notițe)
  const stersCols = ensureSheetColumns(sh, ["Sters", "DataStergere", "DataModificare"]);
  const maxCols = Math.max(Math.min(Math.max(sh.getLastColumn(), 4), 10), stersCols["Sters"], stersCols["DataStergere"], stersCols["DataModificare"]);
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, maxCols).getValues();

  // Structură coloane registru (10 coloane conform foii noteBIBLIA_Anania):
  // Col A (0): Colecție (ex: Vechiul Testament)
  // Col B (1): Carte (ex: Facerea)
  // Col C (2): Capitol (ex: Capitol 1)
  // Col D (3): Verset (ex: 1)
  // Col E (4): Marcaj (ex: (A))
  // Col F (5): Text verset complet (ex: 1Întru'nceput[A]a făcut[B]Dumnezeu cerul și pământul[C].)
  // Col G (6): Context / fragment adnotat (ex: 1Întru'nceput)
  // Col H (7): Nota de subsol (ex: [Nota A] În interiorul...)
  // Col I (8): URL sursă (ex: https://biblia.dervent.ro/VT-Fc-01)
  // Col J (9): Capitol # / # (ex: 1/50) — progres capitol curent/total din carte
  return rows.map((r, idx) => {
    let colectie = "", cartea = "", capitol = "", versetul = "", marcaj = "", textVerset = "", context = "", nota = "", derventUrl = "", capitolProgres = "";

    if (maxCols >= 8) {
      colectie = (r[0] || "Vechiul Testament").toString().trim();
      cartea = (r[1] || "").toString().trim();
      capitol = (r[2] || "").toString().trim();
      versetul = (r[3] || "").toString().trim();
      marcaj = (r[4] || "").toString().trim();
      textVerset = (r[5] || "").toString().trim();
      context = (r[6] || "").toString().trim();
      nota = (r[7] || "").toString().trim();
      derventUrl = (r[8] || "").toString().trim();
      capitolProgres = (r[9] || "").toString().trim();
    } else {
      cartea = (r[0] || "").toString().trim();
      versetul = (r[1] || "").toString().trim();
      marcaj = (r[2] || "").toString().trim();
      nota = (r[3] || "").toString().trim();
      if (maxCols >= 5 && r[4]) textVerset = (r[4] || "").toString().trim();
    }

    if (!cartea && !nota) return null;
    const sters = (r[stersCols["Sters"] - 1] || "").toString().trim().toUpperCase() === "DA";
    if (sters) return null;

    return {
      id: idx + 1,
      rowId: idx + 2,
      colectie: colectie,
      cartea: cartea,
      capitol: capitol,
      versetul: versetul,
      marcaj: marcaj,
      textVerset: textVerset,
      context: context,
      nota: nota,
      derventUrl: derventUrl,
      capitolProgres: capitolProgres,
      dataModificare: r[stersCols["DataModificare"] - 1] ? parseDateToText(r[stersCols["DataModificare"] - 1]) : "",
      // Păstrăm și denumirile istorice pentru compatibilitate inversă totală:
      indice: textVerset || marcaj,
      explicatie: context || nota,
      moment: nota
    };
  }).filter(x => x !== null);
}

  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 8).getValues();
  return rows.map((r, i) => ({
    rowId: i + 2,
    modelId: (r[0] || "").toString().trim(),
    modelNume: (r[1] || "").toString().trim(),
    activ: (r[2] || "DA").toString().trim().toUpperCase() !== "NU",

  })).filter(x => x.modelId);
}

function findRowIndexById(sheet, id) {
  const last = sheet.getLastRow();
  if (last < 2) return -1;
  const ids = sheet.getRange(2, 1, last - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) {
    if (ids[i][0] && ids[i][0].toString().trim() === id.toString().trim()) return i + 2;
  }
  return -1;
}

function deleteRowById(sheet, targetVal, colIndex) {
  const maxRows = sheet.getLastRow();
  if (maxRows < 2) return;
  const values = sheet.getRange(2, colIndex, maxRows - 1, 1).getValues();
  for (let i = values.length - 1; i >= 0; i--) {
    if (values[i][0] && values[i][0].toString().trim() === targetVal.toString().trim()) {
      sheet.deleteRow(i + 2);
    }
  }
}

function logToSystemSheet(ss, d) {
  let sh = ss.getSheetByName("LOG_Sistem");
  if (!sh) {
    sh = ss.insertSheet("LOG_Sistem");
    sh.appendRow(["Timestamp", "CategorieLog", "RunId_Sesiune", "Dispozitiv_Sursa", "Modul", "Actiune", "Status", "DetaliiModificare", "MesajEroare", "StackTrace"]);
    sh.getRange("A1:J1").setFontWeight("bold");
  }
  const timestamp = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
  sh.appendRow([
    timestamp,
    d.categorie || "AUDIT_APP",
    d.runId || "-",
    d.sursa || "Browser",
    d.modul || "General",
    d.actiune || "-",
    d.status || "INFO",
    d.detalii || "",
    d.eroare || "",
    d.stack || ""
  ]);
  if (sh.getLastRow() > 2000) sh.deleteRows(2, 500);
}

function parseDateToText(val) {
  if (!val) return "";
  if (Object.prototype.toString.call(val) === '[object Date]') {
    return Utilities.formatDate(val, "Europe/Bucharest", "dd.MM.yyyy");
  }
  return val.toString().trim();
}

function parseDateToISO(val) {
  if (!val) return "";
  if (Object.prototype.toString.call(val) === '[object Date]') {
    return Utilities.formatDate(val, "Europe/Bucharest", "yyyy-MM-dd");
  }
  return val.toString().trim().substring(0, 10);
}

    const rows = sh.getLastRow() >= 2 ? sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues() : [];
    for (let i = 0; i < rows.length; i++) {
      if ((rows[i][0] || "").toString().trim() === FOLDER_KEY && rows[i][1]) {
        try {
          return DriveApp.getFolderById(rows[i][1].toString().trim());
        } catch (e) { /* folder ID invalid/șters — recreăm mai jos */ }
      }
    }
  }
  const folder = DriveApp.createFolder("DIiH_App_Fisiere");
  const settingsSheet = sh || ss.insertSheet("Setari");
  if (!sh) {
    settingsSheet.appendRow(["Cheie", "Valoare", "UltimaModificare"]);
    settingsSheet.getRange("A1:C1").setFontWeight("bold");
  }
  setSettingValue(settingsSheet, FOLDER_KEY, folder.getId());
  return folder;
}

/**
 * Primește fișiere binare (imagini) prin POST, fiindcă URL-urile GET au o limită de lungime
 * prea mică pentru poze reale. Token-ul se verifică la fel ca la doGet.
 */
function doPost(e) {
  const ss = getSpreadsheet();
  const action = e.parameter.action;

  // Adservio: date trimise din browserul părintelui (sesiune Adservio deja autentificată).
  // Nu folosește sesiunea DIiH; se autorizează cu o cheie proprie (ADSERVIO_PUSH_KEY din Proprietăți script).
  if (action === "adservioPush") {
    try {
      const rez = (typeof adservioHandlePush === "function")
        ? adservioHandlePush(ss, e)
        : { error: "Lipsește fișierul Adservio.gs din proiect." };
      return ContentService.createTextOutput(JSON.stringify(rez)).setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.message || String(err) })).setMimeType(ContentService.MimeType.JSON);
    }
  }

  const auth = authenticateRequest(ss, e);
  if (!auth.ok || !authIsActionAllowed(auth, action)) {
    return ContentService.createTextOutput(JSON.stringify({ error: auth.ok ? "Nu ai drept pentru această acțiune." : auth.error, authRequired: !auth.ok }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  if (action === "uploadDriveFile") {
    try {
      const folder = getOrCreateAppDriveFolder(ss);
      const bytes = Utilities.base64Decode(e.parameter.base64Data);
      const blob = Utilities.newBlob(bytes, e.parameter.mimeType || "image/png", e.parameter.fileName || ("file_" + Date.now()));
      const file = folder.createFile(blob);
      return ContentService.createTextOutput(JSON.stringify({ success: true, fileId: file.getId() }))
        .setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.message || String(err) }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }

  return ContentService.createTextOutput(JSON.stringify({ error: "Acțiune POST necunoscută" }))
    .setMimeType(ContentService.MimeType.JSON);
}

// Returnează conținutul unui fișier din Drive ca base64, pentru afișare în <img> cu data: URI
// (evită partajarea publică a fișierului). Doar fișierele din folderul aplicației (DIiH_App_Fisiere) pot fi citite,
// altfel oricine cu acces la API ar putea citi orice fișier din Drive-ul proprietarului, știindu-i ID-ul.
function getDriveImageAsBase64(ss, fileId) {
  try {
    const file = DriveApp.getFileById(fileId);
    const appFolderId = getOrCreateAppDriveFolder(ss).getId();
    const parents = file.getParents();
    let inAppFolder = false;
    while (parents.hasNext()) { if (parents.next().getId() === appFolderId) inAppFolder = true; }
    if (!inAppFolder) return { success: false, error: "Fișierul nu aparține folderului aplicației." };
    const blob = file.getBlob();
    return {
      success: true,
      mimeType: blob.getContentType(),
      base64: Utilities.base64Encode(blob.getBytes())
    };
  } catch (e) {
    return { success: false, error: e.message || String(e) };
  }
}

function authLevelOk(have, need) {

function authSha256(text) {
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, text, Utilities.Charset.UTF_8)
    .map(b => ((b + 256) % 256).toString(16).padStart(2, "0")).join("");
}

function authFindSession(ss, token) {
  const hash = authSha256(token);
  const cache = CacheService.getScriptCache();
  const cached = cache.get("sess_" + hash);
  if (cached) {
    const c = JSON.parse(cached);
    if (c.exp > Date.now()) return c;
  }
  const sh = getOrCreateSessionSheet(ss);
  if (sh.getLastRow() < 2) return null;
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 6).getValues();
  for (let i = rows.length - 1; i >= 0; i--) {
    if (rows[i][0] !== hash) continue;
    if ((rows[i][5] || "").toString().toUpperCase() === "DA") return null;
    const exp = rows[i][3] instanceof Date ? rows[i][3].getTime() : new Date(rows[i][3].toString().replace(" ", "T")).getTime();
    if (!exp || exp < Date.now()) return null;
    const s = { email: rows[i][1].toString().toLowerCase(), exp: exp };
    cache.put("sess_" + hash, JSON.stringify(s), Math.max(60, Math.min(21600, Math.floor((exp - Date.now()) / 1000))));
    return s;
  }
  return null;
}

// Revocă sesiunile: după email (toate ale lui) sau după hash-ul unei singure sesiuni
function authRevoke(ss, email, onlyHash) {
  const sh = getOrCreateSessionSheet(ss);
  if (sh.getLastRow() < 2) return 0;
  const cache = CacheService.getScriptCache();
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 6).getValues();
  let n = 0;
  rows.forEach((r, i) => {
    const match = onlyHash ? r[0] === onlyHash : r[1].toString().toLowerCase() === (email || "").toLowerCase();
    if (match && (r[5] || "").toString().toUpperCase() !== "DA") {
      sh.getRange(i + 2, 6).setValue("DA");
      cache.remove("sess_" + r[0]);
      n++;
    }
  });
  return n;
}

// Verifică la Google că ID token-ul e valid, emis pentru aplicația noastră, cu email confirmat
function authVerifyGoogleIdToken(idToken) {
  if (!idToken) return null;
  const res = UrlFetchApp.fetch("https://oauth2.googleapis.com/tokeninfo?id_token=" + encodeURIComponent(idToken), { muteHttpExceptions: true });
  if (res.getResponseCode() !== 200) return null;
  const p = JSON.parse(res.getContentText());
  if (p.aud !== GOOGLE_CLIENT_ID) return null;
  if (p.iss !== "accounts.google.com" && p.iss !== "https://accounts.google.com") return null;
  if (String(p.email_verified) !== "true") return null;
  if (!p.exp || parseInt(p.exp, 10) * 1000 < Date.now()) return null;
  return { email: (p.email || "").toLowerCase(), name: p.name || "" };
}

function authenticateRequest(ss, e) {
  const p = e.parameter || {};
  if (p.session) {
    const s = authFindSession(ss, p.session);
    if (!s) return { ok: false, error: "Sesiune expirată sau invalidă. Conectează-te din nou." };
    const prof = authFindProfile(ss, s.email);
    if (!prof || !prof.activ) return { ok: false, error: "Profilul nu există sau a fost dezactivat." };
    return { ok: true, legacy: false, sessionHash: authSha256(p.session), profile: prof };
  }
  return { ok: false, error: "Acces neautorizat! Conectează-te cu Google." };
}

function authIsActionAllowed(auth, action) {
  if (auth.profile.rol === "admin") return true;
  if (AUTH_ADMIN_ACTIONS.indexOf(action) !== -1) return false;
  const rule = AUTH_ACTION_RULES[action || "getAllData"];
  if (!rule) return false;
  if (rule[0] === "*") return true;
  return authLevelOk(auth.profile.permisiuni[rule[0]], rule[1]);
}

// Scoate din răspunsul de sincronizare modulele la care profilul nu are drept de vizualizare
function authFilterResponse(data, auth, ss) {

    Object.keys(AUTH_RESPONSE_MODULES).forEach(k => {
      if (k in data && !authLevelOk(auth.profile.permisiuni[AUTH_RESPONSE_MODULES[k]], "view")) data[k] = [];

  // Cumpărăturile sunt comune, în afară de cele marcate „Nu partaja” (doar ale autorului).
  // ownerNume = numele afișat al celui care a adăugat produsul (pentru notificarea „X a adăugat…”).
  if (Array.isArray(data.cumparaturi)) {
    const profiles = authGetProfiles(ss);
    data.cumparaturi = data.cumparaturi.filter(it => shopItemVisibleTo(it, auth.profile)).map(it => {
      const o = (it.owner || "").toLowerCase();
      const pr = o.indexOf("@") !== -1 ? profiles.find(x => x.email === o) : null;
      return Object.assign({}, it, { ownerNume: pr ? (pr.nume || pr.email) : (it.owner || "") });
    });
  }
  // Notițele sunt per profil: fiecare vede doar notițele lui și pe cele pe care le-a scris pentru alții
  ["ganduri", "notite"].forEach(k => {
    if (Array.isArray(data[k])) data[k] = data[k].filter(n => noteVisibleTo(n, auth.profile));
  });
  // Notificările sunt per profil: fiecare primește doar configurația lui (adminul preia, o singură dată,
  // vechea configurație comună, dacă nu și-a salvat încă una proprie)
  if (data.settings) {
    const raw = ss.getSheetByName("Setari") ? getSettingValueRaw(ss, "notif_cfg_" + auth.profile.email) : "";
    data.settings.notificationsJson = raw || (auth.profile.rol === "admin" ? data.settings.notificationsJson : "");
  }
  if ("modelCitire" in data) {
    const sp = scripturaFindRow(ss, auth.profile.email);
    data.scripturaProgres = sp ? scripturaPublic(sp) : null;
  }
  data.auth = { legacy: auth.legacy, mode: authGetMode(ss), profile: auth.legacy ? null : authPublicProfile(auth.profile) };
  return data;
}

function authHandleAction(ss, e, action, auth) {
  const p = e.parameter || {};

  if (action === "authGoogle") {
    const g = authVerifyGoogleIdToken(p.idToken);
    if (!g) return returnResponse(e, { success: false, error: "Login Google invalid sau expirat." });
    let prof = authFindProfile(ss, g.email);
    // Primul admin: dacă tabul Profile e gol, proprietarul scriptului devine admin (fără email scris în cod)
    if (!prof && authGetProfiles(ss).length === 0 && g.email === (Session.getEffectiveUser().getEmail() || "").toLowerCase()) {
      getOrCreateProfileSheet(ss).appendRow([g.email, g.name || "Admin", "admin", "DA", "", "", "", "", "", "", "", authNow()]);
      CacheService.getScriptCache().remove("auth_profiles");
      prof = authFindProfile(ss, g.email);
    }
    if (!prof || !prof.activ) {
      logToAppSheet(ss, { modul: "Auth", actiune: "login_refuzat", status: "WARN", detalii: g.email, user: g.email });
      return returnResponse(e, { success: false, error: "Contul " + g.email + " nu are acces. Cere-i administratorului să te adauge." });
    }
    const s = authCreateSession(ss, prof.email, p.device);
    logToAppSheet(ss, { modul: "Auth", actiune: "login_google", status: "OK", detalii: p.device || "", user: prof.email });
    return returnResponse(e, { success: true, sessionToken: s.token, expira: s.expira, profile: authPublicProfile(prof) });
  }

  if (action === "authPair") {
    const code = (p.code || "").toString().trim().toUpperCase();
    const cache = CacheService.getScriptCache();
    const email = code ? cache.get("pair_" + authSha256(code)) : null;
    if (!email) return returnResponse(e, { success: false, error: "Cod invalid sau expirat." });
    cache.remove("pair_" + authSha256(code));
    const prof = authFindProfile(ss, email);
    if (!prof || !prof.activ) return returnResponse(e, { success: false, error: "Profilul nu există sau a fost dezactivat." });
    const s = authCreateSession(ss, prof.email, p.device);
    logToAppSheet(ss, { modul: "Auth", actiune: "login_cod", status: "OK", detalii: p.device || "", user: prof.email });
    return returnResponse(e, { success: true, sessionToken: s.token, expira: s.expira, profile: authPublicProfile(prof) });
  }

  if (action === "authProfileNames") {
    return returnResponse(e, { success: true, profiles: authGetProfiles(ss).filter(pr => pr.activ).map(pr => ({ email: pr.email, nume: pr.nume })) });
  }

  if (action === "authMe") {
    return returnResponse(e, { success: true, mode: authGetMode(ss), legacy: auth.legacy, profile: auth.legacy ? null : authPublicProfile(auth.profile) });
  }

  if (action === "authLogout") {
    if (auth.sessionHash) authRevoke(ss, null, auth.sessionHash);
    return returnResponse(e, { success: true });
  }

  if (action === "authListProfiles") {
    const counts = {};
    const sh = getOrCreateSessionSheet(ss);
    if (sh.getLastRow() >= 2) {
      sh.getRange(2, 1, sh.getLastRow() - 1, 6).getValues().forEach(r => {
        const exp = r[3] instanceof Date ? r[3].getTime() : new Date(r[3].toString().replace(" ", "T")).getTime();
        if ((r[5] || "").toString().toUpperCase() !== "DA" && exp > Date.now()) {
          const k = r[1].toString().toLowerCase();
          counts[k] = (counts[k] || 0) + 1;
        }
      });
    }
    const list = authGetProfiles(ss).map(pr => Object.assign(authPublicProfile(pr), { activ: pr.activ, permisiuniCustom: pr.permisiuniCustom, sesiuniActive: counts[pr.email] || 0 }));
    return returnResponse(e, { success: true, mode: authGetMode(ss), modules: AUTH_MODULES, profiles: list });
  }

  if (action === "authSaveProfile") {
    const email = (p.email || "").toString().trim().toLowerCase();
    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) return returnResponse(e, { success: false, error: "Email invalid." });
    const existing = authFindProfile(ss, email);
    const rol = ["admin", "membru", "copil"].indexOf(p.rol) !== -1 ? p.rol : (existing ? existing.rol : "membru");
    const activ = p.activ !== undefined ? (p.activ === "NU" ? "NU" : "DA") : (existing && !existing.activ ? "NU" : "DA");
    if (!auth.legacy && email === auth.profile.email && (rol !== "admin" || activ === "NU")) {
      return returnResponse(e, { success: false, error: "Nu îți poți scoate singur rolul de admin sau dezactiva propriul profil." });
    }
    let perms = "";
    if (p.permisiuni) {
      try {
        const parsed = JSON.parse(p.permisiuni);
        const clean = {};
        Object.keys(parsed).forEach(k => { if (AUTH_MODULES.indexOf(k) !== -1 && ["none", "view", "edit"].indexOf(parsed[k]) !== -1) clean[k] = parsed[k]; });
        perms = JSON.stringify(clean);
      } catch (err) { return returnResponse(e, { success: false, error: "Permisiuni invalide." }); }
    }
    const sh = getOrCreateProfileSheet(ss);
    const row = [email, (p.nume || "").toString(), rol, activ, perms, (p.widgeturi || "").toString(), (p.meniu || "").toString(),
      (p.titlu || "").toString(), (p.logo || "").toString(), (p.pozaFileId || "").toString(), (p.iconaFileId || "").toString(), authNow()];
    if (existing) {
      // Câmpurile netrimise își păstrează valoarea existentă
      const old = sh.getRange(existing.rowId, 1, 1, 12).getValues()[0];
      const keys = ["email", "nume", "rol", "activ", "permisiuni", "widgeturi", "meniu", "titlu", "logo", "pozaFileId", "iconaFileId"];
      keys.forEach((k, i) => { if (i > 3 && p[k] === undefined) row[i] = old[i]; });
      if (p.nume === undefined) row[1] = old[1];
      sh.getRange(existing.rowId, 1, 1, 12).setValues([row]);
    } else {
      sh.appendRow(row);
    }
    CacheService.getScriptCache().remove("auth_profiles");
    if (activ === "NU") authRevoke(ss, email, null);
    logToAppSheet(ss, { modul: "Auth", actiune: existing ? "profil_modificat" : "profil_adaugat", status: "OK", detalii: email + " / " + rol + " / " + activ, user: auth.profile.email || "token" });
    return returnResponse(e, { success: true });
  }

  // Personalizarea proprie (titlu, logo, widgeturi, meniu, poză, iconiță) — doar pe profilul celui conectat;
  // rolul, accesul și permisiunile le schimbă numai adminul, prin authSaveProfile
  if (action === "authSaveMyProfile") {
    const me = authFindProfile(ss, auth.profile.email);
    if (!me) return returnResponse(e, { success: false, error: "Profil inexistent." });
    const sh = getOrCreateProfileSheet(ss);
    const cols = { widgeturi: 6, meniu: 7, titlu: 8, logo: 9, pozaFileId: 10, iconaFileId: 11 };
    Object.keys(cols).forEach(k => {
      if (p[k] !== undefined) sh.getRange(me.rowId, cols[k]).setValue(String(p[k]).slice(0, 1000));
    });
    sh.getRange(me.rowId, 12).setValue(authNow());
    CacheService.getScriptCache().remove("auth_profiles");
    return returnResponse(e, { success: true });
  }

  if (action === "authRevokeSessions") {
    const n = authRevoke(ss, (p.email || "").toString(), null);
    logToAppSheet(ss, { modul: "Auth", actiune: "sesiuni_revocate", status: "OK", detalii: (p.email || "") + " (" + n + ")", user: auth.profile.email || "token" });
    return returnResponse(e, { success: true, revocate: n });
  }

  if (action === "authCreatePairCode") {
    const prof = authFindProfile(ss, p.email);
    if (!prof || !prof.activ) return returnResponse(e, { success: false, error: "Profil inexistent sau inactiv." });
    const alphabet = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
    let code = "";
    for (let i = 0; i < 8; i++) code += alphabet.charAt(Math.floor(Math.random() * alphabet.length));
    CacheService.getScriptCache().put("pair_" + authSha256(code), prof.email, PAIR_CODE_MINUTES * 60);
    logToAppSheet(ss, { modul: "Auth", actiune: "cod_asociere", status: "OK", detalii: prof.email, user: auth.profile.email || "token" });
    return returnResponse(e, { success: true, code: code, expiraMinute: PAIR_CODE_MINUTES });
  }

  if (action === "authSetMode") {
    const mode = p.mode === "strict" ? "strict" : "tranzitie";
    // Modul strict se poate porni doar dintr-o sesiune Google de admin, ca să nu te blochezi singur pe dinafară
    if (mode === "strict" && auth.legacy) return returnResponse(e, { success: false, error: "Pornește modul strict după ce te-ai conectat cu Google." });
    let sh = ss.getSheetByName("Setari");
    if (!sh) {
      sh = ss.insertSheet("Setari");
      sh.appendRow(["Cheie", "Valoare", "UltimaModificare"]);
    }
    setSettingValue(sh, "auth_mode", mode);
    CacheService.getScriptCache().remove("auth_mode");
    logToAppSheet(ss, { modul: "Auth", actiune: "mod_" + mode, status: "OK", detalii: "", user: auth.profile.email || "token" });
    return returnResponse(e, { success: true, mode: mode });
  }

  return returnResponse(e, { success: false, error: "Acțiune de autentificare necunoscută." });
}

// =====================================================================================================
// COPII DE SIGURANȚĂ (v24.4) — o copie a întregului sheet în Drive, săptămânal, păstrând ultimele BACKUP_COPII
// Copia e un fișier NOU (makeCopy); sheet-ul original doar se citește, nu se modifică.
// Copiile stau în subfolderul „Backup” din DIiH_App_Fisiere; cele vechi ajung în Coșul Drive (30 de zile).
// Pornire: rulează O DATĂ din editor instaleazaBackupSaptamanal() (duminică, 03:00, ora României).
// =====================================================================================================
const BACKUP_COPII = 8;
const BACKUP_PREFIX = "DIiH_Backup_";

function backupReadSetting(ss, key) {
  const sh = ss.getSheetByName("Setari");
  if (!sh || sh.getLastRow() < 2) return "";
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues();
  for (let i = 0; i < rows.length; i++) {
    if ((rows[i][0] || "").toString().trim() === key) return (rows[i][1] || "").toString().trim();
  }
  return "";
}

function getOrCreateBackupFolder(ss) {
  const id = backupReadSetting(ss, "backup_folder_id");
  if (id) {
    try {
      const f = DriveApp.getFolderById(id);
      if (!f.isTrashed()) return f;
    } catch (e) { /* folder șters — îl recreăm */ }
  }
  const folder = getOrCreateAppDriveFolder(ss).createFolder("Backup");
  folder.setDescription("Copii de siguranță ale registrului DIiH (automat, săptămânal). Se păstrează ultimele " + BACKUP_COPII + ".");
  setSettingValue(ss.getSheetByName("Setari"), "backup_folder_id", folder.getId());
  return folder;
}

function backupListeaza(folder) {
  const list = [];
  const it = folder.getFiles();
  while (it.hasNext()) {
    const f = it.next();
    if (f.getName().indexOf(BACKUP_PREFIX) === 0) list.push(f);
  }
  return list.sort((a, b) => b.getDateCreated() - a.getDateCreated());
}

function backupCreeazaCopie(ss, motiv) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(30000)) throw new Error("O altă copie e în curs. Încearcă peste un minut.");
  try {
    const folder = getOrCreateBackupFolder(ss);
    const nume = BACKUP_PREFIX + Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd_HHmm");
    const copie = DriveApp.getFileById(ss.getId()).makeCopy(nume, folder);
    const vechi = backupListeaza(folder).slice(BACKUP_COPII);
    vechi.forEach(f => f.setTrashed(true));
    setSettingValue(ss.getSheetByName("Setari"), "backup_ultim", authNow() + " (" + motiv + ")");
    logToAppSheet(ss, { modul: "Backup", actiune: "copie_sheet", status: "OK", detalii: nume + "; " + motiv + "; mutate în Coș: " + vechi.length, user: "-" });
    return { id: copie.getId(), nume: nume, sterse: vechi.length };
  } finally {
    lock.releaseLock();
  }
}

function backupStatus(ss) {
  const folder = getOrCreateBackupFolder(ss);
  const copii = backupListeaza(folder).map(f => ({
    id: f.getId(),
    nume: f.getName(),
    data: Utilities.formatDate(f.getDateCreated(), "Europe/Bucharest", "yyyy-MM-dd HH:mm")
  }));
  const programat = ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === "backupSaptamanal");
  return { success: true, copii: copii, pastrate: BACKUP_COPII, programat: programat, ultim: backupReadSetting(ss, "backup_ultim"), folderId: folder.getId() };
}

// Rulată de trigger (duminică noaptea)
function backupSaptamanal() {
  try {
    backupCreeazaCopie(getSpreadsheet(), "automat, săptămânal");
  } catch (err) {
    logToAppSheet(getSpreadsheet(), { modul: "Backup", actiune: "copie_sheet", status: "EROARE", detalii: err.message || String(err), user: "-" });
    throw err;
  }
}

// Rulează O DATĂ din editor (Run). Atinge doar triggerul backupSaptamanal; celelalte triggere rămân neschimbate.
function instaleazaBackupSaptamanal() {
  ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === "backupSaptamanal") ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger("backupSaptamanal").timeBased().onWeekDay(ScriptApp.WeekDay.SUNDAY).atHour(3).inTimezone("Europe/Bucharest").create();
  const r = backupCreeazaCopie(getSpreadsheet(), "prima copie, la instalare");
  Logger.log("Backup programat: duminică, 03:00. Prima copie: " + r.nume + ". Se păstrează ultimele " + BACKUP_COPII + ".");
}

// =====================================================================================================
// PROGRES SCRIPTURĂ PER PROFIL (v24.1) — tabul ProgresScriptura, un rând per profil
// Capitole = listă compactă „indexCarte:capitole” (ex. „0:1-50|1:1-3,5”), indexul fiind ordinea canonului din aplicație.
// Istoric = lecturile anterioare (JSON); ultimele copii păstrează și capitolele, pentru restaurare.
// =====================================================================================================
const SCRIPTURA_PROGRES_COLUMNS = ["Email", "Capitole", "ModelActiv", "NrLectura", "InceputLa", "Istoric", "Actualizat"];
const SCRIPTURA_ISTORIC_COPII = 5;

function getOrCreateScripturaProgresSheet(ss) {
  let sh = ss.getSheetByName("ProgresScriptura");
  if (!sh) {
    sh = ss.insertSheet("ProgresScriptura");
    sh.appendRow(SCRIPTURA_PROGRES_COLUMNS);
    sh.getRange("A1:G1").setFontWeight("bold");
  }
  return sh;
}

function scripturaFindRow(ss, email) {
  const sh = ss.getSheetByName("ProgresScriptura");
  const em = (email || "").toString().trim().toLowerCase();
  if (!sh || sh.getLastRow() < 2 || !em) return null;
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 7).getValues();
  for (let i = 0; i < rows.length; i++) {
    if ((rows[i][0] || "").toString().trim().toLowerCase() !== em) continue;
    let istoric = [];
    try { istoric = rows[i][5] ? JSON.parse(rows[i][5].toString()) : []; } catch (err) { istoric = []; }
    return {
      rowId: i + 2,
      email: em,
      capitole: (rows[i][1] || "").toString(),
      modelActiv: (rows[i][2] || "").toString(),
      nrLectura: parseInt(rows[i][3], 10) || 1,
      inceputLa: rows[i][4] ? (rows[i][4] instanceof Date ? Utilities.formatDate(rows[i][4], "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss") : rows[i][4].toString()) : "",
      istoric: Array.isArray(istoric) ? istoric : [],
      actualizat: rows[i][6] ? rows[i][6].toString() : ""
    };
  }
  return null;
}

function scripturaGetRow(ss, email) {
  return scripturaFindRow(ss, email) || { rowId: 0, email: (email || "").toString().trim().toLowerCase(), capitole: "", modelActiv: "", nrLectura: 1, inceputLa: "", istoric: [], actualizat: "" };
}

function scripturaSaveRow(ss, row) {
  const sh = getOrCreateScripturaProgresSheet(ss);
  // Doar ultimele copii păstrează capitolele (limita de caractere a unei celule)
  row.istoric.forEach((h, i) => { if (i < row.istoric.length - SCRIPTURA_ISTORIC_COPII) delete h.capitole; });
  row.actualizat = authNow();
  const values = [row.email, row.capitole, row.modelActiv, row.nrLectura, row.inceputLa, JSON.stringify(row.istoric), row.actualizat];
  if (row.rowId) sh.getRange(row.rowId, 1, 1, 7).setValues([values]);
  else { sh.appendRow(values); row.rowId = sh.getLastRow(); }
}

function scripturaArchive(ss, email, motiv) {
  const row = scripturaGetRow(ss, email);
  if (row.capitole || row.inceputLa) {
    row.istoric.push({ nr: row.nrLectura, model: row.modelActiv, inceput: row.inceputLa, sfarsit: authNow(), motiv: motiv, capitole: row.capitole });
    row.nrLectura = row.nrLectura + 1;
  }
  row.capitole = "";
  row.inceputLa = authNow();
  scripturaSaveRow(ss, row);
  return row;
}

function scripturaPublic(row) {
  return {
    capitole: row.capitole, modelActiv: row.modelActiv, nrLectura: row.nrLectura, inceputLa: row.inceputLa, actualizat: row.actualizat,
    istoric: row.istoric.map(h => ({ nr: h.nr, model: h.model, inceput: h.inceput, sfarsit: h.sfarsit, motiv: h.motiv, areCopie: h.capitole !== undefined }))
  };
}
