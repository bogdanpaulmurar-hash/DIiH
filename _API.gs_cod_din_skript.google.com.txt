const API_SECURITY_TOKEN = "diih_secret_2026_bm";
const SPREADSHEET_ID = "107cZOxLdMMTuOsytriNmZt_d3FM0Nwz1z2ykzpnCcfY";

function getSpreadsheet() {
  try {
    return SpreadsheetApp.openById(SPREADSHEET_ID);
  } catch (e) {
    return SpreadsheetApp.getActiveSpreadsheet();
  }
}

function doGet(e) {
  const token = e.parameter.token;
  if (token !== API_SECURITY_TOKEN) {
    return ContentService.createTextOutput(JSON.stringify({ error: "Acces neautorizat!" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const ss = getSpreadsheet();
  const action = e.parameter.action;

  // 1. AUDIT AVANSAT ÎN LOG_Sistem
  if (action === "logEvent") {
    logToSystemSheet(ss, {
      categorie: e.parameter.category || "AUDIT_APP",
      runId: e.parameter.runId || "-",
      sursa: e.parameter.deviceName || e.parameter.deviceId || "App",
      modul: e.parameter.entity || "General",
      actiune: e.parameter.appAction || "-",
      status: e.parameter.status || "INFO",
      detalii: e.parameter.message || "",
      eroare: e.parameter.errorMessage || "",
      stack: e.parameter.stack || ""
    });
    return returnResponse(e, { success: true });
  }

  // 1b. AUDIT LA NIVEL DE APLICAȚIE ÎN LOG_app (separat de LOG_Sistem; folosit ex. la revenirea unei plăți pe neachitat)
  if (action === "logAppEvent") {
    logToAppSheet(ss, {
      modul: e.parameter.modul || "General",
      actiune: e.parameter.actiune || "-",
      status: e.parameter.status || "INFO",
      detalii: e.parameter.detalii || "",
      user: e.parameter.user || "-"
    });
    return returnResponse(e, { success: true });
  }

  // 2. SETĂRI GLOBALE (Inclusiv Număr Persoane Pomelnic Zilnic, Iconiță & Configurare Partajare)
  if (action === "updateGlobalSettings") {
    let sh = ss.getSheetByName("Setari");
    if (!sh) {
      sh = ss.insertSheet("Setari");
      sh.appendRow(["Cheie", "Valoare", "UltimaModificare"]);
      sh.getRange("A1:C1").setFontWeight("bold");
    }
    if (e.parameter.appTitle) setSettingValue(sh, "app_title", e.parameter.appTitle);
    if (e.parameter.appLogo) setSettingValue(sh, "app_logo", e.parameter.appLogo);
    if (e.parameter.appIconPath) setSettingValue(sh, "app_icon_path", e.parameter.appIconPath);
    if (e.parameter.pomelnicBatchSize) setSettingValue(sh, "pomelnic_batch_size", e.parameter.pomelnicBatchSize);
    if (e.parameter.psaltireBatchSize) setSettingValue(sh, "psaltire_batch_size", e.parameter.psaltireBatchSize);
    if (e.parameter.sharingConfigJson) setSettingValue(sh, "sharing_config_json", e.parameter.sharingConfigJson);
    if (e.parameter.randuialaTrackerJson) setSettingValue(sh, "randuiala_tracker_json", e.parameter.randuialaTrackerJson);
    if (e.parameter.userProfilesJson) setSettingValue(sh, "user_profiles_json", e.parameter.userProfilesJson);
    return returnResponse(e, { success: true });
  }

  // 2b. SETARE / CITIRE CHEIE ARBITRARĂ ÎN Setari (folosit de funcții noi: ModelCitire activ, iconiță/logo per profil,
  //     calendar școlar per an, repere zoom, date Săptămâna Verde/Altfel) — fără să modifice cheile cunoscute existente
  if (action === "setSettingRaw") {
    let sh = ss.getSheetByName("Setari");
    if (!sh) {
      sh = ss.insertSheet("Setari");
      sh.appendRow(["Cheie", "Valoare", "UltimaModificare"]);
      sh.getRange("A1:C1").setFontWeight("bold");
    }
    if (e.parameter.key) setSettingValue(sh, e.parameter.key, e.parameter.value || "");
    return returnResponse(e, { success: true });
  }

  if (action === "getSettingsRaw") {
    return returnResponse(e, { settingsRaw: getAllSettingsRaw(ss) });
  }

  // Returnează o imagine salvată în Drive (calendar școlar, iconiță/logo custom) ca base64, pentru <img src="data:...">
  if (action === "getDriveImage") {
    return returnResponse(e, getDriveImageAsBase64(ss, e.parameter.fileId));
  }

  // 3. SALVARE ACTIVITĂȚI ȘCOALĂ (Păstrează structura de 14 coloane a foii existente + stocare metadate în Col 13
  //    + coloane NOI O/P: Data_Achitat, Comentariu_Plata)
  if (action === "saveSchoolActivities") {
    let sh = ss.getSheetByName("Scoala_Activitati") || ss.getSheetByName("Activitati_Scoala");
    if (!sh) {
      sh = ss.insertSheet("Scoala_Activitati");
      sh.appendRow(["ID", "Data", "Copil", "Institutie", "Titlu", "TipPlata", "Ore", "TarifOra", "TotalPlata", "StatusPlata", "TermenPlata", "MesajZi", "RemindereJson", "DataCreare"]);
      sh.getRange("A1:N1").setFontWeight("bold");
    }
    ensureSheetColumns(sh, ["Data_Achitat", "Comentariu_Plata"]);
    const activities = JSON.parse(e.parameter.data || "[]");
    const dataCreare = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");

    activities.forEach(item => {
      deleteRowById(sh, item.id, 1);
      // Păstrăm metadatele extinse (culoare, ore început/sfârșit, autor) în Col 13 (RemindereJson)
      const metaObj = {
        reminders: item.reminders || [60, 30, 10],
        color: item.color || "auto",
        startHour: item.startHour || "08:00",
        endHour: item.endHour || "16:00",
        author: item.author || "Bogdan"
      };

      sh.appendRow([
        item.id,
        item.date,
        item.child || "Copil",
        item.institution || "-",
        item.title,
        item.paymentType || "Pe oră",
        item.hours || 1,
        item.rate || 0,
        item.total || 0,
        item.paymentStatus || "În așteptare",
        item.deadline || "-",
        item.dayMessage || "",
        JSON.stringify(metaObj),
        dataCreare,
        item.dataAchitat || "",
        item.comentariuPlata || ""
      ]);
    });
    return returnResponse(e, { success: true, count: activities.length });
  }

  if (action === "deleteSchoolActivity") {
    let sh = ss.getSheetByName("Scoala_Activitati") || ss.getSheetByName("Activitati_Scoala");
    if (sh && e.parameter.id) deleteRowById(sh, e.parameter.id, 1);
    return returnResponse(e, { success: true });
  }

  // Actualizare în calup status plată (pentru restanțe rezolvate direct din sumar)
  if (action === "batchUpdateSchoolStatus") {
    let sh = ss.getSheetByName("Scoala_Activitati") || ss.getSheetByName("Activitati_Scoala");
    const ids = JSON.parse(e.parameter.ids || "[]");
    const newStatus = e.parameter.status || "Achitat";
    // dataAchitat/comentariuPlata: completate la achitare; golite (string gol) la revenire la neachitat
    const dataAchitat = e.parameter.dataAchitat !== undefined ? e.parameter.dataAchitat : null;
    const comentariuPlata = e.parameter.comentariuPlata !== undefined ? e.parameter.comentariuPlata : null;
    if (sh && ids.length > 0) {
      const cols = ensureSheetColumns(sh, ["Data_Achitat", "Comentariu_Plata"]);
      const maxRows = sh.getLastRow();
      if (maxRows >= 2) {
        const idCol = sh.getRange(2, 1, maxRows - 1, 1).getValues();
        for (let i = 0; i < idCol.length; i++) {
          if (ids.indexOf(String(idCol[i][0]).trim()) !== -1) {
            sh.getRange(i + 2, 10).setValue(newStatus); // Col J = StatusPlata
            if (dataAchitat !== null) sh.getRange(i + 2, cols["Data_Achitat"]).setValue(dataAchitat);
            if (comentariuPlata !== null) sh.getRange(i + 2, cols["Comentariu_Plata"]).setValue(comentariuPlata);
          }
        }
      }
    }
    return returnResponse(e, { success: true });
  }

  // 4. CRUD FOAIA Psaltire_Grup
  if (action === "addPsaltirePerson") {
    let sh = ss.getSheetByName("Psaltire_Grup");
    if (!sh) {
      sh = ss.insertSheet("Psaltire_Grup");
      sh.appendRow(["Sectiune", "Subgrup_Coloana", "Numar_Rand", "Catisma", "Persoana_Raw", "Exclus_La_Pomenire"]);
      sh.getRange("A1:F1").setFontWeight("bold");
    }
    sh.appendRow([
      e.parameter.sectiune || "Grup Psaltire",
      e.parameter.subgrup || "Coloana 1",
      parseInt(e.parameter.numarRand, 10) || sh.getLastRow(),
      e.parameter.catisma || "",
      e.parameter.nume,
      e.parameter.exclus === "true" ? "DA" : "NU"
    ]);
    return returnResponse(e, { success: true });
  }

  if (action === "updatePsaltirePerson") {
    let sh = ss.getSheetByName("Psaltire_Grup");
    let rowId = parseInt(e.parameter.rowId, 10);
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      if (e.parameter.nume) sh.getRange(rowId, 5).setValue(e.parameter.nume);
      if (e.parameter.sectiune) sh.getRange(rowId, 1).setValue(e.parameter.sectiune);
      if (e.parameter.subgrup) sh.getRange(rowId, 2).setValue(e.parameter.subgrup);
      if (e.parameter.catisma) sh.getRange(rowId, 4).setValue(e.parameter.catisma);
      if (e.parameter.exclus !== undefined) sh.getRange(rowId, 6).setValue(e.parameter.exclus === "true" ? "DA" : "NU");
      return returnResponse(e, { success: true });
    }
  }

  if (action === "deletePsaltirePerson") {
    let sh = ss.getSheetByName("Psaltire_Grup");
    let rowId = parseInt(e.parameter.rowId, 10);
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      sh.deleteRow(rowId);
      return returnResponse(e, { success: true });
    }
  }

  // 5. NOTIȚE & GÂNDURI (Suport pentru tab separat "Notite" sau fallback pe "Ganduri")
  if (action === "addGand" || action === "addNote" || action === "addNotita") {
    let sh = ss.getSheetByName("Notite") || ss.getSheetByName("Ganduri");
    if (!sh) {
      sh = ss.insertSheet("Notite");
      sh.appendRow(["ID", "Gand", "Sursa", "DataCreare"]);
      sh.getRange("A1:D1").setFontWeight("bold");
    }
    const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
    const dataCreare = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
    const noteContent = e.parameter.gand || e.parameter.nota || e.parameter.notita || e.parameter.text || "";
    if (header === "ID") {
      const gandId = e.parameter.id || ("NOTE-" + Date.now());
      sh.appendRow([
        gandId,
        noteContent,
        e.parameter.sursa || "Personal",
        dataCreare,
        e.parameter.user || e.parameter.author || ""
      ]);
    } else {
      sh.appendRow([
        noteContent,
        e.parameter.sursa || "Personal",
        dataCreare,
        e.parameter.user || e.parameter.author || ""
      ]);
    }
    return returnResponse(e, { success: true });
  }

  if (action === "updateGand" || action === "updateNote" || action === "updateNotita") {
    let sh = ss.getSheetByName("Notite") || ss.getSheetByName("Ganduri");
    let rowId = parseInt(e.parameter.rowId, 10);
    const noteContent = e.parameter.gand || e.parameter.nota || e.parameter.notita || e.parameter.text || "";
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
      const colOffset = (header === "ID") ? 1 : 0;
      sh.getRange(rowId, 1 + colOffset).setValue(noteContent);
      if (e.parameter.sursa) sh.getRange(rowId, 2 + colOffset).setValue(e.parameter.sursa);
      return returnResponse(e, { success: true });
    }
  }

  if (action === "deleteGand" || action === "deleteNote" || action === "deleteNotita") {
    let sh = ss.getSheetByName("Notite") || ss.getSheetByName("Ganduri");
    if (!sh || sh.getLastRow() < 2) return returnResponse(e, { success: true });
    
    let targetGand = (e.parameter.gand || e.parameter.nota || e.parameter.notita || e.parameter.text || "").toString().trim();
    let rowId = parseInt(e.parameter.rowId, 10);
    let deleted = false;

    // 1. Căutăm după textul gândului / notiței (asigură ștergerea corectă chiar dacă rândurile s-au decalat)
    if (targetGand) {
      const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
      const col = (header === "ID") ? 2 : 1;
      const vals = sh.getRange(2, col, sh.getLastRow() - 1, 1).getValues();
      for (let i = vals.length - 1; i >= 0; i--) {
        if (vals[i][0] && vals[i][0].toString().trim() === targetGand) {
          sh.deleteRow(i + 2);
          deleted = true;
          break;
        }
      }
    }

    // 2. Fallback după rowId dacă nu s-a găsit după text
    if (!deleted && !isNaN(rowId) && rowId >= 2 && rowId <= sh.getLastRow()) {
      sh.deleteRow(rowId);
      deleted = true;
    }

    return returnResponse(e, { success: true, deleted: deleted });
  }

  // 5b. ȘTERGERE SOFT NOTIȚE (marcaj Sters=DA/NU, nu elimină rânduri; suportă selecție multiplă via rowIds="2,5,7")
  if (action === "softDeleteGand" || action === "restoreGand") {
    let sh = ss.getSheetByName("Notite") || ss.getSheetByName("Ganduri");
    if (!sh) return returnResponse(e, { success: true });
    const cols = getNoteStersColumns(sh);
    const isDelete = (action === "softDeleteGand");
    const timestamp = isDelete ? Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss") : "";
    const rowIdsStr = e.parameter.rowIds || e.parameter.rowId || "";
    const ids = rowIdsStr.toString().split(",").map(x => parseInt(x.trim(), 10)).filter(n => !isNaN(n));
    ids.forEach(rId => {
      if (rId >= 2 && rId <= sh.getLastRow()) {
        sh.getRange(rId, cols.stersCol).setValue(isDelete ? "DA" : "NU");
        sh.getRange(rId, cols.dataStergereCol).setValue(timestamp);
      }
    });
    return returnResponse(e, { success: true, count: ids.length });
  }

  // 6. CUMPĂRĂTURI (Structură suportată: cu ID pe Col 1 sau direct Magazin pe Col 1; suport autor & opțiune "Nu partaja")
  if (action === "addCumparaturi") {
    let sh = ss.getSheetByName("lista_cumparaturi");
    if (!sh) {
      sh = ss.insertSheet("lista_cumparaturi");
      sh.appendRow(["ID", "Magazin", "Produs", "Cantitate", "DataCreare", "NuPartaja", "Owner"]);
      sh.getRange("A1:G1").setFontWeight("bold");
    }
    const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
    const dataCreare = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
    const author = e.parameter.author ? ` (${e.parameter.author})` : "";
    const produsComplet = (e.parameter.produs || "") + author;
    const nuPartaja = (e.parameter.nuPartaja === "true" || e.parameter.nuPartaja === true) ? "DA" : "NU";
    const owner = e.parameter.owner || e.parameter.user || e.parameter.author || "";
    if (header === "ID") {
      const shopId = e.parameter.id || ("SHOP-" + Date.now());
      sh.appendRow([
        shopId,
        e.parameter.magazin || "General",
        produsComplet,
        e.parameter.cantitate || "1",
        dataCreare,
        nuPartaja,
        owner
      ]);
    } else {
      sh.appendRow([
        e.parameter.magazin || "General",
        produsComplet,
        e.parameter.cantitate || "1",
        nuPartaja,
        owner
      ]);
    }
    return returnResponse(e, { success: true });
  }

  if (action === "deleteCumparaturi") {
    let sh = ss.getSheetByName("lista_cumparaturi");
    if (!sh || sh.getLastRow() < 2) return returnResponse(e, { success: true });

    let targetProdus = (e.parameter.produs || "").toString().trim();
    let targetMagazin = (e.parameter.magazin || "").toString().trim();
    let rowId = parseInt(e.parameter.rowId, 10);
    let deleted = false;

    // 1. Căutăm după produs și magazin (asigură ștergerea corectă chiar dacă rândurile s-au decalat)
    if (targetProdus) {
      const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
      const hasIdCol = (header === "ID");
      const magazinCol = hasIdCol ? 2 : 1;
      const produsCol = hasIdCol ? 3 : 2;
      const maxRows = sh.getLastRow() - 1;
      const vals = sh.getRange(2, 1, maxRows, hasIdCol ? 3 : 2).getValues();
      for (let i = vals.length - 1; i >= 0; i--) {
        let pVal = (vals[i][produsCol - 1] || "").toString().trim();
        let mVal = (vals[i][magazinCol - 1] || "").toString().trim();
        if (pVal === targetProdus || pVal.startsWith(targetProdus) || (targetMagazin && mVal === targetMagazin && pVal.includes(targetProdus))) {
          sh.deleteRow(i + 2);
          deleted = true;
          break;
        }
      }
    }

    // 2. Fallback după rowId dacă nu s-a găsit după conținut
    if (!deleted && !isNaN(rowId) && rowId >= 2 && rowId <= sh.getLastRow()) {
      sh.deleteRow(rowId);
      deleted = true;
    }

    return returnResponse(e, { success: true, deleted: deleted });
  }

  if (action === "deleteMagazinCumparaturi") {
    let sh = ss.getSheetByName("lista_cumparaturi");
    const magazin = e.parameter.magazin;
    if (sh && magazin && sh.getLastRow() >= 2) {
      const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
      const magazinCol = (header === "ID") ? 2 : 1;
      const vals = sh.getRange(2, magazinCol, sh.getLastRow() - 1, 1).getValues();
      for (let i = vals.length - 1; i >= 0; i--) {
        if (vals[i][0] && vals[i][0].toString().trim() === magazin.toString().trim()) {
          sh.deleteRow(i + 2);
        }
      }
    }
    return returnResponse(e, { success: true });
  }

  // 7. CRUD COMPLET PERSOANĂ POMELNIC (Adăugare pe ultimul rând & Editare completă)
  if (action === "addPerson") {
    let sh = ss.getSheetByName("Pomelnic");
    if (!sh) {
      return returnResponse(e, { error: "Foaia Pomelnic nu a fost găsită" });
    }
    const newRowId = sh.getLastRow() + 1;
    const isActiv = (e.parameter.isActivPomelnic === "true" || e.parameter.isActivPomelnic === true || e.parameter.isActivPomelnic === "1" || e.parameter.isActivPomelnic === 1);
    const stare = (e.parameter.stare || "Viu").toString().trim();
    const van = (stare === "Adormit") ? "A" : "V";
    const necesitaAtentie = (e.parameter.necesitaAtentie === "true" || e.parameter.necesitaAtentie === true);
    const bolnav = (e.parameter.bolnav === "true" || e.parameter.bolnav === true);
    const dnAprox = (e.parameter.dnAprox === "true" || e.parameter.dnAprox === true);
    const daAprox = (e.parameter.daAprox === "true" || e.parameter.daAprox === true);
    const pomelnicAndreea = (e.parameter.pomelnicAndreea === "true" || e.parameter.pomelnicAndreea === true);

    sh.appendRow([
      newRowId,                                       // Col 1 (A): RowID
      van,                                            // Col 2 (B): (F) V/A/N
      isActiv,                                        // Col 3 (C): (F) Activ_Inactiv/calcul_pomelnic
      pomelnicAndreea,                                // Col 4 (D): Pomelnic_Andreea
      false,                                          // Col 5 (E): Pomelnic_calendar
      (e.parameter.prenume || "").trim(),             // Col 6 (F): Prenume
      (e.parameter.nume || "").trim(),                // Col 7 (G): Nume
      (e.parameter.info || "").trim(),                // Col 8 (H): Informatii
      (e.parameter.istoricMedical || "").trim(),      // Col 9 (I): Istoric medical
      (e.parameter.dn || "").trim(),                  // Col 10 (J): Data nasterii
      dnAprox,                                        // Col 11 (K): D A N A
      (e.parameter.da || "").trim(),                  // Col 12 (L): Data adormirii
      daAprox,                                        // Col 13 (M): D A D A
      bolnav,                                         // Col 14 (N): Bolnav
      stare,                                          // Col 15 (O): Starea sufleteasca
      (e.parameter.grad || "").trim(),                // Col 16 (P): Grad (Coloana P!)
      (e.parameter.copii || "-").trim(),              // Col 17 (Q): Copil
      (e.parameter.sot || "-").trim(),                // Col 18 (R): Sot/Sotie
      (e.parameter.parinti || "-").trim(),            // Col 19 (S): Parinti
      (e.parameter.apropiati || "-").trim(),          // Col 20 (T): Frati, bunici, apropiati
      "09:00",                                        // Col 21 (U): StartTime
      "10:00",                                        // Col 22 (V): EndTime
      necesitaAtentie                                 // Col 23 (W): NECESITA_Atentie!!!
    ]);
    return returnResponse(e, { success: true, rowId: newRowId });
  }

  if (action === "updatePerson") {
    let sh = ss.getSheetByName("Pomelnic");
    let rowId = parseInt(e.parameter.rowId, 10);
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      if (e.parameter.prenume !== undefined) sh.getRange(rowId, 6).setValue(e.parameter.prenume);
      if (e.parameter.nume !== undefined) sh.getRange(rowId, 7).setValue(e.parameter.nume);
      if (e.parameter.info !== undefined) sh.getRange(rowId, 8).setValue(e.parameter.info);
      if (e.parameter.istoricMedical !== undefined) sh.getRange(rowId, 9).setValue(e.parameter.istoricMedical);
      if (e.parameter.dn !== undefined) sh.getRange(rowId, 10).setValue(e.parameter.dn);
      if (e.parameter.dnAprox !== undefined) {
        const dnVal = (e.parameter.dnAprox === "true" || e.parameter.dnAprox === true);
        sh.getRange(rowId, 11).setValue(dnVal);
      }
      if (e.parameter.da !== undefined) sh.getRange(rowId, 12).setValue(e.parameter.da);
      if (e.parameter.daAprox !== undefined) {
        const daVal = (e.parameter.daAprox === "true" || e.parameter.daAprox === true);
        sh.getRange(rowId, 13).setValue(daVal);
      }
      if (e.parameter.bolnav !== undefined) {
        const bVal = (e.parameter.bolnav === "true" || e.parameter.bolnav === true);
        sh.getRange(rowId, 14).setValue(bVal);
      }
      if (e.parameter.stare !== undefined) {
        sh.getRange(rowId, 15).setValue(e.parameter.stare);
        sh.getRange(rowId, 2).setValue(e.parameter.stare === "Adormit" ? "A" : "V");
      }
      if (e.parameter.grad !== undefined) sh.getRange(rowId, 16).setValue(e.parameter.grad);
      if (e.parameter.copii !== undefined) sh.getRange(rowId, 17).setValue(e.parameter.copii);
      if (e.parameter.sot !== undefined) sh.getRange(rowId, 18).setValue(e.parameter.sot);
      if (e.parameter.parinti !== undefined) sh.getRange(rowId, 19).setValue(e.parameter.parinti);
      if (e.parameter.apropiati !== undefined) sh.getRange(rowId, 20).setValue(e.parameter.apropiati);
      if (e.parameter.necesitaAtentie !== undefined) {
        const valBool = (e.parameter.necesitaAtentie === "true" || e.parameter.necesitaAtentie === true);
        sh.getRange(rowId, 23).setValue(valBool);
      }
      if (e.parameter.isActivPomelnic !== undefined) {
        const valAct = (e.parameter.isActivPomelnic === "true" || e.parameter.isActivPomelnic === true);
        sh.getRange(rowId, 3).setValue(valAct);
      }
      if (e.parameter.pomelnicAndreea !== undefined) {
        const valAndreea = (e.parameter.pomelnicAndreea === "true" || e.parameter.pomelnicAndreea === true || e.parameter.pomelnicAndreea === "1" || e.parameter.pomelnicAndreea === 1);
        sh.getRange(rowId, 4).setValue(valAndreea);
      }
      return returnResponse(e, { success: true });
    }
  }

  // Actualizare în calup Pomelnic Andreea (Coloana D)
  if (action === "batchUpdateAndreea") {
    let sh = ss.getSheetByName("Pomelnic");
    const rowIdsStr = e.parameter.rowIds || "";
    const flagValue = (e.parameter.pomelnicAndreea === "true" || e.parameter.pomelnicAndreea === true);
    if (sh && rowIdsStr) {
      const ids = rowIdsStr.split(",").map(x => parseInt(x.trim(), 10)).filter(n => !isNaN(n));
      ids.forEach(rId => {
        if (rId >= 2 && rId <= sh.getLastRow()) {
          sh.getRange(rId, 4).setValue(flagValue);
        }
      });
      return returnResponse(e, { success: true, count: ids.length, flag: flagValue });
    }
    return returnResponse(e, { error: "Parametri lipsă" });
  }

  // Actualizare în calup Necesită Atenție (Pomelnic - Coloana W = 23)
  if (action === "batchUpdateAtentie") {
    let sh = ss.getSheetByName("Pomelnic");
    const rowIdsStr = e.parameter.rowIds || "";
    const flagValue = (e.parameter.necesitaAtentie === "true" || e.parameter.necesitaAtentie === true);
    if (sh && rowIdsStr) {
      const ids = rowIdsStr.split(",").map(x => parseInt(x.trim(), 10)).filter(n => !isNaN(n));
      ids.forEach(rId => {
        if (rId >= 2 && rId <= sh.getLastRow()) {
          sh.getRange(rId, 23).setValue(flagValue);
        }
      });
    }
    return returnResponse(e, { success: true });
  }

  // 8. SALVARE CONFIGURĂRI NOTIFICĂRI ÎN SHEET
  if (action === "saveNotificationConfigs") {
    let sh = ss.getSheetByName("Setari");
    if (!sh) {
      sh = ss.insertSheet("Setari");
      sh.appendRow(["Cheie", "Valoare", "UltimaModificare"]);
    }
    setSettingValue(sh, "notifications_config_json", e.parameter.configJson);
    return returnResponse(e, { success: true });
  }

  // 8b. CRUD INVENTAR PERSONAL (Obiecte, spații de depozitare, mentenanță)
  if (action === "getInventoryData") {
    return returnResponse(e, { inventar: getInventoryData(ss) });
  }

  if (action === "addInventoryItem" || action === "updateInventoryItem") {
    let sh = ss.getSheetByName("Inventar");
    if (!sh) {
      sh = ss.insertSheet("Inventar");
      sh.appendRow(["ID", "Denumire", "Categorie", "Spatiu", "Cantitate", "Stare", "Valoare", "DataAchizitie", "UrmatoareaMentenanta", "Observatii", "PozaBase64", "IstoricMutariJson", "DataCreare", "DataModificare"]);
      sh.getRange("A1:N1").setFontWeight("bold");
    }
    const now = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
    const itemId = e.parameter.id || ("INV-" + Date.now());

    if (action === "updateInventoryItem") {
      deleteRowById(sh, itemId, 1);
    }

    sh.appendRow([
      itemId,
      e.parameter.name || "Obiect",
      e.parameter.category || "-",
      e.parameter.space || "-",
      e.parameter.qty || "1",
      e.parameter.status || "Funcțional",
      e.parameter.value || "",
      e.parameter.purchaseDate || "",
      e.parameter.nextMaintenance || "",
      e.parameter.notes || "",
      e.parameter.photoBase64 || "",
      e.parameter.historyJson || "[]",
      action === "addInventoryItem" ? now : (e.parameter.dataCreare || now),
      now
    ]);
    return returnResponse(e, { success: true, id: itemId });
  }

  if (action === "deleteInventoryItem") {
    let sh = ss.getSheetByName("Inventar");
    if (sh && e.parameter.id) deleteRowById(sh, e.parameter.id, 1);
    return returnResponse(e, { success: true });
  }

  // 8c. GÂNDUL ZILEI (tab nou GandulZilei; adăugare/editare/dezactivare — niciodată ștergere de rând)
  if (action === "getGandulZileiData") {
    return returnResponse(e, { gandulZilei: getGandulZileiData(ss) });
  }

  if (action === "addGandulZilei") {
    let sh = getOrCreateGandulZileiSheet(ss);
    const now = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
    const newId = "GZ-" + Date.now();
    sh.appendRow([newId, e.parameter.text || "", e.parameter.autor || "", e.parameter.tema || "Înțelepciune", "DA", now, now]);
    return returnResponse(e, { success: true, id: newId });
  }

  if (action === "updateGandulZilei") {
    let sh = getOrCreateGandulZileiSheet(ss);
    const id = e.parameter.id;
    const maxRows = sh.getLastRow();
    if (sh && id && maxRows >= 2) {
      const ids = sh.getRange(2, 1, maxRows - 1, 1).getValues();
      for (let i = 0; i < ids.length; i++) {
        if ((ids[i][0] || "").toString().trim() === id.toString().trim()) {
          const r = i + 2;
          if (e.parameter.text !== undefined) sh.getRange(r, 2).setValue(e.parameter.text);
          if (e.parameter.autor !== undefined) sh.getRange(r, 3).setValue(e.parameter.autor);
          if (e.parameter.tema !== undefined) sh.getRange(r, 4).setValue(e.parameter.tema);
          if (e.parameter.activ !== undefined) sh.getRange(r, 5).setValue(e.parameter.activ === "true" || e.parameter.activ === true ? "DA" : "NU");
          sh.getRange(r, 7).setValue(Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss"));
          break;
        }
      }
    }
    return returnResponse(e, { success: true });
  }

  // 8d. MODEL CITIRE (Secțiunea 4-5): tab NOU ModelCitire, singurul tab complet nou cerut — populat
  //     idempotent cu cele 3 modele predefinite DOAR dacă foaia e goală la prima utilizare
  if (action === "getModelCitireData") {
    return returnResponse(e, { modelCitire: getModelCitireData(ss) });
  }

  if (action === "setModelCitirePozitieBifat") {
    let sh = getOrCreateModelCitireSheet(ss);
    const rowId = parseInt(e.parameter.rowId, 10);
    const bifat = (e.parameter.bifat === "true" || e.parameter.bifat === true);
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      sh.getRange(rowId, 6).setValue(bifat ? "DA" : "NU");
      sh.getRange(rowId, 7).setValue(bifat ? Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss") : "");
    }
    return returnResponse(e, { success: true });
  }

  // 8e. EDITARE NOTE BIBLIA ANANIA — actualizează doar câmpurile trimise, completează DataModificare
  if (action === "updateNoteAnania") {
    let sh = ss.getSheetByName("noteBIBLIA_Anania") || ss.getSheetByName("Biblia_Note");
    const rowId = parseInt(e.parameter.rowId, 10);
    if (sh && rowId >= 2 && rowId <= sh.getLastRow()) {
      const cols = ensureSheetColumns(sh, ["Sters", "DataStergere", "DataModificare"]);
      if (e.parameter.textVerset !== undefined) sh.getRange(rowId, 6).setValue(e.parameter.textVerset);
      if (e.parameter.context !== undefined) sh.getRange(rowId, 7).setValue(e.parameter.context);
      if (e.parameter.nota !== undefined) sh.getRange(rowId, 8).setValue(e.parameter.nota);
      if (e.parameter.derventUrl !== undefined) sh.getRange(rowId, 9).setValue(e.parameter.derventUrl);
      if (e.parameter.capitolProgres !== undefined) sh.getRange(rowId, 10).setValue(e.parameter.capitolProgres);
      sh.getRange(rowId, cols["DataModificare"]).setValue(Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss"));
    }
    return returnResponse(e, { success: true });
  }

  // 8f. ȘTERGERE SOFT NOTE BIBLIA ANANIA (ca la Notițe: marcaj Sters=DA/NU, nu elimină rânduri)
  if (action === "softDeleteNoteAnania" || action === "restoreNoteAnania") {
    let sh = ss.getSheetByName("noteBIBLIA_Anania") || ss.getSheetByName("Biblia_Note");
    if (!sh) return returnResponse(e, { success: true });
    const cols = ensureSheetColumns(sh, ["Sters", "DataStergere"]);
    const isDelete = (action === "softDeleteNoteAnania");
    const timestamp = isDelete ? Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss") : "";
    const ids = (e.parameter.rowIds || e.parameter.rowId || "").toString().split(",").map(x => parseInt(x.trim(), 10)).filter(n => !isNaN(n));
    ids.forEach(rId => {
      if (rId >= 2 && rId <= sh.getLastRow()) {
        sh.getRange(rId, cols["Sters"]).setValue(isDelete ? "DA" : "NU");
        sh.getRange(rId, cols["DataStergere"]).setValue(timestamp);
      }
    });
    return returnResponse(e, { success: true, count: ids.length });
  }

  // Populare inițială unică: scrie citatele trimise doar dacă foaia GandulZilei este încă goală (nu suprascrie date reale)
  if (action === "seedGandulZileiIfEmpty") {
    let sh = getOrCreateGandulZileiSheet(ss);
    if (sh.getLastRow() < 2) {
      const items = JSON.parse(e.parameter.itemsJson || "[]");
      const now = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
      const rows = items.map((it, idx) => ["GZ-SEED-" + idx, it.quote || it.text || "", it.author || it.autor || "", it.tema || "Înțelepciune", "DA", now, now]);
      if (rows.length > 0) sh.getRange(2, 1, rows.length, 7).setValues(rows);
      return returnResponse(e, { success: true, seeded: rows.length });
    }
    return returnResponse(e, { success: true, seeded: 0 });
  }

  // 9. PRELUARE COMPLETĂ DATE PENTRU APLICAȚIE (RĂSPUNS 1:1)
  if (action === "getLiveData") {
    // Sincronizare ușoară/rapidă (fără Pomelnic 990 rânduri și fără Biblia 360) - folosită la polling frecvent
    const notesList = getGanduriData(ss);
    const liveData = {
      settings: getSettingsData(ss),
      settingsRaw: getAllSettingsRaw(ss),
      cumparaturi: getCumparaturiData(ss),
      ganduri: notesList,
      notite: notesList,
      gandulZilei: getGandulZileiData(ss),
      modelCitire: getModelCitireData(ss),
      schoolActivities: getSchoolActivitiesData(ss),
      psaltireGrup: getPsaltireGrupData(ss)
    };
    return returnResponse(e, liveData);
  }

  const notesList = getGanduriData(ss);
  const responseData = {
    settings: getSettingsData(ss),
    settingsRaw: getAllSettingsRaw(ss),
    pomelnic: getPomelnicData(ss),
    cumparaturi: getCumparaturiData(ss),
    ganduri: notesList,
    notite: notesList,
    gandulZilei: getGandulZileiData(ss),
    modelCitire: getModelCitireData(ss),
    noteAnania: getNoteAnaniaData(ss),
    bibliaBVA: getBiblia360Data(ss),
    schoolActivities: getSchoolActivitiesData(ss),
    psaltireGrup: getPsaltireGrupData(ss)
  };

  return returnResponse(e, responseData);
}

function returnResponse(e, data) {
  const callback = e.parameter.callback;
  if (callback) {
    if (callback === "void") {
      return ContentService.createTextOutput("/* void */")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(`${callback}(${JSON.stringify(data)})`)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function setSettingValue(sheet, key, value) {
  if (value === undefined || value === null) return;
  const maxRows = sheet.getLastRow();
  const timestamp = Utilities.formatDate(new Date(), "Europe/Bucharest", "yyyy-MM-dd HH:mm:ss");
  if (maxRows >= 2) {
    const keys = sheet.getRange(2, 1, maxRows - 1, 1).getValues();
    for (let i = 0; i < keys.length; i++) {
      if (keys[i][0] && keys[i][0].toString().trim() === key) {
        sheet.getRange(i + 2, 2, 1, 2).setValues([[value, timestamp]]);
        return;
      }
    }
  }
  sheet.appendRow([key, value, timestamp]);
}

// Helper generic: citește header-ul unei foi, creează (idempotent) coloanele lipsă din listă la finalul foii
// fără să atingă coloanele existente. Returnează un obiect { numeColoana: indexColoana(1-based) }.
function ensureSheetColumns(sheet, columnNames) {
  const lastCol = Math.max(sheet.getLastColumn(), 1);
  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => (h || "").toString().trim());
  const result = {};
  let nextCol = lastCol;
  columnNames.forEach(name => {
    const idx = headers.indexOf(name);
    if (idx !== -1) {
      result[name] = idx + 1;
    } else {
      nextCol += 1;
      sheet.getRange(1, nextCol).setValue(name);
      headers.push(name);
      result[name] = nextCol;
    }
  });
  return result;
}

function getSchoolActivitiesData(ss) {
  let sh = ss.getSheetByName("Scoala_Activitati") || ss.getSheetByName("Activitati_Scoala");
  if (!sh || sh.getLastRow() < 2) return [];
  const cols = ensureSheetColumns(sh, ["Data_Achitat", "Comentariu_Plata"]);
  const maxCol = Math.max(14, cols["Data_Achitat"], cols["Comentariu_Plata"]);
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, maxCol).getValues();
  return rows.map(r => {
    let dateVal = r[1];
    let dateStr = "";
    if (dateVal instanceof Date) {
      dateStr = Utilities.formatDate(dateVal, "Europe/Bucharest", "yyyy-MM-dd");
    } else if (typeof dateVal === "string") {
      dateStr = dateVal.substring(0, 10);
    }
    const meta = parseJsonSafe(r[12]);
    const isMetaObj = meta && typeof meta === "object" && !Array.isArray(meta);
    const reminders = isMetaObj ? (meta.reminders || [60, 30, 10]) : (Array.isArray(meta) ? meta : [60, 30, 10]);
    const color = isMetaObj ? (meta.color || "auto") : "auto";
    const startHour = isMetaObj ? (meta.startHour || "08:00") : "08:00";
    const endHour = isMetaObj ? (meta.endHour || "16:00") : "16:00";
    const author = isMetaObj ? (meta.author || "Bogdan") : "Bogdan";

    return {
      id: r[0],
      date: dateStr,
      child: r[2] || "Copil",
      institution: r[3] || "-",
      title: r[4] || "Activitate",
      paymentType: r[5] || "Pe oră",
      hours: parseFloat(r[6]) || 1,
      rate: parseFloat(r[7]) || 0,
      total: parseFloat(r[8]) || 0,
      paymentStatus: r[9] || "În așteptare",
      deadline: r[10] ? r[10].toString().substring(0, 10) : "-",
      dayMessage: r[11] || "",
      reminders: reminders,
      color: color,
      startHour: startHour,
      endHour: endHour,
      author: author,
      dataAchitat: r[cols["Data_Achitat"] - 1] ? parseDateToText(r[cols["Data_Achitat"] - 1]) : "",
      comentariuPlata: r[cols["Comentariu_Plata"] - 1] || ""
    };
  });
}

function parseJsonSafe(val) {
  try { return JSON.parse(val || "[]"); } catch (e) { return []; }
}

function getCumparaturiData(ss) {
  const sh = ss.getSheetByName("lista_cumparaturi");
  if (!sh || sh.getLastRow() < 2) return [];
  const maxRows = sh.getLastRow();
  const maxCols = Math.min(Math.max(sh.getLastColumn(), 3), 7);
  const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
  const hasIdCol = (header === "ID");
  const rows = sh.getRange(2, 1, maxRows - 1, maxCols).getValues();
  return rows.map((r, i) => {
    let id = hasIdCol ? (r[0] || ("SHOP-" + (i + 2))) : ("SHOP-" + (i + 2));
    let magazin = hasIdCol ? (r[1] || "General") : (r[0] || "General");
    let produs = hasIdCol ? (r[2] || "") : (r[1] || "");
    let cantitate = hasIdCol ? (r[3] || "1") : (r[2] || "1");
    let nuPartajaVal = hasIdCol ? (r[5] || "") : (r[3] || "");
    let nuPartaja = (nuPartajaVal === true || nuPartajaVal.toString().trim().toUpperCase() === "DA" || nuPartajaVal.toString().trim().toLowerCase() === "true");
    let owner = hasIdCol ? (r[6] || "").toString().trim() : (r[4] || "").toString().trim();
    return {
      id: id.toString().trim(),
      rowId: i + 2,
      magazin: magazin.toString().trim(),
      produs: produs.toString().trim(),
      cantitate: cantitate.toString().trim(),
      nuPartaja: nuPartaja,
      owner: owner
    };
  }).filter(x => x.produs !== "");
}

function getGanduriData(ss) {
  const sh = ss.getSheetByName("Notite") || ss.getSheetByName("Ganduri");
  if (!sh || sh.getLastRow() < 2) return [];
  const maxRows = sh.getLastRow();
  const maxCols = Math.min(Math.max(sh.getLastColumn(), 3), 8);
  const header = (sh.getRange(1, 1).getValue() || "").toString().trim().toUpperCase();
  const hasIdCol = (header === "ID");
  const cols = getNoteStersColumns(sh);
  const rows = sh.getRange(2, 1, maxRows - 1, Math.max(maxCols, cols.stersCol)).getValues();
  return rows.map((r, i) => {
    let gand = hasIdCol ? (r[1] || "") : (r[0] || "");
    let sursa = hasIdCol ? (r[2] || "Personal") : (r[1] || "Personal");
    let dataVal = hasIdCol ? r[3] : r[2];
    let user = hasIdCol ? (r[4] || "").toString().trim() : (r[3] || "").toString().trim();
    let sters = (r[cols.stersCol - 1] || "").toString().trim().toUpperCase() === "DA";
    return {
      rowId: i + 2,
      gand: gand.toString().trim(),
      nota: gand.toString().trim(),
      sursa: sursa.toString().trim(),
      data: dataVal ? parseDateToText(dataVal) : "",
      user: user,
      sters: sters
    };
  }).filter(x => x.gand !== "" && !x.sters);
}

// Returnează indicii coloanelor Sters / DataStergere pentru foaia Notite/Ganduri, creându-le idempotent dacă lipsesc
function getNoteStersColumns(sheet) {
  const lastCol = Math.max(sheet.getLastColumn(), 1);
  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => (h || "").toString().trim().toUpperCase());
  let stersCol = headers.indexOf("STERS") + 1;
  let dataStergereCol = headers.indexOf("DATASTERGERE") + 1;
  if (stersCol === 0) {
    stersCol = lastCol + 1;
    sheet.getRange(1, stersCol).setValue("Sters");
  }
  if (dataStergereCol === 0) {
    dataStergereCol = Math.max(lastCol, stersCol) + 1;
    sheet.getRange(1, dataStergereCol).setValue("DataStergere");
  }
  return { stersCol: stersCol, dataStergereCol: dataStergereCol };
}

function getNoteAnaniaData(ss) {
  const sh = ss.getSheetByName("noteBIBLIA_Anania") || ss.getSheetByName("Biblia_Note");
  if (!sh || sh.getLastRow() < 2) return [];
  // Sters/DataStergere/DataModificare: coloane adăugate idempotent la finalul foii
  // (pentru ștergere cu confirmare și editare, ca la Notițe)
  const stersCols = ensureSheetColumns(sh, ["Sters", "DataStergere", "DataModificare"]);
  const maxCols = Math.max(Math.min(Math.max(sh.getLastColumn(), 4), 10), stersCols["Sters"], stersCols["DataStergere"], stersCols["DataModificare"]);
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, maxCols).getValues();

  // Structură coloane registru (10 coloane conform foii noteBIBLIA_Anania):
  // Col A (0): Colecție (ex: Vechiul Testament)
  // Col B (1): Carte (ex: Facerea)
  // Col C (2): Capitol (ex: Capitol 1)
  // Col D (3): Verset (ex: 1)
  // Col E (4): Marcaj (ex: (A))
  // Col F (5): Text verset complet (ex: 1Întru'nceput[A]a făcut[B]Dumnezeu cerul și pământul[C].)
  // Col G (6): Context / fragment adnotat (ex: 1Întru'nceput)
  // Col H (7): Nota de subsol (ex: [Nota A] În interiorul...)
  // Col I (8): URL sursă (ex: https://biblia.dervent.ro/VT-Fc-01)
  // Col J (9): Capitol # / # (ex: 1/50) — progres capitol curent/total din carte
  return rows.map((r, idx) => {
    let colectie = "", cartea = "", capitol = "", versetul = "", marcaj = "", textVerset = "", context = "", nota = "", derventUrl = "", capitolProgres = "";

    if (maxCols >= 8) {
      colectie = (r[0] || "Vechiul Testament").toString().trim();
      cartea = (r[1] || "").toString().trim();
      capitol = (r[2] || "").toString().trim();
      versetul = (r[3] || "").toString().trim();
      marcaj = (r[4] || "").toString().trim();
      textVerset = (r[5] || "").toString().trim();
      context = (r[6] || "").toString().trim();
      nota = (r[7] || "").toString().trim();
      derventUrl = (r[8] || "").toString().trim();
      capitolProgres = (r[9] || "").toString().trim();
    } else {
      cartea = (r[0] || "").toString().trim();
      versetul = (r[1] || "").toString().trim();
      marcaj = (r[2] || "").toString().trim();
      nota = (r[3] || "").toString().trim();
      if (maxCols >= 5 && r[4]) textVerset = (r[4] || "").toString().trim();
    }

    if (!cartea && !nota) return null;
    const sters = (r[stersCols["Sters"] - 1] || "").toString().trim().toUpperCase() === "DA";
    if (sters) return null;

    return {
      id: idx + 1,
      rowId: idx + 2,
      colectie: colectie,
      cartea: cartea,
      capitol: capitol,
      versetul: versetul,
      marcaj: marcaj,
      textVerset: textVerset,
      context: context,
      nota: nota,
      derventUrl: derventUrl,
      capitolProgres: capitolProgres,
      dataModificare: r[stersCols["DataModificare"] - 1] ? parseDateToText(r[stersCols["DataModificare"] - 1]) : "",
      // Păstrăm și denumirile istorice pentru compatibilitate inversă totală:
      indice: textVerset || marcaj,
      explicatie: context || nota,
      moment: nota
    };
  }).filter(x => x !== null);
}

function deleteRowById(sheet, targetVal, colIndex) {
  const maxRows = sheet.getLastRow();
  if (maxRows < 2) return;
  const values = sheet.getRange(2, colIndex, maxRows - 1, 1).getValues();
  for (let i = values.length - 1; i >= 0; i--) {
    if (values[i][0] && values[i][0].toString().trim() === targetVal.toString().trim()) {
      sheet.deleteRow(i + 2);
    }
  }
}

function parseDateToText(val) {
  if (!val) return "";
  if (Object.prototype.toString.call(val) === '[object Date]') {
    return Utilities.formatDate(val, "Europe/Bucharest", "dd.MM.yyyy");
  }
  return val.toString().trim();
}

function parseDateToISO(val) {
  if (!val) return "";
  if (Object.prototype.toString.call(val) === '[object Date]') {
    return Utilities.formatDate(val, "Europe/Bucharest", "yyyy-MM-dd");
  }
  return val.toString().trim().substring(0, 10);
}

/**
 * ====== STOCARE FIȘIERE ÎN GOOGLE DRIVE (imagini: calendar școlar, iconițe/logo aplicație) ======
 * Imaginile NU se salvează niciodată în celule — doar ca fișiere în acest folder dedicat,
 * iar în Setari se păstrează DOAR ID-ul fișierului (ex. CalendarScolar_2026-2027 -> fileId).
 */
function getOrCreateAppDriveFolder(ss) {
  const sh = ss.getSheetByName("Setari");
  const FOLDER_KEY = "drive_folder_id";
  if (sh) {
    const rows = sh.getLastRow() >= 2 ? sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues() : [];
    for (let i = 0; i < rows.length; i++) {
      if ((rows[i][0] || "").toString().trim() === FOLDER_KEY && rows[i][1]) {
        try {
          return DriveApp.getFolderById(rows[i][1].toString().trim());
        } catch (e) { /* folder ID invalid/șters — recreăm mai jos */ }
      }
    }
  }
  const folder = DriveApp.createFolder("DIiH_App_Fisiere");
  const settingsSheet = sh || ss.insertSheet("Setari");
  if (!sh) {
    settingsSheet.appendRow(["Cheie", "Valoare", "UltimaModificare"]);
    settingsSheet.getRange("A1:C1").setFontWeight("bold");
  }
  setSettingValue(settingsSheet, FOLDER_KEY, folder.getId());
  return folder;
}

/**
 * Primește fișiere binare (imagini) prin POST, fiindcă URL-urile GET au o limită de lungime
 * prea mică pentru poze reale. Token-ul se verifică la fel ca la doGet.
 */
function doPost(e) {
  const token = e.parameter.token;
  if (token !== API_SECURITY_TOKEN) {
    return ContentService.createTextOutput(JSON.stringify({ error: "Acces neautorizat!" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const ss = getSpreadsheet();
  const action = e.parameter.action;

  if (action === "uploadDriveFile") {
    try {
      const folder = getOrCreateAppDriveFolder(ss);
      const bytes = Utilities.base64Decode(e.parameter.base64Data);
      const blob = Utilities.newBlob(bytes, e.parameter.mimeType || "image/png", e.parameter.fileName || ("file_" + Date.now()));
      const file = folder.createFile(blob);
      return ContentService.createTextOutput(JSON.stringify({ success: true, fileId: file.getId() }))
        .setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.message || String(err) }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }

  return ContentService.createTextOutput(JSON.stringify({ error: "Acțiune POST necunoscută" }))
    .setMimeType(ContentService.MimeType.JSON);
}

// Returnează conținutul unui fișier din Drive ca base64, pentru afișare în <img> cu data: URI
// (evită partajarea publică a fișierului; accesul rămâne protejat de tokenul aplicației)
function getDriveImageAsBase64(ss, fileId) {
  try {
    const file = DriveApp.getFileById(fileId);
    const blob = file.getBlob();
    return {
      success: true,
      mimeType: blob.getContentType(),
      base64: Utilities.base64Encode(blob.getBytes())
    };
  } catch (e) {
    return { success: false, error: e.message || String(e) };
  }
}
